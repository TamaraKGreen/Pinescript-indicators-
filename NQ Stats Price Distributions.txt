// by one constant set measured from 15 years of NQ 1-minute data,
// 2010-06-06 to 2025-03-14, 4,974,698 bars over 3,804 sessions.
//
// STORAGE. Every grid is held in SIGMA units, so a level in percent is
// just sigma(anchor) x grid value. One source of truth: the volatility
// multiplier is frozen at each anchor's open, so a level cannot move
// under you mid-session and the chart, gauge and table are all computed

//
// LAYOUT. 32 anchors: index 0-23 are hours of the NY clock, then 24 Asia,
// 25 London, 26 NY AM, 27 NY PM, 28 Day, 29 Week, 30 RTH, 31 NY full.

// is percentile 5(i+1). S_ONE and S_FADE hold 4 values per anchor at
// k = 0.5 / 1.0 / 1.5 / 2.0.
//
// SAMPLE SIZES. Hours and sessions rest on 3,544-3,795 observations each.
// Week has only 770, so its outer percentiles rest on a few dozen tail
// observations. Hour 17 has 26 because of the maintenance break and is
// blank in every table.
//
// REGENERATING. pipeline.py + generate.py rebuild every constant here
// from the raw minute CSV and reproduce the shipped tables to within
// 0.05%, which is the rounding at 4 significant figures.
//
// The statistics themselves - what the two distributions mean, why the
// textbook 68% is wrong, and the difference between the unconditional
// and conditional probabilities - are explained in the tooltips on the
// inputs they affect, and at length in the script description.
// ==================================================================

// The stdev must be evaluated ON the daily series. Computing it on the chart
// series makes close[1] the previous CHART bar's daily close, which is the same
// day, so the returns are ~0 and the multiplier pins to its floor.
cs = request.security(syminfo.tickerid, "D",
     ta.stdev((close - close[1]) / close[1], i_slen) * 100,
     lookahead=barmerge.lookahead_off)

// Every sigma lookup takes an explicit multiplier. Anchors store the value
// that was live when they opened and pass that, so a level drawn at the open
// keeps its price for the whole session and the table, the gauge ticks and
// the chart lines are computed from one number.
f_sigv(a, vm) => array.get(SIG, a) * vm
f_mfev(a, q, vm) => f_sigv(a, vm) * array.get(MFE, a * NQ + q)
f_maev(a, q, vm) => f_sigv(a, vm) * array.get(MAE, a * NQ + q)
f_rngv(a, q, vm) => f_sigv(a, vm) * array.get(RNG, a * NQ + q)

// ==================================================================
// PERCENTILE RANK against the 19-point grid
// ==================================================================
f_rank(grid, a, x) =>
    base = a * NQ
    lo0  = array.get(grid, base)
    hiN  = array.get(grid, base + NQ - 1)
    r = 0.0
    if na(x) or x < 0
        r := 0.0
    else if lo0 > 0 and x <= lo0
        // Seven grids have an exact zero at P5; for those this branch cannot
        // fire and the loop below handles the interval from zero correctly.
        r := 5.0 * x / lo0
    else if x >= hiN
        r := math.min(99.0, 95.0 + 4.0 * (x - hiN) / hiN)
    else
        for i = 0 to NQ - 2
            a1 = array.get(grid, base + i)
            b1 = array.get(grid, base + i + 1)
            if x >= a1 and x <= b1 and b1 > a1
                r := 5.0 * (i + 1) + 5.0 * (x - a1) / (b1 - a1)
                break
    r

// P(anchor closes below +k SD), unconditional, interpolated across the table.
f_below(a, k) =>
    ak = math.abs(k)
    b  = a * 4
    p  = 0.0
    if ak <= 0.5
        p := 50.0 + (array.get(ONE, b) - 50.0) * (ak / 0.5)
    else if ak <= 1.0
        p := array.get(ONE, b)   + (array.get(ONE, b+1) - array.get(ONE, b))   * ((ak-0.5)/0.5)
    else if ak <= 1.5
        p := array.get(ONE, b+1) + (array.get(ONE, b+2) - array.get(ONE, b+1)) * ((ak-1.0)/0.5)
    else if ak <= 2.0
        p := array.get(ONE, b+2) + (array.get(ONE, b+3) - array.get(ONE, b+2)) * ((ak-1.5)/0.5)
    else
        p := array.get(ONE, b+3)
    p

// P(closes back inside | tagged), interpolated across the same k grid as
// f_below. Held flat outside 0.5-2.0 rather than extrapolated: the measured
// series has no trend in k, so a slope fitted to its noise would be fiction.
// Returns na where the cell is unmeasured, so callers render "-" not 0.
f_fade(a, k) =>
    ak = math.abs(k)
    b  = a * 4
    v0 = array.get(FADE, b)
    v1 = array.get(FADE, b+1)
    v2 = array.get(FADE, b+2)
    v3 = array.get(FADE, b+3)
    p  = float(na)
    if ak <= 0.5
        p := v0
    else if ak <= 1.0
        p := v0 > 0 and v1 > 0 ? v0 + (v1 - v0) * ((ak-0.5)/0.5) : float(na)
    else if ak <= 1.5
        p := v1 > 0 and v2 > 0 ? v1 + (v2 - v1) * ((ak-1.0)/0.5) : float(na)
    else if ak <= 2.0
        p := v2 > 0 and v3 > 0 ? v2 + (v3 - v2) * ((ak-1.5)/0.5) : float(na)
    else
        p := v3
    na(p) or p <= 0 ? float(na) : p

// The text under a gauge column, and in the table's probability cell.
f_probTxt(a, zc) =>
    s = ""
    if i_g_prob == "P fade (conditional)"
        f = f_fade(a, zc)
        s := na(f) ? "-" : str.format("{0,number,#}", f) + "%"
    else if i_g_prob == "P close below (unconditional)"
        p = f_below(a, zc)
        s := str.format("{0,number,#}", zc >= 0 ? p : 100 - p) + "%"
    s

            f_lvAdd(v, o * (1 - f_maev(a,Q75,vm)/100), f_lvTxt(v.tag+" MAE75", f_maev(a,Q75,vm), o), c_mae, line.style_dashed)

f_lvBar(v, isIn, wasIn) =>
    if not isIn and wasIn
        // Retained levels simply stop being extended, so they freeze at the bar
        // the period closed on. The next open clears them before redrawing.
        if not i_lv_keep
            f_lvClear(v)
        v.stale := true
    if isIn and not wasIn
        v.o  := open
        v.h  := high
        v.l  := low
        v.vm := volLive          // frozen here, for the whole session
        v.short := earlyClose and f_hitByEarly(v.anchor)
        v.stale := false
        v.bars  := 1
        f_lvClear(v)
        suppress = v.short and i_hd_mode == "Hide affected levels"
        if v.show and i_lv_on and not suppress
            f_lvDraw(v)
    else if isIn and not na(v.o)
        // Only extends while the session is actually live, so nothing bleeds
        // across the 17:00-19:00 window when no session is running.
        v.h := math.max(v.h, high)
        v.l := math.min(v.l, low)
        v.bars := v.bars + 1
    if isIn and v.show and i_lv_on and not na(v.o)
        f_lvExtend(v)

var array<box>   GB = array.new<box>()
var array<label> GL = array.new<label>()
var array<line>  GN = array.new<line>()

f_gWipe() =>
    for b in GB
        box.delete(b)
    array.clear(GB)
    for l in GL
        label.delete(l)
    array.clear(GL)
    for l in GN
        line.delete(l)
    array.clear(GN)

// Reference excursion, in sigma units, that counts as a full tank.
f_zRef(a, up) =>
    switch i_g_full
        "MFE/MAE P75" => array.get(up ? MFE : MAE, a * NQ + Q75)
        "MFE/MAE P90" => array.get(up ? MFE : MAE, a * NQ + Q90)
        "MFE/MAE P95" => array.get(up ? MFE : MAE, a * NQ + Q95)
        "1.0 SD"      => 1.0
        "1.5 SD"      => 1.5
        => 2.0

// Map an excursion in sigma units to a vertical offset in price.
//   Range fill : normalise by the reference, so the tank tops out at 100%.
//   otherwise  : plot true sigma, clamped to the scale extent.
f_yoff(z, zRef, fill, pctScale, grid, a, unitPx) =>
    if not fill
        math.min(z, i_g_kmax) * unitPx
    else if pctScale
        math.min(f_rank(grid, a, z) / 100.0, 1.0) * i_g_kmax * unitPx
    else
        math.min(z / zRef, 1.0) * i_g_kmax * unitPx

f_gCol(slot, a, o, hi, lo, vm, tag, baseY, unitPx) =>
    if not na(o) and o > 0 and a >= 0 and f_sigv(a, vm) > 0 and not na(hi) and not na(lo)
        fill  = i_g_align == "Range fill"
        pctSc = i_g_full == "Percentile 0-100"
        sigPx = f_sigv(a, vm) / 100.0 * o
        x  = bar_index + i_g_gap + slot * i_g_space
        xl = x - i_g_wid
        xr = x + i_g_wid
        showSD  = i_mode != "Percentiles"
        showPct = i_mode != "Std Dev"

        // The tank axis and the SD ticks both want the left gutter. Give the SD
        // labels a wider berth when both are on, or they overprint each other.
        axX = xl - i_g_wid - 2
        sdX = fill and i_g_tankax and slot == 0 ? axX - 2 * i_g_wid - 6 : axX

        zu = (hi - o) / sigPx
        zd = (o - lo) / sigPx
        zc = (close - o) / sigPx
        ru = f_rank(MFE, a, zu)
        rd = f_rank(MAE, a, zd)
        zRu = f_zRef(a, true)
        zRd = f_zRef(a, false)

        yMid = baseY
        yT   = yMid + i_g_kmax * unitPx
        yB   = yMid - i_g_kmax * unitPx
        yU   = yMid + f_yoff(zu, zRu, fill, pctSc, MFE, a, unitPx)
        yD   = yMid - f_yoff(zd, zRd, fill, pctSc, MAE, a, unitPx)

        // The tank: an empty outline the fill sits inside, so an unfilled
        // column reads as unfilled rather than as missing data.
        array.push(GB, box.new(xl, yT, xr, yB, bgcolor=color.new(color.gray, 92),
             border_color=c_ax, border_width=1, border_style=line.style_dotted))
        array.push(GB, box.new(xl, yU, xr, yMid, bgcolor=c_bup, border_width=0))
        array.push(GB, box.new(xl, yMid, xr, yD, bgcolor=c_bdn, border_width=0))
        // Open line.
        array.push(GN, line.new(xl, yMid, xr, yMid, color=c_ax, style=line.style_solid, width=1))
        // Price NOW. The bar top is the HIGH, which may sit well above this after a
        // pullback - that gap is the point of the gauge, so draw it boldly and wider
        // than the column so it cannot be mistaken for a level tick.
        yC = yMid + (zc >= 0 ? f_yoff(zc, zRu, fill, pctSc, MFE, a, unitPx)
                             : -f_yoff(-zc, zRd, fill, pctSc, MAE, a, unitPx))
        array.push(GN, line.new(xl - 1, yC, xr + 1, yC, color=c_now,
             style=line.style_solid, width=3))
        if i_g_nowlab
            array.push(GL, label.new(x, yC, "now", style=label.style_none,
                 textcolor=c_now, size=f_sz(i_sz_tick),
                 color=color.new(color.white, 100), text_font_family=font.family_monospace))

        if i_g_hdr
            array.push(GL, label.new(x, yT + 0.14 * i_g_kmax * unitPx, tag,
                 style=label.style_none, textcolor=c_hdr, size=f_sz(i_sz_hdr),
                 color=color.new(color.white, 100), text_font_family=font.family_monospace))

        // pNN ranks, and how much of the tank each side has consumed
        if showPct or fill
            uTxt = "p" + str.tostring(int(math.round(ru)))
            dTxt = "p" + str.tostring(int(math.round(rd)))
            if i_g_fillpc and fill and not pctSc
                // Under Percentile 0-100 the fill percent IS the pNN already, so
                // printing both would just repeat the same number.
                uTxt := uTxt + "  " + str.tostring(int(math.round(math.min(zu / zRu, 1.0) * 100))) + "%"
                dTxt := dTxt + "  " + str.tostring(int(math.round(math.min(zd / zRd, 1.0) * 100))) + "%"
            array.push(GL, label.new(x, yU, uTxt, style=label.style_none, textcolor=c_tx,
                 size=f_sz(i_sz_pct), color=color.new(color.white, 100),
                 text_font_family=font.family_monospace))
            array.push(GL, label.new(x, yD, dTxt, style=label.style_none, textcolor=c_tx,
                 size=f_sz(i_sz_pct), color=color.new(color.white, 100),
                 text_font_family=font.family_monospace))

        // Under Range fill each column normalises to its own range, so a given SD
        // sits at a different height in every column. Sharing one label is wrong,
        // so "First column" is promoted to per-column rather than dropped.
        sdLab = fill ? i_g_sdlab != "None"
                     : i_g_sdlab == "Every column" or (i_g_sdlab == "First column" and slot == 0)
        ptLab = i_g_ptlab == "Every column" or (i_g_ptlab == "First column" and slot == 0)

        // SD gridlines. In Range fill these land at their normalised positions,
        // so they still say where 1 SD sits inside this anchor's own range.
        if showSD
            for t = 1 to 4
                k = t * 0.5
                yKu = f_yoff(k, zRu, fill, pctSc, MFE, a, unitPx)
                yKd = f_yoff(k, zRd, fill, pctSc, MAE, a, unitPx)
                if yKu < i_g_kmax * unitPx
                    array.push(GN, line.new(xl, yMid + yKu, xr, yMid + yKu,
                         color=c_ax, style=line.style_dotted, width=1))
                    if sdLab
                        array.push(GL, label.new(sdX, yMid + yKu,
                             "+" + str.format("{0,number,#.#}", k) + f_tickPts(k, sigPx),
                             style=label.style_none, textcolor=c_ax, size=f_sz(i_sz_tick),
                             color=color.new(color.white, 100), text_font_family=font.family_monospace))
                if yKd < i_g_kmax * unitPx
                    array.push(GN, line.new(xl, yMid - yKd, xr, yMid - yKd,
                         color=c_ax, style=line.style_dotted, width=1))
                    if sdLab
                        array.push(GL, label.new(sdX, yMid - yKd,
                             "-" + str.format("{0,number,#.#}", k) + f_tickPts(k, sigPx),
                             style=label.style_none, textcolor=c_ax, size=f_sz(i_sz_tick),
                             color=color.new(color.white, 100), text_font_family=font.family_monospace))

        // The anchor's own MFE/MAE levels, at their positions inside the tank.
        if showPct
            for qi = 0 to 4
                q  = qi == 0 ? Q25 : qi == 1 ? Q50 : qi == 2 ? Q75 : qi == 3 ? Q90 : Q95
                on = qi == 0 ? i_p25 : qi == 1 ? i_p50 : qi == 2 ? i_p75 : qi == 3 ? i_p90 : i_p95
                pl = qi == 0 ? "25" : qi == 1 ? "50" : qi == 2 ? "75" : qi == 3 ? "90" : "95"
                if on
                    zU = array.get(MFE, a * NQ + q)
                    zD = array.get(MAE, a * NQ + q)
                    yQu = f_yoff(zU, zRu, fill, pctSc, MFE, a, unitPx)
                    yQd = f_yoff(zD, zRd, fill, pctSc, MAE, a, unitPx)
                    if yQu <= i_g_kmax * unitPx
                        array.push(GN, line.new(xl, yMid + yQu, xr, yMid + yQu,
                             color=color.new(c_mfe, 25), style=line.style_solid, width=1))
                        if ptLab
                            array.push(GL, label.new(xr + i_g_wid + 2, yMid + yQu,
                                 "MFE" + pl + f_tickPts(zU, sigPx), style=label.style_none,
                                 textcolor=color.new(c_mfe, 10), size=f_sz(i_sz_tick),
                                 color=color.new(color.white, 100), text_font_family=font.family_monospace))
                    if yQd <= i_g_kmax * unitPx
                        array.push(GN, line.new(xl, yMid - yQd, xr, yMid - yQd,
                             color=color.new(c_mae, 25), style=line.style_solid, width=1))
                        if ptLab
                            array.push(GL, label.new(xr + i_g_wid + 2, yMid - yQd,
                                 "MAE" + pl + f_tickPts(zD, sigPx), style=label.style_none,
                                 textcolor=color.new(c_mae, 10), size=f_sz(i_sz_tick),
                                 color=color.new(color.white, 100), text_font_family=font.family_monospace))

        // Percent-of-tank axis. Valid to share because the tank height is common
        // to all columns, unlike the SD scale under normalisation.
        if fill and i_g_tankax and slot == 0
            for t = 1 to 4
                fr = t * 0.25
                yf = fr * i_g_kmax * unitPx
                array.push(GL, label.new(axX, yMid + yf,
                     str.tostring(int(fr * 100)) + "%", style=label.style_none,
                     textcolor=c_ax, size=f_sz(i_sz_tick),
                     color=color.new(color.white, 100), text_font_family=font.family_monospace))
                array.push(GL, label.new(axX, yMid - yf,
                     str.tostring(int(fr * 100)) + "%", style=label.style_none,
                     textcolor=c_ax, size=f_sz(i_sz_tick),
                     color=color.new(color.white, 100), text_font_family=font.family_monospace))

        if i_g_prob != "None"
            array.push(GL, label.new(x, yB - 0.22 * i_g_kmax * unitPx, f_probTxt(a, zc),
                 style=label.style_none, textcolor=c_tx, size=f_sz(i_sz_tick),
                 color=color.new(color.white, 100), text_font_family=font.family_monospace))

f_gTag(v) => str.upper(v.tag) + (v.stale ? "*" : "") + (v.short ? " !" : "")

// Evaluated on every bar, not inside the barstate.islast block below - ta.*
// functions need a continuous series to keep their history correct.
gFitRange = ta.highest(high, i_g_look) - ta.lowest(low, i_g_look)

if barstate.islast
    f_gWipe()
    if i_g_on and isNQ and tfOK
        // Range fill and Common baseline both share one centre and one pixel
        // height; only Own open staggers the columns.
        common = i_g_align != "Own open"
        // Total tank height is 2 * kmax * unit, so solving for a target height
        // of gFitRange * fit% gives the unit below. Own open never reaches here
        // - it passes its own sigma-derived unit per column instead.
        useFit = i_g_hmode == "Fit to recent range" and not na(gFitRange) and gFitRange > 0
        unit   = useFit ? gFitRange * (i_g_fit / 100.0) / (2.0 * i_g_kmax)
             : f_sigv(I_DAY, LV_DAY.vm) / 100.0 * close * i_g_height
        slot = 0
        if i_g_hour
            f_gCol(slot, LV_HOUR.anchor, LV_HOUR.o, LV_HOUR.h, LV_HOUR.l, LV_HOUR.vm, f_gTag(LV_HOUR),
                 common ? close : nz(LV_HOUR.o, close),
                 common ? unit : f_sigv(LV_HOUR.anchor, LV_HOUR.vm)/100.0*nz(LV_HOUR.o, close) * i_g_height)
            slot += 1
        if i_g_sess
            sv = f_gaugeSess()
            f_gCol(slot, sv.anchor, sv.o, sv.h, sv.l, sv.vm, f_gTag(sv),
                 common ? close : nz(sv.o, close),
                 common ? unit : f_sigv(sv.anchor, sv.vm)/100.0*nz(sv.o, close) * i_g_height)
            slot += 1
        if i_g_day
            f_gCol(slot, I_DAY, D_O, D_H, D_L, LV_DAY.vm, f_gTag(LV_DAY),
                 common ? close : D_O,
                 common ? unit : f_sigv(I_DAY, LV_DAY.vm)/100.0*nz(D_O, close) * i_g_height)
            slot += 1
        if i_g_week
            f_gCol(slot, I_WEEK, W_O, W_H, W_L, LV_WEEK.vm, f_gTag(LV_WEEK),
                 common ? close : W_O,
                 common ? unit : f_sigv(I_WEEK, LV_WEEK.vm)/100.0*nz(W_O, close) * i_g_height)
            slot += 1

// One table, nine anchors, three optional column groups. The previous build
// split this in two - one for levels, one for live state - which looked like
// the same table printed twice and meant the Week anchor appeared in one and
// not the other, and only a single session could be shown at a time.

f_pctOf(o, hi, lo, a, vm, up) =>
    r = float(na)
    if not na(o) and o > 0 and f_sigv(a, vm) > 0
        sp = f_sigv(a, vm) / 100.0 * o
        r := f_rank(up ? MFE : MAE, a, up ? (hi - o) / sp : (o - lo) / sp)
    r

// Column layout, resolved once from the group toggles.
NC_LV = i_tb_lvls ? 6 : 0
NC_ST = i_tb_st   ? 4 : 0
NC_DG = i_tb_diag ? 4 : 0
NCOLS = 1 + NC_LV + NC_ST + NC_DG
C_LV  = 1
C_ST  = C_LV + NC_LV
C_DG  = C_ST + NC_ST

f_tbRow(t, row, v, o, hi, lo, isLive) =>
    a  = v.anchor
    vm = v.vm
    ok = not na(o) and o > 0 and f_sigv(a, vm) > 0
    // Live anchors read black, dormant ones grey, short sessions amber - so a
    // half-day announces itself instead of quietly poisoning the numbers.
    bg = v.short ? color.rgb(255,248,230) : isLive ? color.rgb(252,252,252) : color.rgb(240,240,240)
    tc = v.short ? color.rgb(150,95,0)    : isLive ? color.rgb(40,40,40)    : color.rgb(150,150,150)
    z  = f_sz(i_tb_sz)
    fn = font.family_monospace
    op = nz(o, close)
    zc = ok ? (close - o) / (f_sigv(a, vm)/100.0*o) : float(na)

    table.cell(t, 0, row, v.tag + (v.short ? " !" : v.stale ? "*" : ""),
         text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)

    if i_tb_lvls
        table.cell(t, C_LV+0, row, ok ? str.format("{0,number,#.###}", f_sigv(a,vm)) + "%" : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_LV+1, row, ok ? str.format("{0,number,#}", f_mfev(a,Q50,vm)/100*op) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_LV+2, row, ok ? str.format("{0,number,#}", f_mfev(a,Q75,vm)/100*op) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_LV+3, row, ok ? str.format("{0,number,#}", f_mfev(a,Q90,vm)/100*op) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_LV+4, row, ok ? str.format("{0,number,#}", f_maev(a,Q50,vm)/100*op) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_LV+5, row, ok ? str.format("{0,number,#}", f_maev(a,Q90,vm)/100*op) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)

    if i_tb_st
        table.cell(t, C_ST+0, row, ok ? "p"+str.tostring(int(math.round(f_pctOf(o,hi,lo,a,vm,true))))  : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_ST+1, row, ok ? "p"+str.tostring(int(math.round(f_pctOf(o,hi,lo,a,vm,false)))) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_ST+2, row, ok ? str.format("{0,number,#.##}", zc) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_ST+3, row, ok ? f_probTxt(a, zc) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)

    if i_tb_diag
        eb = f_expBars(a)
        table.cell(t, C_DG+0, row, ok ? str.format("{0,number,#.##}", o)  : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_DG+1, row, ok ? str.format("{0,number,#.##}", hi) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_DG+2, row, ok ? str.format("{0,number,#.##}", lo) : "-", text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)
        table.cell(t, C_DG+3, row, str.tostring(v.bars) + "/" + str.tostring(eb), text_size=z, text_color=tc, bgcolor=bg, text_font_family=fn)

if i_tb_on and barstate.islast and isNQ and tfOK
    var table t = table.new(f_pos(i_tb_pos), NCOLS, 11, bgcolor=color.white,
         border_color=color.rgb(200,200,200), frame_color=color.rgb(170,170,170),
         frame_width=1, border_width=1)
    z  = f_sz(i_tb_sz)
    fn = font.family_monospace
    hb = color.rgb(35,35,35)
    ht = color.white
    table.cell(t, 0, 0, "Anchor", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
    if i_tb_lvls
        table.cell(t, C_LV+0, 0, "sigma", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_LV+1, 0, "MFE50", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_LV+2, 0, "MFE75", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_LV+3, 0, "MFE90", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_LV+4, 0, "MAE50", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_LV+5, 0, "MAE90", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
    if i_tb_st
        // "now SD" not "now pNN" - the cell holds a sigma multiple, and the old
        // header invited it to be read as a percentile.
        table.cell(t, C_ST+0, 0, "up pNN", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_ST+1, 0, "dn pNN", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_ST+2, 0, "now SD", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_ST+3, 0, i_g_prob == "P fade (conditional)" ? "P fade" :
                                 i_g_prob == "P close below (unconditional)" ? "P<lvl" : "-",
             text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
    if i_tb_diag
        table.cell(t, C_DG+0, 0, "Open", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_DG+1, 0, "High", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_DG+2, 0, "Low",  text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)
        table.cell(t, C_DG+3, 0, "bars", text_size=z, text_color=ht, bgcolor=hb, text_font_family=fn)

    f_tbRow(t, 1, LV_HOUR, LV_HOUR.o, LV_HOUR.h, LV_HOUR.l, f_sigv(nyHour, 1.0) > 0)
    f_tbRow(t, 2, LV_ASIA, LV_ASIA.o, LV_ASIA.h, LV_ASIA.l, in_asia)
    f_tbRow(t, 3, LV_LON,  LV_LON.o,  LV_LON.h,  LV_LON.l,  in_lon)
    f_tbRow(t, 4, LV_AM,   LV_AM.o,   LV_AM.h,   LV_AM.l,   in_nyam)
    f_tbRow(t, 5, LV_PM,   LV_PM.o,   LV_PM.h,   LV_PM.l,   in_nypm)
    f_tbRow(t, 6, LV_RTH,  LV_RTH.o,  LV_RTH.h,  LV_RTH.l,  in_rth)
    f_tbRow(t, 7, LV_NY,   LV_NY.o,   LV_NY.h,   LV_NY.l,   in_ny)
    f_tbRow(t, 8, LV_DAY,  D_O, D_H, D_L, true)
    f_tbRow(t, 9, LV_WEEK, W_O, W_H, W_L, true)

    // A chart with no lines on it looks broken, but is usually correct: lines
    // are deleted when a period ends, so only live anchors draw. Say which of
    // the possible reasons applies rather than leave it to guesswork.
    anyOn = i_hourL or i_asia or i_lon or i_nyam or i_nypm or i_rth or i_nyf or i_dayL or i_weekL
    anyQ  = i_p25 or i_p50 or i_p75 or i_p90 or i_p95 or i_bb or i_rng
    drawn = 0
    for lv in array.from(LV_HOUR, LV_ASIA, LV_LON, LV_AM, LV_PM, LV_RTH, LV_NY, LV_DAY, LV_WEEK)
        if lv.show and array.size(lv.ln) > 0
            drawn += 1
    lvNote = not i_lv_on          ? "  |  LINES OFF - see Show chart lines"
         : not anyOn              ? "  |  NO LINES - no anchor selected"
         : i_mode == "Percentiles" and not anyQ ? "  |  NO LINES - no percentile selected"
         : drawn == 0             ? "  |  NO LINES - selected anchors are not live now"
         : "  |  " + str.tostring(drawn) + " anchors drawn"

    foot = "15y levels, points  |  src " + i_src
         + "  |  vol " + str.format("{0,number,#.##}", volLive) + " @" + str.tostring(i_slen) + "d"
         + "  |  D " + str.tostring(d_bars) + "  W " + str.tostring(w_bars)
         + (earlyClose ? "  |  ! EARLY CLOSE - affected anchors are short" : "")
         + lvNote
         + "  |  * held, not live"
    table.cell(t, 0, 10, foot, text_size=z, text_color=color.rgb(140,140,140),
         bgcolor=color.rgb(245,245,245), text_font_family=fn)
    table.merge_cells(t, 0, 10, NCOLS - 1, 10)
