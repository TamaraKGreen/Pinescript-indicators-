// === Helper Functions ===
get_line_style(style_string) =>
    style_string == "Solid" ? line.style_solid : style_string == "Dashed" ? line.style_dashed : line.style_dotted

get_label_size(size_string) =>
    size_string == "Tiny" ? size.tiny : size_string == "Small" ? size.small : size_string == "Normal" ? size.normal : size.large

get_label_style() =>
    label.style_label_left

is_newbar(sess) =>
    t = time('D', sess, timezone=time_zone)
    na(t[1]) and not na(t) or t[1] < t

is_session(sess) =>
    not na(time('D', sess, timezone=time_zone))

get_weekday() =>
    day = dayofweek(time, time_zone)
    day == dayofweek.monday ? "Monday" : day == dayofweek.tuesday ? "Tuesday" : day == dayofweek.wednesday ? "Wednesday" : day == dayofweek.thursday ? "Thursday" : day == dayofweek.friday ? "Friday" : "Other"

get_table_position(pos) =>
    switch pos
        "Top Left" => position.top_left
        "Top Center" => position.top_center
        "Top Right" => position.top_right
        "Middle Left" => position.middle_left
        "Middle Center" => position.middle_center
        "Middle Right" => position.middle_right
        "Bottom Left" => position.bottom_left
        "Bottom Center" => position.bottom_center
        "Bottom Right" => position.bottom_right

get_table_text_size(size) =>
    switch size
        "Tiny" => size.tiny
        "Small" => size.small
        "Normal" => size.normal
        "Large" => size.large

                break

        else if current_day == "Thursday"
            thu_upper_10_touches := thu_upper_10_touches + 1
        else if current_day == "Friday"
            fri_upper_10_touches := fri_upper_10_touches + 1

    if not lower_10_touched and low <= lower_10_level
        if current_day == "Monday"
            mon_lower_10_touches := mon_lower_10_touches + 1
        else if current_day == "Tuesday"
            tue_lower_10_touches := tue_lower_10_touches + 1
        else if current_day == "Wednesday"
            wed_lower_10_touches := wed_lower_10_touches + 1
        else if current_day == "Thursday"

        else if current_day == "Friday"
            fri_lower_10_touches := fri_lower_10_touches + 1

    float upper_20_level = current_open + 2.0 * sd
