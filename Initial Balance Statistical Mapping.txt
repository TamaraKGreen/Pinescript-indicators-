//-------------------------------------------------------------------------------------------------
// 1-Minute LTF Data
// This is the KEY fix. Rather than trying to detect the first extreme
// bar-by-bar on the chart timeframe (which fails when a single chart bar
// contains both the session high and low), we collect every 1-minute bar
// during the IB session and walk them in order at session end.
// The first 1-min bar whose high equals the session high  → high was first.
// The first 1-min bar whose low  equals the session low   → low  was first.
// Whichever index is smaller wins — simple, accurate, and reliable.
//-------------------------------------------------------------------------------------------------
[high_1m, low_1m] = request.security_lower_tf(ticker.standard(syminfo.tickerid), "1", [high, low])

var ib_ranges = array.new_float(0)

var array<float> ltf_highs_frozen = array.new<float>()
var array<float> ltf_lows_frozen  = array.new<float>()

var bool stored_is_bullish     = false

//-------------------------------------------------------------------------------------------------
// Functions

inSession(sess) =>
    not na(time(timeframe.period, sess, time_zone))

get_levels(n) =>
    h = high[1]
    l = low[1]
    o = open[n]
    c = close[1]
    for i = 1 to n
        if low[i]  < l
            l := low[i]
        if high[i] > h
            h := high[i]
    [h, l, (h + l) / 2, o, c]

get_line_style(s) =>
    s == "Solid" ? line.style_solid : s == "Dashed" ? line.style_dashed : line.style_dotted

get_label_size(s) =>
    s == "Tiny" ? size.tiny : s == "Small" ? size.small : s == "Normal" ? size.normal : s == "Large" ? size.large : size.huge

get_table_position(pos) =>
    switch pos
        "Bottom Center" => position.bottom_center
        "Bottom Left"   => position.bottom_left
        "Bottom Right"  => position.bottom_right
        "Middle Center" => position.middle_center
        "Middle Left"   => position.middle_left
        "Middle Right"  => position.middle_right
        "Top Center"    => position.top_center
        "Top Left"      => position.top_left
        "Top Right"     => position.top_right

get_table_text_size(sz) =>
    switch sz
        "Auto"   => size.auto
        "Tiny"   => size.tiny
        "Small"  => size.small
        "Normal" => size.normal
        "Large"  => size.large
        "Huge"   => size.huge

get_ordinal_suffix(num) =>
    int last_digit      = math.floor(num) % 10
    int last_two_digits = math.floor(num) % 100
    string suffix = "th"
    if last_two_digits == 11 or last_two_digits == 12 or last_two_digits == 13
        suffix := "th"
    else if last_digit == 1
        suffix := "st"
    else if last_digit == 2
        suffix := "nd"
    else if last_digit == 3
        suffix := "rd"
    suffix

array_mean(arr) =>
    float sum = 0.0
    int sz = array.size(arr)
    if sz > 0
        for i = 0 to sz - 1
            sum += array.get(arr, i)
        sum / sz
    else
        na

calc_percentile_rank(current_value, arr) =>
    int sz = array.size(arr)
    if sz > 0 and not na(current_value)
        int count_below = 0
        for i = 0 to sz - 1
            if array.get(arr, i) < current_value
                count_below += 1
        (count_below / sz) * 100
    else
        na

get_ib_stats(is_bullish, high_first, close_above_mid) =>
    if is_bullish
        if high_first
            close_above_mid ? BULL_HIGH_ABOVE : BULL_HIGH_BELOW
        else
            close_above_mid ? BULL_LOW_ABOVE : BULL_LOW_BELOW
    else
        if high_first
            close_above_mid ? BEAR_HIGH_ABOVE : BEAR_HIGH_BELOW
        else
            close_above_mid ? BEAR_LOW_ABOVE : BEAR_LOW_BELOW

get_retest_stats(is_bullish, close_above_mid, high_broke) =>
    if is_bullish
        if close_above_mid
            high_broke ? BULL_ABOVE_HIGH_BREAKS : BULL_ABOVE_LOW_BREAKS
        else
            high_broke ? BULL_BELOW_HIGH_BREAKS : BULL_BELOW_LOW_BREAKS
    else
        if close_above_mid
            high_broke ? BEAR_ABOVE_HIGH_BREAKS : BEAR_ABOVE_LOW_BREAKS
        else
            high_broke ? BEAR_BELOW_HIGH_BREAKS : BEAR_BELOW_LOW_BREAKS

//-------------------------------------------------------------------------------------------------
// Main Logic
//-------------------------------------------------------------------------------------------------
ins    = inSession(ib_session)
in_rth = inSession(rth_session)

// Reset at IB session start
if ins and not ins[1]
    offset           := 0
    ltf_frozen       := false
    ib_extreme_first := "none"
    ltf_highs_frozen.clear()
    ltf_lows_frozen.clear()

// Accumulate offset and LTF data during session
if ins
    offset += 1
    // Collect all 1-min bars delivered on this chart bar
    if not ltf_frozen and not na(high_1m) and array.size(high_1m) > 0
        for i = 0 to array.size(high_1m) - 1
            ltf_highs_frozen.push(array.get(high_1m, i))
            ltf_lows_frozen.push(array.get(low_1m, i))

// Track breakouts after IB forms
if ib_formed and not na(ib_high) and not na(ib_low)
    if not ib_high_broken and high > ib_high
        ib_high_broken        := true
        table_updated_for_break := false
    if not ib_low_broken and low < ib_low
        ib_low_broken         := true
        table_updated_for_break := false

// Track midpoint retest
if ib_formed and not na(ib_mid_price)
    threshold = ib_mid_price * 0.001
    if not left_midpoint
        if math.abs(close - ib_mid_price) > threshold
            left_midpoint := true
    if left_midpoint and not midpoint_retested
        if math.abs(low - ib_mid_price) <= threshold or math.abs(high - ib_mid_price) <= threshold
            midpoint_retested := true
            retest_bar_index  := bar_index

// End of RTH — clear table and reset
if in_rth[1] and not in_rth
    table.delete(stats_table)
    stats_table             := na
    ib_formed               := false
    table_updated_for_break := false
    midpoint_retested       := false
    left_midpoint           := false

// ============================================================
// SESSION END: determine extreme-first from 1-min LTF data,
// then draw all visuals and build the stats table
// ============================================================
if ins[1] and not ins
    ltf_frozen := true

    // Get IB levels
    [h, l, m, o, c] = get_levels(offset)
    ib_high      := h
    ib_low       := l
    ib_mid_price := m
    ib_open      := o
    ib_close     := c
    ib_formed    := true
    ib_high_broken          := false
    ib_low_broken           := false
    table_updated_for_break := false
    midpoint_retested       := false
    left_midpoint           := false

    // ----------------------------------------------------------------
    // Extreme-first detection via 1-minute data.
    //
    // Walk all accumulated 1-min bars chronologically. Find the earliest
    // index where the bar's high reached the final session high (high_index)
    // and the earliest index where the bar's low reached the final session
    // low (low_index). Whichever index is smaller came first.
    //
    // Using >= / <= rather than == handles floating-point edge cases where
    // a 1-min bar slightly exceeds the OHLC value reported by the HTF bar.
    // ----------------------------------------------------------------
    int high_index = -1
    int low_index  = -1
    int ltf_size   = array.size(ltf_highs_frozen)

    if ltf_size > 0
        for i = 0 to ltf_size - 1
            if high_index == -1 and array.get(ltf_highs_frozen, i) >= h
                high_index := i
            if low_index == -1 and array.get(ltf_lows_frozen, i) <= l
                low_index := i
            if high_index != -1 and low_index != -1
                break

    if high_index != -1 and low_index != -1
        ib_extreme_first := high_index <= low_index ? "high" : "low"
    else if high_index != -1
        ib_extreme_first := "high"
    else if low_index != -1
        ib_extreme_first := "low"
    else
        ib_extreme_first := "none"

    // Store for post-break table updates
    stored_is_bullish      := c >= o
    stored_close_above_mid := c > m
    stored_high_first      := ib_extreme_first == "high"

    // Store IB range
    ib_range = h - l
    array.push(ib_ranges, ib_range)
    if array.size(ib_ranges) > ib_stats_lookback
        array.shift(ib_ranges)

    // ---- Draw visuals ----
    if fill_ib_areas
        if only_current_zone
            box.delete(ib_area)
            line.delete(ib_midpoint)
            line.delete(ib_ext_high_25th)
            line.delete(ib_ext_high_median)
            line.delete(ib_ext_high_75th)
            line.delete(ib_ext_high_90th)
            line.delete(ib_ext_low_25th)
            line.delete(ib_ext_low_median)
            line.delete(ib_ext_low_75th)
            line.delete(ib_ext_low_90th)
            label.delete(ib_ext_high_25th_label)
            label.delete(ib_ext_high_median_label)
            label.delete(ib_ext_high_75th_label)
            label.delete(ib_ext_high_90th_label)
            label.delete(ib_ext_low_25th_label)
            label.delete(ib_ext_low_median_label)
            label.delete(ib_ext_low_75th_label)
            label.delete(ib_ext_low_90th_label)

        ib_area := box.new(
             left=bar_index[offset], top=h, right=bar_index, bottom=l,
             bgcolor=fill_ib_color, border_color=border_color,
             border_style=get_line_style(line_style), border_width=1)
        ib_midpoint := line.new(
             x1=bar_index[offset], y1=m, x2=bar_index, y2=m,
             color=midpoint_color, style=get_line_style(line_style), width=1)

        ext_style  = get_line_style(ext_line_style)
        label_size = get_label_size(ext_label_size)

        if show_ext_25th
            ep25h = h * (1 + EXT_HIGH_25TH / 100)
            ep25l = l * (1 - EXT_LOW_25TH  / 100)
            ib_ext_high_25th := line.new(bar_index[offset], ep25h, bar_index, ep25h, color=ext_color, style=ext_style, width=ext_line_width)
            ib_ext_low_25th  := line.new(bar_index[offset], ep25l, bar_index, ep25l, color=ext_color, style=ext_style, width=ext_line_width)
            if show_ext_labels
                ib_ext_high_25th_label := label.new(bar_index[offset], ep25h, "25th +0.14%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)
                ib_ext_low_25th_label  := label.new(bar_index[offset], ep25l, "25th -0.16%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)

        if show_ext_median
            epmh = h * (1 + EXT_HIGH_MEDIAN / 100)
            epml = l * (1 - EXT_LOW_MEDIAN  / 100)
            ib_ext_high_median := line.new(bar_index[offset], epmh, bar_index, epmh, color=ext_color, style=ext_style, width=ext_line_width)
            ib_ext_low_median  := line.new(bar_index[offset], epml, bar_index, epml, color=ext_color, style=ext_style, width=ext_line_width)
            if show_ext_labels
                ib_ext_high_median_label := label.new(bar_index[offset], epmh, "Median +0.31%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)
                ib_ext_low_median_label  := label.new(bar_index[offset], epml, "Median -0.38%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)

        if show_ext_75th
            ep75h = h * (1 + EXT_HIGH_75TH / 100)
            ep75l = l * (1 - EXT_LOW_75TH  / 100)
            ib_ext_high_75th := line.new(bar_index[offset], ep75h, bar_index, ep75h, color=ext_color, style=ext_style, width=ext_line_width)
            ib_ext_low_75th  := line.new(bar_index[offset], ep75l, bar_index, ep75l, color=ext_color, style=ext_style, width=ext_line_width)
            if show_ext_labels
                ib_ext_high_75th_label := label.new(bar_index[offset], ep75h, "75th +0.60%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)
                ib_ext_low_75th_label  := label.new(bar_index[offset], ep75l, "75th -0.79%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)

        if show_ext_90th
            ep90h = h * (1 + EXT_HIGH_90TH / 100)
            ep90l = l * (1 - EXT_LOW_90TH  / 100)
            ib_ext_high_90th := line.new(bar_index[offset], ep90h, bar_index, ep90h, color=ext_color, style=ext_style, width=ext_line_width)
            ib_ext_low_90th  := line.new(bar_index[offset], ep90l, bar_index, ep90l, color=ext_color, style=ext_style, width=ext_line_width)
            if show_ext_labels
                ib_ext_high_90th_label := label.new(bar_index[offset], ep90h, "90th +1.00%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)
                ib_ext_low_90th_label  := label.new(bar_index[offset], ep90l, "90th -1.37%", style=label.style_label_right, color=color.new(ext_color, 100), textcolor=ext_color, size=label_size)

    // ---- Build stats table ----
    if only_current_zone
        table.delete(stats_table)

    table_pos = get_table_position(table_position)
    ts        = get_table_text_size(table_size)

    header_bg    = color.rgb(64, 64, 64)
    header_text  = color.white
    cell_bg      = color.white
    cell_text    = color.rgb(64, 64, 64)
    highlight_bg = color.rgb(46, 125, 50)
    highlight_text = color.white

    ib_range_current     = h - l
    ib_mean_range        = array_mean(ib_ranges)
    ib_percentile        = calc_percentile_rank(ib_range_current, ib_ranges)
    ib_sessions_captured = array.size(ib_ranges)

    is_bullish      = c >= o
    close_above_mid = c > m
    high_first      = ib_extreme_first == "high"

    stats               = get_ib_stats(is_bullish, high_first, close_above_mid)
    sample_size         = array.get(stats, 0)
    break_high_pct      = array.get(stats, 1)
    break_low_pct       = array.get(stats, 2)
    neither_pct         = array.get(stats, 3)
    midpoint_retest_pct = array.get(stats, 4)
    bullish_close_pct   = array.get(stats, 6)

    stats_table := table.new(table_pos, 3, 20, border_width=1, border_color=color.rgb(64, 64, 64), bgcolor=cell_bg)
    row = 0

    // Header
    table.cell(stats_table, 0, row, "━━━ IB STATISTICS ━━━", bgcolor=header_bg, text_color=header_text, text_size=ts, text_font_family=font.family_monospace)
    table.merge_cells(stats_table, 0, row, 2, row)
    row += 1

    // IB Direction
    table.cell(stats_table, 0, row, "IB Direction", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    dir_color = is_bullish ? color.green : color.red
    table.cell(stats_table, 1, row, is_bullish ? "Bullish" : "Bearish", bgcolor=cell_bg, text_color=dir_color, text_size=ts, text_font_family=font.family_monospace)
    table.cell(stats_table, 2, row, "", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    row += 1

    // Extreme First
    table.cell(stats_table, 0, row, "Extreme First", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    extreme_text = ib_extreme_first == "high" ? "High First" : ib_extreme_first == "low" ? "Low First" : "Unknown"
    table.cell(stats_table, 1, row, extreme_text, bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    table.cell(stats_table, 2, row, "", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    row += 1

    // Close Position
    table.cell(stats_table, 0, row, "Close Position", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    table.cell(stats_table, 1, row, close_above_mid ? "Above Midpoint" : "Below Midpoint", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    table.cell(stats_table, 2, row, "", bgcolor=cell_bg, text_color=cell_text, text_size=ts, text_font_family=font.family_monospace)
    row += 1

    // Separator
    table.cell(stats_table, 0, row, "", bgcolor=cell_bg, text_color=cell_text, text_size=ts)
    table.merge_cells(stats_table, 0, row, 2, row)
    row += 1

    // Historical Pattern
    table.cell(stats_table, 0, row, "Historical Pattern", bgcolor=header_bg, text_color=header_text, text_size=ts, text_font_family=font.family_monospace)
    table.cell(stats_table, 1, row, str.tostring(sample_size, "#") + " sessions", bgcolor=header_bg, text_color=header_text, text_size=ts, text_font_family=font.family_monospace)
    table.cell(stats_table, 2, row, "", bgcolor=header_bg, text_color=header_text, text_size=ts, text_font_family=font.family_monospace)
    row += 1
