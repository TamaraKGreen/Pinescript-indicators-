// Utility Functions
inSession(sess) =>
    na(time(timeframe.period, sess)) == false

get_line_style(s) =>
    s == "Solid" ? line.style_solid : s == "Dotted" ? line.style_dotted : line.style_dashed

get_levels(n) =>
    h = high[1]
    l = low[1]
    for i = 1 to n
        if low[i] < l
            l := low[i]
        if high[i] > h
            h := high[i]
    [h, l, (h + l) / 2]

get_table_position(pos) =>
    result = switch pos
        "Bottom Center" => position.bottom_center
        "Bottom Left" => position.bottom_left
        "Bottom Right" => position.bottom_right
        "Middle Center" => position.middle_center
        "Middle Left" => position.middle_left
        "Middle Right" => position.middle_right
        "Top Center" => position.top_center
        "Top Left" => position.top_left
        "Top Right" => position.top_right

get_table_text_size(size) =>
    result = switch size
        'Auto' => size.auto
        'Tiny' => size.tiny
        'Small' => size.small
        'Normal' => size.normal
        'Large' => size.large
        'Huge' => size.huge

                                     color=ext_color, extend=ext, style=line.style_dotted)

                     style=label.style_none, textcolor=ext_color, size=labelSize)
            sd3_low_line := line.new(bar_index[offset], ib_low - (sd_delta * 3), bar_index, ib_low - (sd_delta * 3),
                                     color=ext_color, extend=ext, style=line.style_dotted)
            label.new(bar_index[offset], ib_low - (sd_delta * 3), "3 SD Low: " + str.tostring(ib_low - (sd_delta * 3), "#.##"),
                     style=label.style_none, textcolor=ext_color, size=labelSize)

            sd4_high_line := line.new(bar_index[offset], ib_high + (sd_delta * 4), bar_index, ib_high + (sd_delta * 4),

            sd4_low_line := line.new(bar_index[offset], ib_low - (sd_delta * 4), bar_index, ib_low - (sd_delta * 4),
                                     color=ext_color, extend=ext, style=line.style_dotted)
            label.new(bar_index[offset], ib_low - (sd_delta * 4), "4 SD Low: " + str.tostring(ib_low - (sd_delta * 4), "#.##"),
                     style=label.style_none, textcolor=ext_color, size=labelSize)    
    offset := 0

if ib_formed and not ins
    if high > current_high_ext
        current_high_ext := high
    if low < current_low_ext
        current_low_ext := low
    
    // Track hits after IB is formed
    if show_extensions and not na(array.avg(high_extensions)) and not mean_hit
        if high >= ib_high + array.avg(high_extensions) or low <= ib_low - array.avg(low_extensions)
            mean_hit := true
    if show_extensions and not na(array.median(high_extensions)) and not median_hit
        if high >= ib_high + array.median(high_extensions) or low <= ib_low - array.median(low_extensions)
            median_hit := true
    if show_sd2 and not na(array.stdev(delta_history)) and not sd2_hit
        sd_delta = array.stdev(delta_history)
        if high >= ib_high + (sd_delta * 2) or low <= ib_low - (sd_delta * 2)
            sd2_hit := true
    if show_sd3 and not na(array.stdev(delta_history)) and not sd3_hit
        sd_delta = array.stdev(delta_history)
        if high >= ib_high + (sd_delta * 3) or low <= ib_low - (sd_delta * 3)
            sd3_hit := true
    if show_sd4 and not na(array.stdev(delta_history)) and not sd4_hit
        sd_delta = array.stdev(delta_history)
        if high >= ib_high + (sd_delta * 4) or low <= ib_low - (sd_delta * 4)
            sd4_hit := true

if not ins[1] and ins and ib_formed
