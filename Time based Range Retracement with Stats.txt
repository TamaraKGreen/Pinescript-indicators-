    range_start_time_1 := time
    new_session_1 := true

    track_1.high_taken := false
    track_1.low_taken := false

    session_low_1 := math.min(session_low_1, low)
    range_end_time_1 := time

if in_hl_check_time_1
    if not track_1.high_taken and high >= session_high_1
        track_1.high_taken := true
        track_1.count_high_taken += 1
    if not track_1.low_taken and low <= session_low_1
        track_1.low_taken := true
        track_1.count_low_taken += 1

if not in_range_time_1 and in_range_time_1[1] and new_session_1
    if show_box_1
        track_1.range_box := box.new(range_start_time_1, session_high_1, range_end_time_1, session_low_1, border_color=range_col_1, bgcolor=range_col_1, xloc=xloc.bar_time)
    float range_size = session_high_1 - session_low_1
    track_1.ret_price := session_low_1 + range_size * (ret_pct_1 / 100)
    if show_line_1
        track_1.ret_line := line.new(range_start_time_1, track_1.ret_price, range_end_time_1 + timeframe.in_seconds("D") * 1000, track_1.ret_price, xloc=xloc.bar_time, color=ret_line_col_1, style=style_1, width=line_width_1)
    track_1.ret_distance.unshift(range_size * (ret_pct_1 / 100))
    new_session_1 := false

if in_ret_time_1 and not track_1.returned
    if high >= track_1.ret_price and low <= track_1.ret_price
        track_1.returned := true

            track_1.count_returned_high_taken += 1

        if track_1.low_taken
            track_1.count_returned_low_taken += 1
