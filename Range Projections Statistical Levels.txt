get_label_size(size_str) =>
    switch size_str
        'Tiny' => size.tiny
        'Small' => size.small
        'Normal' => size.normal
        'Large' => size.large
        'Huge' => size.huge
        => size.normal

get_line_style(style_str) =>
    switch style_str
        'Solid' => line.style_solid
        'Dotted' => line.style_dotted
        'Dashed' => line.style_dashed
        => line.style_dotted

get_gradient_color(level_index, total_levels) =>
    if use_gradient
        transparency = int(80 - (level_index * 10))
        transparency := math.max(10, math.min(90, transparency))
        color.new(base_color, transparency)
    else
        color.new(base_color, 70)

label_text_size = get_label_size(label_size)

// Calculate approximate session duration in bars
get_session_end_bar(start_bar) =>
    // Get session duration in minutes (all sessions are 1 hour)
    session_duration_minutes = 60
    // Get chart timeframe in minutes
    timeframe_minutes = timeframe.in_seconds() / 60
    // Calculate number of bars in session
    bars_in_session = math.round(session_duration_minutes / timeframe_minutes)
    // Return estimated end bar
    start_bar + bars_in_session

if t and not t[1]
    // Session start
    start := bar_index
    session_open := open
    session_high := high
    session_low := low
    session_end_bar := get_session_end_bar(bar_index)
    
    // Clear previous drawings
    array.clear(proj_boxes)
    array.clear(proj_lines)
    
    // Create projection levels at session open
    [levels_up, levels_down, level_labels] = get_levels_for_session_and_day()
    
    if session_open > 0 and array.size(levels_up) > 0
        // Create session open line
        session_open_line := line.new(start, session_open, session_end_bar, session_open, color = open_line_color, style = open_line_style_converted, width = 1)
        
        // Create level boxes and lines
        for i = 0 to array.size(levels_up) - 1
            level_up_pct = array.get(levels_up, i)
            level_down_pct = array.get(levels_down, i)
            label_text = array.get(level_labels, i)
            
            if level_up_pct > 0
                // Calculate level prices
                upper_level = session_open * (1 + level_up_pct)
                lower_level = session_open * (1 - level_down_pct)
                
                // Create boxes (stacked approach - only between levels)
                var box upper_box = na
                var box lower_box = na
                box_color = get_gradient_color(i, array.size(levels_up))
                
                if i == 0
                    // For first level, create boxes from session open to first level
                    upper_box := box.new(start, upper_level, session_end_bar, session_open, border_color = color.new(box_color, 100), bgcolor = box_color)
                    lower_box := box.new(start, session_open, session_end_bar, lower_level, border_color = color.new(box_color, 100), bgcolor = box_color)
                else
                    // For subsequent levels, create boxes between previous level and current level
                    prev_upper = session_open * (1 + array.get(levels_up, i-1))
                    prev_lower = session_open * (1 - array.get(levels_down, i-1))
                    upper_box := box.new(start, upper_level, session_end_bar, prev_upper, border_color = color.new(box_color, 100), bgcolor = box_color)
                    lower_box := box.new(start, prev_lower, session_end_bar, lower_level, border_color = color.new(box_color, 100), bgcolor = box_color)
                
                array.push(proj_boxes, upper_box)
                array.push(proj_boxes, lower_box)
                
                // Create labels
                if show_labels
                    label.new(session_end_bar, upper_level, label_text, textcolor = label_text_color, color = label_color, style = label.style_label_center, size = label_text_size)
                    label.new(session_end_bar, lower_level, "-" + label_text, textcolor = label_text_color, color = label_color, style = label.style_label_center, size = label_text_size)
                
                // Create level lines
                if horiz_lines
                    upper_line = line.new(start, upper_level, session_end_bar, upper_level, color = line_color, style = level_line_style, width = 1)
                    lower_line = line.new(start, lower_level, session_end_bar, lower_level, color = line_color, style = level_line_style, width = 1)
                    array.push(proj_lines, upper_line)
                    array.push(proj_lines, lower_line)

    if high > session_high

    if low < session_low
        session_low := low
