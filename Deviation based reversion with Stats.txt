// === MOVING AVERAGE CALCULATIONS (UNCHANGED) === //
gd(src, len, factor) =>
    ta.ema(src, len) * (1 + factor) - ta.ema(ta.ema(src, len), len) * factor

ma(type, src, len, alma_offset, alma_sigma, tvma_length, t3_factor, kama_fast_len, kama_slow_len) =>
    var float result = na
    if type == "SMA"
        result := ta.sma(src, len)
    else if type == "EMA"
        result := ta.ema(src, len)
    else if type == "WMA"
        result := ta.wma(src, len)
    else if type == "RMA"
        result := ta.rma(src, len)
    else if type == "HMA"
        result := ta.hma(src, len)
    else if type == "DEMA"
        e = ta.ema(src, len)
        result := 2 * e - ta.ema(e, len)
    else if type == "TEMA"
        e = ta.ema(src, len)
        result := 3 * (e - ta.ema(e, len)) + ta.ema(ta.ema(e, len), len)
    else if type == "VWMA"
        result := ta.vwma(src, len)
    else if type == "ALMA"
        result := ta.alma(src, len, alma_offset, alma_sigma)
    else if type == "TVMA"
        alpha = 2.0 / (tvma_length + 1)
        alphaSum = 0.0
        vmaSum = 0.0
        for i = 0 to len - 1 by 1
            alphaSum += math.pow(alpha, i)
            vmaSum += src[len - i - 1] * math.pow(alpha, i)
        result := alphaSum != 0 ? vmaSum / alphaSum : na
    else if type == "T3"
        result := gd(gd(gd(src, len, t3_factor), len, t3_factor), len, t3_factor)
    else if type == "ZLEMA"
        lag = math.floor((len - 1) / 2)
        result := ta.ema(src + src - src[lag], len)
    else if type == "KAMA"
        mom_kauf = math.abs(ta.change(src))
        volatility_kauf = math.sum(math.abs(ta.change(src)), len)
        er_kauf = volatility_kauf != 0 ? mom_kauf / volatility_kauf : 0
        fastAlpha_kauf = 2 / (kama_fast_len + 1)
        slowAlpha_kauf = 2 / (kama_slow_len + 1)
        alpha_kauf = math.pow(er_kauf * (fastAlpha_kauf - slowAlpha_kauf) + slowAlpha_kauf, 2)
        result := alpha_kauf * src + (1 - alpha_kauf) * nz(result[1], src)
    result

// === DEVIATION CALCULATIONS === //
calc_deviation(type, src, length, linreg) =>
    var float deviation = na
    raw_dev = src - linreg
    if type == "MAD"
        medianPrice = ta.median(src, length)
        absDev = math.abs(src - medianPrice)
        mad = ta.median(absDev, length)
        deviation := mad > 0 ? raw_dev / mad : 0
    else if type == "Standard Deviation"
        std = ta.stdev(src, length)
        deviation := std > 0 ? raw_dev / std : 0
    else if type == "ATR"
        atr = ta.atr(length)
        deviation := atr > 0 ? raw_dev / atr : 0
    else if type == "Z-Score"
        mean = ta.sma(src, length)
        std = ta.stdev(src, length)
        deviation := std > 0 ? (src - mean) / std : 0
    else if type == "IQR"
        q1 = ta.percentile_nearest_rank(src, length, 25)
        q3 = ta.percentile_nearest_rank(src, length, 75)
        iqr = q3 - q1
        deviation := iqr > 0 ? raw_dev / iqr : 0
    else if type == "Mean AD"
        mean = ta.sma(src, length)
        mean_ad = ta.sma(math.abs(src - mean), length)
        deviation := mean_ad > 0 ? raw_dev / mean_ad : 0
    deviation
