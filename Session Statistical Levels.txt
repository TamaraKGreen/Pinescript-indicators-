TZ = "America/New_York"

f_lbl_size() =>
    i_label_size == "Tiny" ? size.tiny :
     i_label_size == "Small" ? size.small : size.normal

f_tbl_size(s) =>
    s == "Tiny" ? size.tiny : s == "Small" ? size.small : size.normal

f_tpos(s) =>
    switch s
        "Top Left"      => position.top_left
        "Top Right"     => position.top_right
        "Top Center"    => position.top_center
        "Middle Left"   => position.middle_left
        "Middle Center" => position.middle_center
        "Middle Right"  => position.middle_right
        "Bottom Left"   => position.bottom_left
        "Bottom Center" => position.bottom_center
        => position.bottom_right

f_line_style(s) =>
    switch s
        "Solid"  => line.style_solid
        "Dashed" => line.style_dashed
        => line.style_dotted

f_pct_val(arr, p) =>
    array.size(arr) >= 2 ? array.percentile_nearest_rank(arr, p) : float(na)

f_pct_rank(arr, val) =>
    n = array.size(arr)
    if n == 0
        int(na)
    else
        count = 0
        for i = 0 to n - 1
            if array.get(arr, i) <= val
                count += 1
        math.round((count / n) * 100)

f_fmt_pts(v) =>
    not na(v) ?
     (v >= 0 ? "+" : "") + str.format("{0,number,#.#}", v) + " pts" : "—"

f_fmt_p(v) =>
    not na(v) ? str.format("{0,number,#.#}", v) : "—"

f_rev_str(tot, rev) =>
    tot > 0 ? str.tostring(math.round((rev / tot) * 100)) + "%" : "—"

// ─── f_record_session ─────────────────────────────────────────────
// Called at session end with all OHLC values.
// 1. Computes mfe/mae/rng/cls and pushes to arrays (trimmed to 100)
// 2. Splits into bull/bear directional arrays
// 3. Determines if session "reverted" (close within cls P25-P75)
// 4. Returns 22 counter deltas:
//    [mfe25_t, mfe25_r, mfe25_nxt,
//     mfe50_t, mfe50_r, mfe50_nxt,
//     mfe75_t, mfe75_r, mfe75_nxt,
//     mfe90_t, mfe90_r,
//     mae25_t, mae25_r, mae25_nxt,
//     mae50_t, mae50_r, mae50_nxt,
//     mae75_t, mae75_r, mae75_nxt,
//     mae90_t, mae90_r]

f_record_session(rng_a, mfe_a, mae_a,
                 bull_mfe_a, bull_mae_a,
                 bear_mfe_a, bear_mae_a,
                 cls_a,
                 s_open, s_high, s_low, s_close,
                 hit_mfe90, ret_mfe90,
                 hit_mfe75, ret_mfe75,
                 hit_mfe50, ret_mfe50,
                 hit_mae90, ret_mae90,
                 hit_mae75, ret_mae75,
                 hit_mae50, ret_mae50) =>

    mfe = s_high - s_open   // max upside from open
    mae = s_open - s_low    // max downside from open
    rng = s_high - s_low
    cls = s_close - s_open  // signed close deviation
    is_bull = s_close >= s_open

    // Push to combined arrays
    array.push(rng_a, rng)
    array.push(mfe_a, mfe)
    array.push(mae_a, mae)
    array.push(cls_a, cls)

    // Push to directional arrays
    if is_bull
        array.push(bull_mfe_a, mfe)
        array.push(bull_mae_a, mae)
    else
        array.push(bear_mfe_a, mfe)
        array.push(bear_mae_a, mae)

    // Trim all arrays to 100 sessions
    if array.size(rng_a)      > 100
        array.shift(rng_a)
    if array.size(mfe_a)      > 100
        array.shift(mfe_a)
    if array.size(mae_a)      > 100
        array.shift(mae_a)
    if array.size(cls_a)      > 100
        array.shift(cls_a)
    if array.size(bull_mfe_a) > 100
        array.shift(bull_mfe_a)
    if array.size(bull_mae_a) > 100
        array.shift(bull_mae_a)
    if array.size(bear_mfe_a) > 100
        array.shift(bear_mfe_a)
    if array.size(bear_mae_a) > 100
        array.shift(bear_mae_a)

    // MFE level thresholds
    mfe_p25 = array.size(mfe_a) >= 5 ? array.percentile_nearest_rank(mfe_a, 25) : float(na)
    mfe_p50 = array.size(mfe_a) >= 5 ? array.percentile_nearest_rank(mfe_a, 50) : float(na)
    mfe_p75 = array.size(mfe_a) >= 5 ? array.percentile_nearest_rank(mfe_a, 75) : float(na)
    mfe_p90 = array.size(mfe_a) >= 5 ? array.percentile_nearest_rank(mfe_a, 90) : float(na)

    // MAE level thresholds
    mae_p25 = array.size(mae_a) >= 5 ? array.percentile_nearest_rank(mae_a, 25) : float(na)
    mae_p50 = array.size(mae_a) >= 5 ? array.percentile_nearest_rank(mae_a, 50) : float(na)
    mae_p75 = array.size(mae_a) >= 5 ? array.percentile_nearest_rank(mae_a, 75) : float(na)
    mae_p90 = array.size(mae_a) >= 5 ? array.percentile_nearest_rank(mae_a, 90) : float(na)

    // MFE counter deltas
    mfe25_t = not na(mfe_p25) and mfe >= mfe_p25 ? 1 : 0
    mfe50_t = not na(mfe_p50) and mfe >= mfe_p50 ? 1 : 0
    mfe75_t = not na(mfe_p75) and mfe >= mfe_p75 ? 1 : 0
    mfe90_t = not na(mfe_p90) and mfe >= mfe_p90 ? 1 : 0

    // MAE counter deltas
    mae25_t = not na(mae_p25) and mae >= mae_p25 ? 1 : 0
    mae50_t = not na(mae_p50) and mae >= mae_p50 ? 1 : 0
    mae75_t = not na(mae_p75) and mae >= mae_p75 ? 1 : 0
    mae90_t = not na(mae_p90) and mae >= mae_p90 ? 1 : 0

    // Next-level deltas — reached this level AND the next level in same direction
    mfe25_nxt = mfe25_t == 1 and mfe50_t == 1 ? 1 : 0
    mfe50_nxt = mfe50_t == 1 and mfe75_t == 1 ? 1 : 0
    mfe75_nxt = mfe75_t == 1 and mfe90_t == 1 ? 1 : 0
    mae25_nxt = mae25_t == 1 and mae50_t == 1 ? 1 : 0
    mae50_nxt = mae50_t == 1 and mae75_t == 1 ? 1 : 0
    mae75_nxt = mae75_t == 1 and mae90_t == 1 ? 1 : 0

    // Opposite-direction sequenced deltas
    // hit_mfe75 then hit_mae25_seq = price went to MFE P75 THEN reversed to MAE P25
    // hit_mae75 then hit_mfe25_seq = price went to MAE P75 THEN reversed to MFE P25
    // Same-side retrace deltas — after hitting level, did price retrace to level below?
    mfe25_opp = 0
    mfe50_opp = mfe50_t == 1 and ret_mfe50 ? 1 : 0
    mfe75_opp = mfe75_t == 1 and ret_mfe75 ? 1 : 0
    mfe90_opp = mfe90_t == 1 and ret_mfe90 ? 1 : 0
    mae25_opp = 0
    mae50_opp = mae50_t == 1 and ret_mae50 ? 1 : 0
    mae75_opp = mae75_t == 1 and ret_mae75 ? 1 : 0
    mae90_opp = mae90_t == 1 and ret_mae90 ? 1 : 0

    [mfe25_t, mfe25_nxt, mfe25_opp,
     mfe50_t, mfe50_nxt, mfe50_opp,
     mfe75_t, mfe75_nxt, mfe75_opp,
     mfe90_t, mfe90_opp,
     mae25_t, mae25_nxt, mae25_opp,
     mae50_t, mae50_nxt, mae50_opp,
     mae75_t, mae75_nxt, mae75_opp,
     mae90_t, mae90_opp]

// ─── Call f_record_session at each session end ────────────────────

if asia_end and not na(asia_o) and asia_o != 0
    [a1,a2,a3,a4,a5,a6,a7,a8,a9,a10,a11,a12,a13,a14,a15,a16,a17,a18,a19,a20,a21,a22] =
         f_record_session(asia_rng, asia_mfe, asia_mae,
                          asia_bull_mfe, asia_bull_mae,
                          asia_bear_mfe, asia_bear_mae,
                          asia_cls, asia_o, asia_h, asia_l, close[1],
                          asia_hit_mfe90, asia_ret_mfe90,
                          asia_hit_mfe75, asia_ret_mfe75,
                          asia_hit_mfe50, asia_ret_mfe50,
                          asia_hit_mae90, asia_ret_mae90,
                          asia_hit_mae75, asia_ret_mae75,
                          asia_hit_mae50, asia_ret_mae50)
    asia_sess_tot += 1
    if close[1] >= asia_o
        asia_bull_ct += 1
    else
        asia_bear_ct += 1
    asia_mfe25_tot += a1,  asia_mfe25_nxt += a2,  asia_mfe25_opp += a3
    asia_mfe50_tot += a4,  asia_mfe50_nxt += a5,  asia_mfe50_opp += a6
    asia_mfe75_tot += a7,  asia_mfe75_nxt += a8,  asia_mfe75_opp += a9
    asia_mfe90_tot += a10, asia_mfe90_opp += a11
    asia_mae25_tot += a12, asia_mae25_nxt += a13, asia_mae25_opp += a14

// SECTION 7 — LIVE SCORING
// Computes on every bar:
//   curr_sess     : which session is active
//   curr_open     : session open price
//   curr_dist     : signed distance from open in points
//   curr_pct_rank : percentile of current excursion vs history
//                   (vs MFE array if above open, MAE if below)
// ═══════════════════════════════════════════════════════════════════

     in_asia ? "ASIA" :
     in_lon  ? "LON"  :
     in_nyam ? "NYAM" :
     in_nypm ? "NYPM" :
     in_cust ? "CUST" : ""

     in_asia ? asia_o :
     in_lon  ? lon_o  :
     in_nyam ? nyam_o :
     in_nypm ? nypm_o :
     in_cust ? cust_o : float(na)

curr_dist = not na(curr_open) ? close - curr_open : float(na)

curr_mfe_arr =
     in_asia ? asia_mfe :
     in_lon  ? lon_mfe  :
     in_nyam ? nyam_mfe :
     in_nypm ? nypm_mfe :
     in_cust ? cust_mfe : array.new_float(0)

     in_asia ? asia_mae :
     in_lon  ? lon_mae  :
     in_nyam ? nyam_mae :
     in_nypm ? nypm_mae :
     in_cust ? cust_mae : array.new_float(0)

curr_excursion  = not na(curr_dist) ? math.abs(curr_dist) : float(na)
curr_score_arr  = not na(curr_dist) and curr_dist >= 0 ? curr_mfe_arr : curr_mae_arr
curr_pct_rank   = not na(curr_excursion) and array.size(curr_score_arr) >= 5 ?
     f_pct_rank(curr_score_arr, curr_excursion) : int(na)

// SECTION 8 — BAND PLOTTING
//
// Three layers per session:
//   A) Session open line  — dashed, session colour
//   B) Range bands        — symmetric H-L percentile offsets from open
//                           style/colour from Range Band inputs
//   C) MFE/MAE bands      — directional excursion offsets from open
//                           optionally split bull/bear
//                           style/colour from MFE/MAE inputs
//
// All objects drawn at session start, extended every bar, cleared
// when the next session begins.
// Labels track the right edge via label.set_x() every bar.
// ═══════════════════════════════════════════════════════════════════

// ─── Per-session object stores ────────────────────────────────────
// We use arrays of lines/labels/boxes so any number of levels
// can be added dynamically without declaring every object up front.

// ─── Clear / extend helpers ───────────────────────────────────────

f_clear(lns, lbs, bxs) =>
    for ln in lns
        line.delete(ln)
    array.clear(lns)
    for lb in lbs
        label.delete(lb)
    array.clear(lbs)
    for bx in bxs
        box.delete(bx)
    array.clear(bxs)

f_extend(lns, lbs, bxs) =>
    x2 = bar_index + i_proj_bars
    for ln in lns
        line.set_x2(ln, x2)
    for lb in lbs
        label.set_x(lb, x2)
    for bx in bxs
        box.set_right(bx, x2)

// ─── Add a single level line + label ─────────────────────────────

f_add_level(lns, lbs, x1, y, txt, col, ls, lw) =>
    array.push(lns, line.new(x1, y, x1 + 1, y,
         color=col, style=ls, width=lw))
    array.push(lbs, label.new(x1 + 1, y, " " + txt,
         style=label.style_none, textcolor=color.new(col, 30),
         color=color.new(color.white, 100),
         size=f_lbl_size(), text_font_family=font.family_monospace))

// ─── Add a band (two lines + optional fill box) ───────────────────

f_add_band(lns, lbs, bxs, x1, y_hi, y_lo, txt_hi, txt_lo, col, ls, lw) =>
    array.push(lns, line.new(x1, y_hi, x1 + 1, y_hi,
         color=col, style=ls, width=lw))
    array.push(lns, line.new(x1, y_lo, x1 + 1, y_lo,
         color=col, style=ls, width=lw))
    array.push(lbs, label.new(x1 + 1, y_hi, " " + txt_hi,
         style=label.style_none, textcolor=color.new(col, 30),
         color=color.new(color.white, 100),
         size=f_lbl_size(), text_font_family=font.family_monospace))
    array.push(lbs, label.new(x1 + 1, y_lo, " " + txt_lo,
         style=label.style_none, textcolor=color.new(col, 30),
         color=color.new(color.white, 100),
         size=f_lbl_size(), text_font_family=font.family_monospace))
    if i_rng_fills
        array.push(bxs, box.new(x1, y_hi, x1 + 1, y_lo,
             bgcolor=color.new(col, 93), border_width=0))

// ─── Draw all bands for one session ──────────────────────────────
// Called once at session start. Draws range bands then MFE/MAE bands.

f_draw_bands(lns, lbs, bxs, x1, s_open, slbl, col,
              rng_a, mfe_a, mae_a,
              bull_mfe_a, bull_mae_a,
              bear_mfe_a, bear_mae_a) =>

    rng_ls  = f_line_style(i_rng_style)
    exc_ls  = f_line_style(i_exc_style)

    // ── A) Range bands (symmetric, H-L percentiles) ──────────────
    if array.size(rng_a) >= 5

        p10 = f_pct_val(rng_a, 10)
        p25 = f_pct_val(rng_a, 25)
        p50 = f_pct_val(rng_a, 50)
        p75 = f_pct_val(rng_a, 75)
        p90 = f_pct_val(rng_a, 90)
        avg = array.size(rng_a) >= 2 ? array.avg(rng_a) : float(na)

        if i_rng_p10p90 and not na(p90)
            f_add_band(lns, lbs, bxs, x1,
                 s_open + p90, s_open - p90,
                 slbl + " P90", slbl + " -P90",
                 i_col_p10p90, rng_ls, i_rng_width)
        if i_rng_p10p90 and not na(p10)
            f_add_band(lns, lbs, bxs, x1,
                 s_open + p10, s_open - p10,
                 slbl + " P10", slbl + " -P10",
                 i_col_p10p90, rng_ls, i_rng_width)
        if i_rng_iqr and not na(p75)
            f_add_band(lns, lbs, bxs, x1,
                 s_open + p75, s_open - p75,
                 slbl + " P75", slbl + " -P75",
                 i_col_iqr, rng_ls, i_rng_width)
        if i_rng_iqr and not na(p25)
            f_add_band(lns, lbs, bxs, x1,
                 s_open + p25, s_open - p25,
                 slbl + " P25", slbl + " -P25",
                 i_col_iqr, rng_ls, i_rng_width)
        if i_rng_p50 and not na(p50)
            f_add_level(lns, lbs, x1, s_open + p50,
                 slbl + " P50",  i_col_p50, line.style_dashed, i_rng_width)
            f_add_level(lns, lbs, x1, s_open - p50,
                 slbl + " -P50", i_col_p50, line.style_dashed, i_rng_width)
        if i_rng_mean and not na(avg)
            f_add_level(lns, lbs, x1, s_open + avg,
                 slbl + " Mean",  i_col_mean, line.style_dashed, i_rng_width)
            f_add_level(lns, lbs, x1, s_open - avg,
                 slbl + " -Mean", i_col_mean, line.style_dashed, i_rng_width)

    // ── B) MFE / MAE bands (directional excursion) ───────────────
    if i_show_exc

        if i_bull_bear
            // Bull session MFE (upside) levels
            if array.size(bull_mfe_a) >= 5
                for p in array.from(25, 50, 75, 90)
                    v = f_pct_val(bull_mfe_a, p)
                    if not na(v)
                        f_add_level(lns, lbs, x1, s_open + v,
                             slbl + " Bull MFE" + str.tostring(p),
                             i_col_mfe, exc_ls, i_exc_width)
            // Bull session MAE (downside) levels
            if array.size(bull_mae_a) >= 5
                for p in array.from(25, 50, 75, 90)
                    v = f_pct_val(bull_mae_a, p)
                    if not na(v)
                        f_add_level(lns, lbs, x1, s_open - v,
                             slbl + " Bull MAE" + str.tostring(p),
                             i_col_mae, exc_ls, i_exc_width)
            // Bear session MFE (downside) levels
            if array.size(bear_mfe_a) >= 5
                for p in array.from(25, 50, 75, 90)
                    v = f_pct_val(bear_mfe_a, p)
                    if not na(v)
                        f_add_level(lns, lbs, x1, s_open - v,
                             slbl + " Bear MFE" + str.tostring(p),
                             i_col_mfe, exc_ls, i_exc_width)
            // Bear session MAE (upside) levels
            if array.size(bear_mae_a) >= 5
                for p in array.from(25, 50, 75, 90)
                    v = f_pct_val(bear_mae_a, p)
                    if not na(v)
                        f_add_level(lns, lbs, x1, s_open + v,
                             slbl + " Bear MAE" + str.tostring(p),
                             i_col_mae, exc_ls, i_exc_width)
        else
            // Combined MFE (above open)
            if array.size(mfe_a) >= 5
                for p in array.from(25, 50, 75, 90)
                    v = f_pct_val(mfe_a, p)
                    if not na(v)
                        f_add_level(lns, lbs, x1, s_open + v,
                             slbl + " MFE" + str.tostring(p),
                             i_col_mfe, exc_ls, i_exc_width)
            // Combined MAE (below open)
            if array.size(mae_a) >= 5

                    if not na(v)

f_tbl1_row(tbl, row, name, rng_a, bull, bear, sess_col, sz) =>

         text_size=sz, text_color=sess_col, bgcolor=bg, text_font_family=fnt)
    table.cell(tbl, 1, row, str.tostring(array.size(rng_a)),
         text_size=sz, text_color=c_tc, bgcolor=bg, text_font_family=fnt)
    table.cell(tbl, 2, row, f_fmt_p(f_pct_val(rng_a, 25)),
         text_size=sz, text_color=c_tc, bgcolor=bg, text_font_family=fnt)
    table.cell(tbl, 3, row, f_fmt_p(f_pct_val(rng_a, 50)),
         text_size=sz, text_color=c_tc, bgcolor=bg, text_font_family=fnt)

         text_size=sz, text_color=c_tc, bgcolor=bg, text_font_family=fnt)
    table.cell(tbl, 5, row, f_fmt_p(f_pct_val(rng_a, 90)),
         text_size=sz, text_color=c_tc, bgcolor=bg, text_font_family=fnt)
    table.cell(tbl, 6, row,
         n >= 2 ? f_fmt_p(array.avg(rng_a)) : "—",
         text_size=sz, text_color=c_tc, bgcolor=bg, text_font_family=fnt)
    table.cell(tbl, 7, row,
         str.tostring(bull) + " / " + str.tostring(bear),
         text_size=sz, text_color=c_dim, bgcolor=bg, text_font_family=fnt)
