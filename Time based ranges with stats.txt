// Median calculation
f_median(float[] arr) =>
    var float result = na
    if arr.size() > 0
        sorted = array.copy(arr)
        array.sort(sorted)
        if arr.size() % 2 == 0
            result := (sorted.get(arr.size()/2 - 1) + sorted.get(arr.size()/2)) / 2
        else
            result := sorted.get(arr.size()/2)
    result

// Get stats for day of week
get_dow_stats(ses_stats S, int dow, bool D) =>
    sum = 0.0
    count = 0
    for [i, s] in S.day
        if D ? s == dow : true
            sum += (dist_show_pcts ? S.range_pcts.get(i) : S.ranges.get(i))
            count += 1
    result = str.tostring(math.round(sum / count, 2))+(dist_show_pcts?"%":"")
    result += sample_size ? (" ["+str.tostring(count)+"]") : ""

// FIXED: Updated median/mean calculation to work with overall data
get_dow_median_mean(ses_stats S, bool is_median) =>
    result = is_median ? (S.ranges.size() > 0 ? f_median(S.ranges) : 0) : (S.ranges.size() > 0 ? S.ranges.avg() : 0)
    str.tostring(math.round(result, 2))

get_hit_stats(ses_stats S, int dow, string target, bool D) =>
    sum = 0
    count = 0
    target_array = target == "Open" ? S.hit_open : target == "High" ? S.hit_likely_high : S.hit_likely_low
    for [i, s] in S.day
        if D ? s == dow : true
            sum += target_array.get(i) ? 1 : 0
            count += 1
    percentage = count > 0 ? (sum / count) * 100 : 0
    result = str.tostring(math.round(percentage, 1)) + "%"
    result += sample_size ? (" ["+str.tostring(count)+"]") : ""
    result

// Get average open-to-high/low for day of week
get_dow_high_low(ses_stats S, int dow, bool is_high) =>
    dow_values = array.new_float(0)
    for [i, d] in S.day
        if d == dow
            dow_values.push(is_high ? S.open_to_high.get(i) : S.open_to_low.get(i))
    result = dow_values.size() > 0 ? dow_values.avg() : 0
    result

// FIXED: Updated session stats method with improved average calculation
method update_ses_stats(ses_stats S, bool t, bool enabled) =>
    if enabled
        if t and not t[1]
            S.count += 1
            S.cur_hi := high
            S.cur_lo := low
            S.cur_op := open
            S.day.unshift(dayofweek)
            S.ranges.unshift(0)
            S.range_pcts.unshift(0)
            S.open_to_high.unshift(0)
            S.open_to_low.unshift(0)
            S.hit_open.unshift(false)
            S.hit_likely_high.unshift(false)
            S.hit_likely_low.unshift(false)
            S.has_hit_open := false
            S.has_hit_likely_high := false
            S.has_hit_likely_low := false
            
            // FIXED: Calculate average for current day of week specifically
            current_dow_ranges = array.new_float(0)
            current_dow_range_pcts = array.new_float(0)
            if S.day.size() > 1  // Only if we have historical data
                for [i, d] in S.day
                    if i > 0 and d == dayofweek  // Skip current session (index 0) and match day of week
                        current_dow_ranges.push(S.ranges.get(i))
                        current_dow_range_pcts.push(S.range_pcts.get(i))
            
            avg = if current_dow_ranges.size() > 0
                dist_show_pcts ? current_dow_range_pcts.avg() : current_dow_ranges.avg()
            else
                // Fallback to overall average if no historical data for this day
                if S.ranges.size() > 1
                    dist_show_pcts ? S.range_pcts.avg() : S.ranges.avg()
                else
                    0.0
            
            // High-low box
            S.boxes.unshift(box.new(time, S.cur_hi, time, S.cur_lo, xloc = xloc.bar_time, bgcolor = show_dist_boxes and fill_dist_boxes ? color.new(S.obj_color, 80) : na, border_color = show_dist_boxes ? S.obj_color : na))
            // Draw open line
            S.open_lines.unshift(line.new(time, S.cur_op, time, S.cur_op, xloc = xloc.bar_time, color = show_open_lines ? S.obj_color : na, style = line.style_solid))
            // Calculate average high and low for current day of week
            avg_high = get_dow_high_low(S, dayofweek, true)
            avg_low = get_dow_high_low(S, dayofweek, false)
            // Define likely high and low boxes based on average high/low
            high_top = S.cur_op + avg_high
            high_bottom = S.cur_op
            low_top = S.cur_op
            low_bottom = S.cur_op + avg_low
            // Likely high box
            S.likely_high_boxes.unshift(box.new(time, high_top, time, high_bottom, xloc = xloc.bar_time, bgcolor = show_likely_boxes and fill_likely_boxes ? color.new(S.obj_color, 80) : na, border_color = show_likely_boxes ? S.obj_color : na, border_style = line.style_dashed))
            // Likely low box
            S.likely_low_boxes.unshift(box.new(time, low_top, time, low_bottom, xloc = xloc.bar_time, bgcolor = show_likely_boxes and fill_likely_boxes ? color.new(S.obj_color, 80) : na, border_color = show_likely_boxes ? S.obj_color : na, border_style = line.style_dashed))
            // Draw average high/low lines
            S.high_lines.unshift(line.new(time, S.cur_op + avg_high, time, S.cur_op + avg_high, xloc = xloc.bar_time, color = show_high_low_lines ? expected_range_line_color : na, style = line.style_solid))
