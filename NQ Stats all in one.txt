    (high + low) / 2

// Session Range Function
get_range(session, session_name, session_css, high_array, low_array)=>
    var t = 0 
    var max = high
    var min = low
    var box bx = na
    var label lbl = na 
    var int start_idx = na
    var float vol = 0.0
    var float atr_sum = 0.0
    var int bar_count = 0
    
    if session > session[1]
        t := time
        max := high
        min := low
        start_idx := bar_index
        vol := 0.0
        atr_sum := 0.0
        bar_count := 0
        bx := box.new(bar_index, max, bar_index, min, bgcolor=color.new(session_css, bg_transp), border_color=show_outline ? session_css : na, border_style=line.style_dotted)
        if show_txt
            lbl := label.new(t, max, session_name, xloc=xloc.bar_time, textcolor=session_css, style=label.style_label_down, color=color.new(color.white, 100), size=size.tiny)
    
    if session and session == session[1]
        max := math.max(high, max)
        min := math.min(low, min)
        vol := vol + volume
        atr_sum := atr_sum + (high - low)
        bar_count := bar_count + 1
        box.set_top(bx, max)
        box.set_rightbottom(bx, bar_index, min)
        if show_txt
            label.set_xy(lbl, int(math.avg(t, time)), max)
    
    if session[1] and not session
        array.push(high_array, max)
        array.push(low_array, min)
    
    [session ? na : max, session ? na : min, start_idx, session ? na : vol, session ? na : (bar_count > 0 ? atr_sum / bar_count : na)]

// Fibonacci Midpoint
get_fibonacci_midpoint(high_price, low_price) =>
    (high_price + low_price) / 2

// Session Position Classification
get_session_position(curr_high, curr_low, prev_high, prev_low) =>
    if curr_high > prev_high and curr_low < prev_low
        "Full Engulfing"
    else if curr_high > prev_high and curr_low >= prev_low
        "Partial Engulfing Up"
    else if curr_high <= prev_high and curr_low < prev_low
        "Partial Engulfing Down"
    else
        "Full Inside"

// Table Position Function
get_table_position(pos) =>
    result = switch pos
        "Bottom Center" => position.bottom_center
        "Bottom Left" => position.bottom_left
        "Bottom Right" => position.bottom_right
        "Middle Center" => position.middle_center
        "Middle Left" => position.middle_left
        "Middle Right" => position.middle_right
        "Top Center" => position.top_center
        "Top Left" => position.top_left
        "Top Right" => position.top_right

// Table Text Size Function
get_table_text_size(size) =>
    result = switch size
        'Auto' => size.auto
        'Tiny' => size.tiny
        'Small' => size.small
        'Normal' => size.normal
        'Large' => size.large
        'Huge' => size.huge

// Function to Get Filter Levels
get_filter_level(level, sesa_high, sesa_low, sesb_high, sesb_low, sesc_high, sesc_low, rth_open) =>
    sesa_mid = get_fibonacci_midpoint(sesa_high, sesa_low)
    sesb_mid = get_fibonacci_midpoint(sesb_high, sesb_low)
    sesc_mid = get_fibonacci_midpoint(sesc_high, sesc_low)
    prev_day_high = request.security(syminfo.tickerid, "D", high[1], lookahead=barmerge.lookahead_on)
    prev_day_low = request.security(syminfo.tickerid, "D", low[1], lookahead=barmerge.lookahead_on)
    prev_day_mid = (prev_day_high + prev_day_low) / 2
    prev_week_high = request.security(syminfo.tickerid, "W", high[1], lookahead=barmerge.lookahead_on)
    prev_week_low = request.security(syminfo.tickerid, "W", low[1], lookahead=barmerge.lookahead_on)
    prev_week_mid = (prev_week_high + prev_week_low) / 2
    
    switch level
        "London high" => sesb_high
        "London low" => sesb_low
        "London mid" => sesb_mid
        "Asia high" => sesc_high
        "Asia low" => sesc_low
        "Asia mid" => sesc_mid
        "Midnight open" => openMidnight
        "Other time based open" => open0800
        "Previous day high" => prev_day_high
        "Previous day low" => prev_day_low
        "Previous day 50%" => prev_day_mid
        "Previous week high" => prev_week_high
        "Previous week low" => prev_week_low
        "Previous week 50%" => prev_week_mid

            sesb_low = array.get(sesb_lows, i)
            sesc_high = array.get(sesc_highs, i)
            sesc_low = array.get(sesc_lows, i)

            yest_rth_low = i == 0 ? yest_rth_lo : array.get(sesa_lows, math.min(i + 1, min_size - 1))    // Proxy for yesterday’s RTH low

                    filt_lon_part_down_ny_part_down := filt_lon_part_down_ny_part_down + (ny_vs_london == "Partial Engulfing Down" ? 1 : 0)
                    filt_lon_part_down_ny_inside := filt_lon_part_down_ny_inside + (ny_vs_london == "Full Inside" ? 1 : 0)
                
                if london_vs_asia == "Full Inside"
                    filt_lon_inside_ny_full := filt_lon_inside_ny_full + (ny_vs_london == "Full Engulfing" ? 1 : 0)
                    filt_lon_inside_ny_part_up := filt_lon_inside_ny_part_up + (ny_vs_london == "Partial Engulfing Up" ? 1 : 0)
                    filt_lon_inside_ny_part_down := filt_lon_inside_ny_part_down + (ny_vs_london == "Partial Engulfing Down" ? 1 : 0)
                    filt_lon_inside_ny_inside := filt_lon_inside_ny_inside + (ny_vs_london == "Full Inside" ? 1 : 0)
