// Get color based on MAE percentage
get_color_for_mae(mae_pct) =>
    color result = color.gray
    
    if mae_pct <= 0.25
        result := color.rgb(33, 150, 243)      // Blue
    else if mae_pct <= 0.30
        result := color.rgb(156, 39, 176)      // Purple
    else if mae_pct <= 0.35
        result := color.rgb(255, 87, 34)       // Deep Orange
    else if mae_pct <= 0.40
        result := color.rgb(0, 150, 136)       // Teal
    else if mae_pct <= 0.45
        result := color.rgb(255, 152, 0)       // Orange
    else if mae_pct <= 0.50
        result := color.rgb(233, 30, 99)       // Pink
    else
        result := color.rgb(96, 125, 139)      // Blue Grey
    
    result

// Get color for the live distance cell
// Green  = price is within MAE (level very much in play)
// Yellow = price is within 2× MAE (approaching, worth watching)
// Orange = price is within 3× MAE (possible but stretched)
// Gray   = price is beyond 3× MAE (unlikely / far away)
get_distance_color(dist_pct, mae_pct) =>
    color result = color.rgb(180, 180, 180)  // default gray
    if dist_pct <= mae_pct
        result := color.rgb(76, 175, 80)     // Green  – within MAE
    else if dist_pct <= mae_pct * 2
        result := color.rgb(255, 235, 59)    // Yellow – within 2× MAE
    else if dist_pct <= mae_pct * 3
        result := color.rgb(255, 152, 0)     // Orange – within 3× MAE
    result

get_distance_text_color(dist_pct, mae_pct) =>
    color result = color.rgb(64, 64, 64)
    if dist_pct <= mae_pct
        result := color.white
    else if dist_pct <= mae_pct * 2
        result := color.rgb(64, 64, 64)
    else if dist_pct <= mae_pct * 3
        result := color.rgb(64, 64, 64)
    result

// ---------------------------------------------- Session Ranges --------------------------------------------------

get_range(session, session_css)=>
    var max = high
    var min = low
    var sess_open = open
    var box bx = na
    
    if session > session[1]
        max := high
        min := low
        sess_open := open
        bx := box.new(bar_index, max, bar_index, min, 
             bgcolor=color.new(session_css, bg_transp), 
             border_color=show_outline ? session_css : na, 
             border_style=line.style_dotted)
    
    if session and session == session[1]
        max := math.max(high, max)
        min := math.min(low, min)
        box.set_top(bx, max)
        box.set_rightbottom(bx, bar_index, min)
    
    [session ? na : max, session ? na : min, sess_open]

// Track hourly opens and ranges
track_hourly(is_hour, prev_is_hour) =>
    var float hour_open = na
    var float hour_max = na
    var float hour_min = na
    
    if is_hour and is_hour > prev_is_hour
        hour_open := open
        hour_max := high
        hour_min := low
    else if is_hour
        hour_max := math.max(high, hour_max)
        hour_min := math.min(low, hour_min)
    
    [hour_open, hour_max, hour_min]

// Clear old lines and labels
clear_hourly_levels() =>
    if array.size(hourly_lines) > 0
        for i = 0 to array.size(hourly_lines) - 1
            line.delete(array.get(hourly_lines, i))
        array.clear(hourly_lines)
    if array.size(hourly_labels) > 0
        for i = 0 to array.size(hourly_labels) - 1
            label.delete(array.get(hourly_labels, i))
        array.clear(hourly_labels)
    if array.size(hourly_prices) > 0
        array.clear(hourly_prices)

// Draw a level
draw_level(price, level_name, hit_rate, mae_pct, ny_open_bar, ny_open_price, is_midpoint) =>
    if hit_rate >= min_hit_rate
        style      = get_line_style(line_style_input)
        lbl_size   = get_label_size(label_size_input)
        level_color = get_color_for_mae(mae_pct)

        new_line = line.new(ny_open_bar, price, ny_open_bar, price,
             color=level_color, style=style, width=line_width)
        array.push(hourly_lines, new_line)

        label_text = str.format("{0}: {1,number,#.#}%", level_name, hit_rate)
        new_label  = label.new(ny_open_bar, price, label_text,
             style=label.style_none, textcolor=level_color, size=lbl_size,
             textalign=text.align_left, text_font_family=font.family_monospace)
        array.push(hourly_labels, new_label)
        array.push(hourly_prices, price)

// Check if levels have been hit and remove them
check_and_remove_hit_levels() =>
    if remove_on_hit and array.size(hourly_prices) > 0
        i = array.size(hourly_prices) - 1
        while i >= 0
            level_price = array.get(hourly_prices, i)
            if low <= level_price and level_price <= high
                line.delete(array.get(hourly_lines, i))
                label.delete(array.get(hourly_labels, i))
                array.remove(hourly_lines, i)
                array.remove(hourly_labels, i)
                array.remove(hourly_prices, i)
            i := i - 1
