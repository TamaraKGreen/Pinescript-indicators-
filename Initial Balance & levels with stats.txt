inSession(sess) =>
    na(time(timeframe.period, sess)) == false

get_line_style(s) =>
    s == "Solid" ? line.style_solid : s == "Dotted" ? line.style_dotted : line.style_dashed

get_levels(n) =>
    h = high[1]
    l = low[1]
    for i = 1 to n
        if low[i] < l
            l := low[i]
        if high[i] > h
            h := high[i]
    [h, l, (h + l) / 2]

get_table_position(pos) =>
    result = switch pos
        "Bottom Center" => position.bottom_center
        "Bottom Left" => position.bottom_left
        "Bottom Right" => position.bottom_right
        "Middle Center" => position.middle_center
        "Middle Left" => position.middle_left
        "Middle Right" => position.middle_right
        "Top Center" => position.top_center
        "Top Left" => position.top_left
        "Top Right" => position.top_right

get_table_text_size(size) =>
    result = switch size
        'Auto' => size.auto
        'Tiny' => size.tiny
        'Small' => size.small
        'Normal' => size.normal
        'Large' => size.large
        'Huge' => size.huge

labelSize = (label_size == "Huge") ? size.huge :
     (label_size == "Large") ? size.large :

     (label_size == "Tiny") ? size.tiny :

var offset = 0
var float ib_delta = na
var float current_high_ext = na
var float current_low_ext = na
var float ib_high = na
var float ib_low = na
var bool ib_formed = false

ins = inSession(ib_session)

bgcolor(show_ib_calculation_area and ins ? #673ab730 : na, title="IB calculation zone")

if ins
    offset += 1

if ins[1] and not ins
    [h, l, m] = get_levels(offset)
    ib_delta := h - l
    array.push(delta_history, ib_delta)
    
    // Store IB levels and reset extension tracking
    ib_high := h
    ib_low := l
    ib_formed := true
    current_high_ext := h
    current_low_ext := l

    // 1. Analyze IB Range Z-Score
    if array.size(delta_history) >= 2
        mean_delta = array.avg(delta_history)
        sd_delta = array.stdev(delta_history)
        z_score_ib_delta := (ib_delta - mean_delta) / sd_delta
        line.delete(z_score_ib_line)
        label.delete(z_score_ib_label)
        if show_intermediate_levels
            z_score_ib_line := line.new(bar_index[offset], m, bar_index, m, color=color.purple, style=line.style_dashed, width=1)
            if show_labels
                z_score_ib_label := label.new(bar_index[offset], m, "Z-Score 50% IB: " + str.tostring(z_score_ib_delta, "#.##"), 
                                             style=label.style_none, textcolor=color.purple, size=labelSize)

    line.set_extend(ibh, extend.none)
    line.set_extend(ibl, extend.none)
    if show_intermediate_levels
        line.set_extend(ibm, extend.none)
        if fib1_show
            line.set_extend(fib_ln_1, extend.none)
        if fib2_show
            line.set_extend(fib_ln_2, extend.none)
        if fib3_show
            line.set_extend(fib_ln_3, extend.none)
        if fib4_show
            line.set_extend(fib_ln_4, extend.none)
    
    if show_labels
        if only_current_levels
            label.delete(labelh)
            label.delete(labell)
            if show_intermediate_levels
                label.delete(labelm)
        labelh := label.new(bar_index[offset], h, text="IBH: " + str.tostring(h), style=label.style_none, textcolor=ib_high_color, size=labelSize)
        labell := label.new(bar_index[offset], l, text="IBL: " + str.tostring(l), style=label.style_none, textcolor=ib_low_color, size=labelSize)
        
        // Fixed: Only show intermediate level labels when both show_labels AND show_intermediate_levels are true
        if show_intermediate_levels
            if only_current_levels
                label.delete(labelm)
                if fib1_show
                    label.delete(labelfib1)
                if fib2_show
                    label.delete(labelfib2)
                if fib3_show
                    label.delete(labelfib3)
                if fib4_show
                    label.delete(labelfib4)
            labelm := label.new(bar_index[offset], m, text="50%: " + str.tostring(m), style=label.style_none, textcolor=mid_color, size=labelSize)

    // Fixed: Only create Fibonacci labels when respective show flags are true AND show_labels AND show_intermediate_levels are true
    if show_labels and show_intermediate_levels
        if fib1_show
            labelfib1 := label.new(bar_index[offset], l + (h - l) * fib1, text=str.tostring(fib1), style=label.style_none, textcolor=fib_color, size=labelSize)
        if fib2_show
            labelfib2 := label.new(bar_index[offset], l + (h - l) * fib2, text=str.tostring(fib2), style=label.style_none, textcolor=fib_color, size=labelSize)
        if fib3_show
            labelfib3 := label.new(bar_index[offset], l + (h - l) * fib3, text=str.tostring(fib3), style=label.style_none, textcolor=fib_color, size=labelSize)
        if fib4_show
            labelfib4 := label.new(bar_index[offset], l + (h - l) * fib4, text=str.tostring(fib4), style=label.style_none, textcolor=fib_color, size=labelSize)
    
    if fill_ib_areas
        if only_current_zone
            box.delete(ib_area)
        ib_area := box.new(bar_index[offset], h, bar_index, l, bgcolor=fill_ib_color, border_color=#00000000)
    if only_current_levels
        line.delete(ibh)
        line.delete(ibl)
        if show_intermediate_levels
            line.delete(ibm)
            if fib1_show
                line.delete(fib_ln_1)
            if fib2_show
                line.delete(fib_ln_2)
            if fib3_show
                line.delete(fib_ln_3)
            if fib4_show
                line.delete(fib_ln_4)
     
    ibh := line.new(bar_index[offset], h, bar_index, h, color=ib_high_color, extend=ext, width=ib_high_width, style=get_line_style(ib_high_style))
    ibl := line.new(bar_index[offset], l, bar_index, l, color=ib_low_color, extend=ext, width=ib_low_width, style=get_line_style(ib_low_style))
    if show_intermediate_levels
        ibm := line.new(bar_index[offset], m, bar_index, m, color=mid_color, style=get_line_style(mid_style), extend=ext, width=mid_width)
        if fib1_show
            fib_ln_1 := line.new(bar_index[offset], l + (h - l) * fib1, bar_index, l + (h - l) * fib1, color=fib_color, style=get_line_style(fib_style), extend=ext, width=fib_width)
        if fib2_show
            fib_ln_2 := line.new(bar_index[offset], l + (h - l) * fib2, bar_index, l + (h - l) * fib2, color=fib_color, style=get_line_style(fib_style), extend=ext, width=fib_width)
        if fib3_show
            fib_ln_3 := line.new(bar_index[offset], l + (h - l) * fib3, bar_index, l + (h - l) * fib3, color=fib_color, style=get_line_style(fib_style), extend=ext, width=fib_width)
        if fib4_show    
            fib_ln_4 := line.new(bar_index[offset], l + (h - l) * fib4, bar_index, l + (h - l) * fib4, color=fib_color, style=get_line_style(fib_style), extend=ext, width=fib_width)    
    
    // Plot extensions
    if show_extensions
        mean_high_ext = array.avg(high_extensions)
        median_high_ext = array.median(high_extensions)
        mean_low_ext = array.avg(low_extensions)
        median_low_ext = array.median(low_extensions)
        high_sd = array.stdev(high_extensions)
        low_sd = array.stdev(low_extensions)
        
        line.delete(mean_high_ext_line)
        line.delete(median_high_ext_line)
        line.delete(mean_low_ext_line)
        line.delete(median_low_ext_line)
        line.delete(sd1_high_line)
        line.delete(sd2_high_line)
        line.delete(sd3_high_line)
        line.delete(sd4_high_line)
        line.delete(sd1_low_line)
        line.delete(sd2_low_line)
        line.delete(sd3_low_line)
        line.delete(sd4_low_line)
        line.delete(sd_high_line)
        line.delete(sd_low_line)
        line.delete(p25_high_line)
        line.delete(p75_high_line)
        line.delete(p25_low_line)
        line.delete(p75_low_line)
        
        if show_mean_high and not na(mean_high_ext)
            mean_high_ext_line := line.new(bar_index[offset], ib_high + mean_high_ext, bar_index, ib_high + mean_high_ext, 
                                          color=ext_color, extend=ext, style=get_line_style(ext_style), width=ext_width)
            if show_labels
                label.new(bar_index[offset], ib_high + mean_high_ext, "Mean High Ext: " + str.tostring(mean_high_ext, "#.##"),

        if show_median_high and not na(median_high_ext)
            median_high_ext_line := line.new(bar_index[offset], ib_high + median_high_ext, bar_index, ib_high + median_high_ext,
                                             color=ext_color, extend=ext, style=get_line_style(ext_style), width=ext_width)
            if show_labels
                label.new(bar_index[offset], ib_high + median_high_ext, "Median High Ext: " + str.tostring(median_high_ext, "#.##"), 
                         style=label.style_none, textcolor=ext_color, size=labelSize)
                
        if show_mean_low and not na(mean_low_ext)
            mean_low_ext_line := line.new(bar_index[offset], ib_low - mean_low_ext, bar_index, ib_low - mean_low_ext, 
                                         color=ext_color, extend=ext, style=get_line_style(ext_style), width=ext_width)
            if show_labels
                label.new(bar_index[offset], ib_low - mean_low_ext, "Mean Low Ext: " + str.tostring(mean_low_ext, "#.##"),

        if show_median_low and not na(median_low_ext)
            median_low_ext_line := line.new(bar_index[offset], ib_low - median_low_ext, bar_index, ib_low - median_low_ext, 
                                           color=ext_color, extend=ext, style=get_line_style(ext_style), width=ext_width)
            if show_labels
                label.new(bar_index[offset], ib_low - median_low_ext, "Median Low Ext: " + str.tostring(median_low_ext, "#.##"), 
                         style=label.style_none, textcolor=ext_color, size=labelSize)
        
        // Add percentile lines for break extensions
        if show_percentiles and show_p25_high and array.size(high_break_extensions) > 0
            high_break_25th = array.percentile_linear_interpolation(high_break_extensions, 25)
            
            if not na(high_break_25th)
                p25_high_line := line.new(bar_index[offset], ib_high + high_break_25th, bar_index, ib_high + high_break_25th, 
                                         color=percentile_color, extend=ext, style=get_line_style(percentile_style), width=percentile_width)
                if show_labels
                    label.new(bar_index[offset], ib_high + high_break_25th, "25th %ile", 
                             style=label.style_none, textcolor=percentile_color, size=labelSize)
        
        if show_percentiles and show_p75_high and array.size(high_break_extensions) > 0
            high_break_75th = array.percentile_linear_interpolation(high_break_extensions, 75)
            
            if not na(high_break_75th)
                p75_high_line := line.new(bar_index[offset], ib_high + high_break_75th, bar_index, ib_high + high_break_75th, 
                                         color=percentile_color, extend=ext, style=get_line_style(percentile_style), width=percentile_width)
                if show_labels
                    label.new(bar_index[offset], ib_high + high_break_75th, "75th %ile", 
                             style=label.style_none, textcolor=percentile_color, size=labelSize)
        
        if show_percentiles and show_p25_low and array.size(low_break_extensions) > 0
            low_break_25th = array.percentile_linear_interpolation(low_break_extensions, 25)
            
            if not na(low_break_25th)
                p25_low_line := line.new(bar_index[offset], ib_low - low_break_25th, bar_index, ib_low - low_break_25th,
                                         color=percentile_color, extend=ext, style=get_line_style(percentile_style), width=percentile_width)
                if show_labels
                    label.new(bar_index[offset], ib_low - low_break_25th, "25th %ile", 
                             style=label.style_none, textcolor=percentile_color, size=labelSize)
        
        if show_percentiles and show_p75_low and array.size(low_break_extensions) > 0
            low_break_75th = array.percentile_linear_interpolation(low_break_extensions, 75)
            
            if not na(low_break_75th)
                p75_low_line := line.new(bar_index[offset], ib_low - low_break_75th, bar_index, ib_low - low_break_75th,
                                         color=percentile_color, extend=ext, style=get_line_style(percentile_style), width=percentile_width)
                if show_labels
                    label.new(bar_index[offset], ib_low - low_break_75th, "75th %ile", 
                             style=label.style_none, textcolor=percentile_color, size=labelSize)
                        
        // SD Levels
        if show_sd1 and not na(high_sd)
            sd1_high_line := line.new(bar_index[offset], ib_high + mean_high_ext + high_sd, bar_index, ib_high + mean_high_ext + high_sd, 
                                     color=ext_color, extend=ext, style=get_line_style(ext_style), width=ext_width)
            if show_labels
                label.new(bar_index[offset], ib_high + mean_high_ext + high_sd, "1 SD High: " + str.tostring(mean_high_ext + high_sd, "#.##"), 
                         style=label.style_none, textcolor=ext_color, size=labelSize)
            sd1_low_line := line.new(bar_index[offset], ib_low - mean_low_ext - low_sd, bar_index, ib_low - mean_low_ext - low_sd,
