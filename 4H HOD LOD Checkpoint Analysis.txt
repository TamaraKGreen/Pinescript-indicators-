update_candle(candle_data c, bool in_session) =>
    if in_session and not in_session[1]
        c.hi := high
        c.lo := low
        c.op := open
        c.start_bar := bar_index
        c.completed := false
        
        if show_4h_boxes
            c.candle_box := box.new(bar_index, high, bar_index, low,
                 bgcolor=c.box_color,
                 border_color=color.new(c.box_color, 0),
                 border_width=1)
    
    if in_session and in_session[1]
        if na(c.hi)
            c.hi := high
        else
            c.hi := math.max(c.hi, high)
        
        if na(c.lo)
            c.lo := low
        else
            c.lo := math.min(c.lo, low)
        
        if na(c.start_bar)
            c.start_bar := bar_index
            c.op := open
        
        if show_4h_boxes and na(c.candle_box)
            c.candle_box := box.new(c.start_bar, c.hi, bar_index, c.lo,
                 bgcolor=c.box_color,
                 border_color=color.new(c.box_color, 0),
                 border_width=1)
        
        if show_4h_boxes and not na(c.candle_box)
            box.set_top(c.candle_box, c.hi)
            box.set_bottom(c.candle_box, c.lo)
            box.set_right(c.candle_box, bar_index)
    
    if not in_session and in_session[1]
        c.cl := close[1]
        c.end_bar := bar_index - 1
        c.completed := true
        
        if not na(c.cl) and not na(c.op)
            c.is_bullish := c.cl > c.op
            c.is_bearish := c.cl < c.op
        
        if show_4h_boxes and not na(c.candle_box)
            box.set_right(c.candle_box, c.end_bar)
            array.push(all_boxes, c.candle_box)

// Function to count eliminations and determine structure
count_and_lock_state() =>
    int num_h_elim = 0
    int num_l_elim = 0
    
    // Count eliminations based on completed candles
    if c_6pm.completed and c_6pm.high_eliminated
        num_h_elim += 1
    if c_6pm.completed and c_6pm.low_eliminated
        num_l_elim += 1
    
    if c_10pm.completed and c_10pm.high_eliminated
        num_h_elim += 1
    if c_10pm.completed and c_10pm.low_eliminated
        num_l_elim += 1
    
    if c_2am.completed and c_2am.high_eliminated
        num_h_elim += 1
    if c_2am.completed and c_2am.low_eliminated
        num_l_elim += 1
    
    if c_6am.completed and c_6am.high_eliminated
        num_h_elim += 1
    if c_6am.completed and c_6am.low_eliminated
        num_l_elim += 1
    
    if c_10am.completed and c_10am.high_eliminated
        num_h_elim += 1
    if c_10am.completed and c_10am.low_eliminated
        num_l_elim += 1
    
    // Determine structure for highs
    string struct_h = "none"
    if num_h_elim > 0
        bool is_seq_h = true
        if num_h_elim >= 1 and c_10pm.high_eliminated and not c_6pm.high_eliminated
            is_seq_h := false
        if num_h_elim >= 2 and c_2am.high_eliminated and not c_10pm.high_eliminated
            is_seq_h := false
        if num_h_elim >= 3 and c_6am.high_eliminated and not c_2am.high_eliminated
            is_seq_h := false
        if num_h_elim >= 4 and c_10am.high_eliminated and not c_6am.high_eliminated
            is_seq_h := false
        struct_h := is_seq_h ? "sequential" : "skip"
    
    // Determine structure for lows
    string struct_l = "none"
    if num_l_elim > 0
        bool is_seq_l = true
        if num_l_elim >= 1 and c_10pm.low_eliminated and not c_6pm.low_eliminated
            is_seq_l := false
        if num_l_elim >= 2 and c_2am.low_eliminated and not c_10pm.low_eliminated
            is_seq_l := false
        if num_l_elim >= 3 and c_6am.low_eliminated and not c_2am.low_eliminated
            is_seq_l := false
        if num_l_elim >= 4 and c_10am.low_eliminated and not c_6am.low_eliminated
            is_seq_l := false
        struct_l := is_seq_l ? "sequential" : "skip"
    
    [num_h_elim, num_l_elim, struct_h, struct_l]

// Lock state at checkpoints
