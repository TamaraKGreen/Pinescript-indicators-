// Custom function to check if timestamp is at 00:00 EST
isNYMidnightSession(timestamp) =>
    h = hour(timestamp, "America/New_York")
    m = minute(timestamp, "America/New_York")
    h == 0 and m == 0

// Custom function to check if timestamp is at 18:00 EST (futures session start)
isFuturesSessionStart(timestamp) =>
    h = hour(timestamp, "America/New_York")
    m = minute(timestamp, "America/New_York")
    h == 18 and m == 0

// Get NY midnight timestamp for a given date
getNYMidnightTimestamp(timestamp) =>
    t_year = year(timestamp, "America/New_York")
    t_month = month(timestamp, "America/New_York")
    t_day = dayofmonth(timestamp, "America/New_York")
    timestamp("America/New_York", t_year, t_month, t_day, 0, 0, 0)

// Get futures session start timestamp (18:00 EST) for a given date
getFuturesSessionTimestamp(timestamp) =>
    t_year = year(timestamp, "America/New_York")
    t_month = month(timestamp, "America/New_York")
    t_day = dayofmonth(timestamp, "America/New_York")
    // For futures, the daily session starts at 18:00 EST
    // If current time is before 18:00, use previous day's 18:00
    currentHour = hour(timestamp, "America/New_York")
    if currentHour < 18
        // Use previous day's 18:00
        prevDay = timestamp - 86400000  // Subtract 24 hours
        prev_year = year(prevDay, "America/New_York")
        prev_month = month(prevDay, "America/New_York")
        prev_day = dayofmonth(prevDay, "America/New_York")
        timestamp("America/New_York", prev_year, prev_month, prev_day, 18, 0, 0)
    else
        // Use current day's 18:00
        timestamp("America/New_York", t_year, t_month, t_day, 18, 0, 0)

// NEW: Percentile calculation function
array_percentile(arr, percentile) =>
    if array.size(arr) == 0
        na
    else
        sorted = array.copy(arr)
        array.sort(sorted)
        n = array.size(sorted)
        index = (percentile / 100) * (n - 1)
        lower_index = math.floor(index)
        upper_index = math.ceil(index)
        
        if lower_index == upper_index
            array.get(sorted, int(lower_index))
        else
            lower_value = array.get(sorted, int(lower_index))
            upper_value = array.get(sorted, int(upper_index))
            fraction = index - lower_index
            lower_value + fraction * (upper_value - lower_value)

// Fixed filter function for NY midnight and futures sessions
filterArrayByCondition(arr, timestamps, conditionValue, lookback, timeframe, nyMidnightEnabled) =>
    filtered = array.new_float(0)
    count = 0
    for i = 0 to array.size(timestamps) - 1
        if count >= lookback
            break
        timestamp = array.get(timestamps, i)
        conditionMet = false
        
        if timeframe == "1D" or timeframe == "1W" or timeframe == "1M"
            // For daily, weekly, and monthly timeframes, always include all data regardless of NY midnight setting
            conditionMet := true
        else if nyMidnightEnabled and (timeframe == "240" or timeframe == "480" or timeframe == "720")
            // For intraday NY midnight sessions, check if it matches the current session slot
            currentHour = hour(timestamp, "America/New_York")
            if timeframe == "240"
                // 4H sessions: 0-4, 4-8, 8-12, 12-16, 16-20, 20-24
                sessionSlot = math.floor(currentHour / 4)
                currentSessionSlot = math.floor(hour(time, "America/New_York") / 4)
                conditionMet := sessionSlot == currentSessionSlot
            else if timeframe == "480"
                // 8H sessions: 0-8, 8-16, 16-24
                sessionSlot = math.floor(currentHour / 8)
                currentSessionSlot = math.floor(hour(time, "America/New_York") / 8)
                conditionMet := sessionSlot == currentSessionSlot
            else if timeframe == "720"
                // 12H sessions: 0-12, 12-24
                sessionSlot = math.floor(currentHour / 12)
                currentSessionSlot = math.floor(hour(time, "America/New_York") / 12)
                conditionMet := sessionSlot == currentSessionSlot
        else
            // Regular timeframe filtering
            conditionMet := timeframe == "60" ? getHourEST(timestamp) == conditionValue :
                           timeframe == "240" ? get4HSlotEST(timestamp) == conditionValue :
                           timeframe == "480" ? get8HSlotEST(timestamp) == conditionValue :
                           timeframe == "720" ? get12HSlotEST(timestamp) == conditionValue :
                           false
        
        if conditionMet
            array.push(filtered, array.get(arr, i))
            count += 1
    filtered

// Determine when a new session starts for NY midnight mode and futures mode
isNewSession(timeframe, nyMidnightEnabled) =>
    if nyMidnightEnabled and (timeframe == "240" or timeframe == "480" or timeframe == "720")
        currentHour = hour(time, "America/New_York")
        prevHour = hour(time[1], "America/New_York")
        
        if timeframe == "240"
            // New 4H session every 4 hours: 0, 4, 8, 12, 16, 20
            currentSession = math.floor(currentHour / 4)
            prevSession = math.floor(prevHour / 4)
            currentSession != prevSession or ta.change(time("1D"))
        else if timeframe == "480"
            // New 8H session every 8 hours: 0, 8, 16
            currentSession = math.floor(currentHour / 8)
            prevSession = math.floor(prevHour / 8)
            currentSession != prevSession or ta.change(time("1D"))
        else if timeframe == "720"
            // New 12H session every 12 hours: 0, 12
            currentSession = math.floor(currentHour / 12)
            prevSession = math.floor(prevHour / 12)
            currentSession != prevSession or ta.change(time("1D"))
        else
            false
    else if nyMidnightEnabled and timeframe == "1D"
        // Daily NY midnight mode - new session at NY midnight (00:00 EST)
        currentDay = dayofmonth(time, "America/New_York")
        prevDay = dayofmonth(time[1], "America/New_York")
        currentDay != prevDay
    else if timeframe == "1D"
        // Daily futures mode - new session at futures open (18:00 EST)
        currentHour = hour(time, "America/New_York")
        prevHour = hour(time[1], "America/New_York")
        // Check if we crossed 18:00 or if it's a new day
        (currentHour >= 18 and prevHour < 18) or (currentHour < 18 and prevHour >= 18)
    else if timeframe == "1W"
        // Weekly timeframe - new session on Monday
        ta.change(time("1W"))
    else if timeframe == "1M"
        // Monthly timeframe - new session on first day of month
        ta.change(time("1M"))
    else
        // Regular timeframe change for 60min and non-NY midnight modes
        ta.change(time(timeframe))

// Fixed candle time calculations
getCurrentSessionStart(timeframe, nyMidnightEnabled) =>
    if nyMidnightEnabled and (timeframe == "240" or timeframe == "480" or timeframe == "720")
        currentHour = hour(time, "America/New_York")
        sessionStartHour = 0  // Initialize the variable
        if timeframe == "240"
            // 4H sessions starting at: 0, 4, 8, 12, 16, 20
            sessionStartHour := math.floor(currentHour / 4) * 4
        else if timeframe == "480"
            // 8H sessions starting at: 0, 8, 16
            sessionStartHour := math.floor(currentHour / 8) * 8
        else if timeframe == "720"
            // 12H sessions starting at: 0, 12
            sessionStartHour := math.floor(currentHour / 12) * 12
        
        t_year = year(time, "America/New_York")
        t_month = month(time, "America/New_York")
        t_day = dayofmonth(time, "America/New_York")
        timestamp("America/New_York", t_year, t_month, t_day, sessionStartHour, 0, 0)
    else if nyMidnightEnabled and timeframe == "1D"
        // Daily NY midnight mode - start at NY midnight
        t_year = year(time, "America/New_York")
        t_month = month(time, "America/New_York")
        t_day = dayofmonth(time, "America/New_York")
        timestamp("America/New_York", t_year, t_month, t_day, 0, 0, 0)
    else if timeframe == "1D"
        // Daily futures mode - start at futures session open (18:00 EST)
        getFuturesSessionTimestamp(time)
    else
        time(timeframe)

getCurrentSessionEnd(timeframe, nyMidnightEnabled) =>
    if nyMidnightEnabled and (timeframe == "240" or timeframe == "480" or timeframe == "720")
        sessionStart = getCurrentSessionStart(timeframe, nyMidnightEnabled)
        if timeframe == "240"
            sessionStart + 14400000  // 4 hours in milliseconds
        else if timeframe == "480"
            sessionStart + 28800000  // 8 hours in milliseconds
        else if timeframe == "720"
            sessionStart + 43200000  // 12 hours in milliseconds
        else
            sessionStart + 14400000
    else if nyMidnightEnabled and timeframe == "1D"
        // Daily NY midnight mode - end at next NY midnight
        sessionStart = getCurrentSessionStart(timeframe, nyMidnightEnabled)
        sessionStart + 86400000  // 24 hours in milliseconds
    else if timeframe == "1D"
        // Daily futures mode - end at next futures session (next day's 18:00)
        sessionStart = getCurrentSessionStart(timeframe, nyMidnightEnabled)
        sessionStart + 86400000  // 24 hours in milliseconds
    else
        time_close(timeframe)

AverageManipulationWickLength(_open, _high, _low, _close) =>
    _close > _open ? _open - _low : _high - _open

AverageDistributionDistance(_open, _high, _low, _close) =>
    _close > _open ? _high - _open : _open - _low

// Function to calculate max range from open
MaxRangeFromOpen(_open, _high, _low, _close) =>
    math.max(math.abs(_high - _open), math.abs(_low - _open))

// NEW: Function to calculate actual price range (high - low)
ActualRange(_open, _high, _low, _close) =>
    _high - _low

// NEW: Functions to calculate range extensions from open
HighRangeFromOpen(_open, _high, _low, _close) =>
    _high - _open

LowRangeFromOpen(_open, _high, _low, _close) =>
    _open - _low
