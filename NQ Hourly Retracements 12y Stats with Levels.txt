get_position(pos) =>
    switch pos
        'Bottom Center' => position.bottom_center
        'Bottom Left' => position.bottom_left
        'Bottom Right' => position.bottom_right
        'Middle Center' => position.middle_center
        'Middle Left' => position.middle_left
        'Middle Right' => position.middle_right

        'Top Left' => position.top_left

get_size(s) =>
    switch s
        'Tiny' => size.tiny
        'Small' => size.small
        'Normal' => size.normal
        => size.large

get_line_style(s) =>
    switch s
        "Solid" => line.style_solid
        "Dotted" => line.style_dotted
        => line.style_dashed

get_shape(s) =>
    switch s
        "Circle" => shape.circle
        "Triangle Up" => shape.triangleup
        "Triangle Down" => shape.triangledown

// ============================================================================
// DATA RETRIEVAL - STORE VALUES AT HOUR START
// ============================================================================

bool newHour = ta.change(time("60"))

// Variables to store static values for the current hour
var float stored_prevHigh = na
var float stored_prevLow = na
var float stored_prevOpen = na
var float stored_prevRange = na
var float stored_prev50 = na
var float stored_currHourOpen = na
var int stored_hourStartTime = na
var int stored_hourEndTime = na
var bool stored_isAbove = false

// Track if sweeps have occurred this hour
var bool highSweptThisHour = false
var bool lowSweptThisHour = false

// Get hourly data
[h1_open, h1_high, h1_low, h1_close] = request.security(syminfo.tickerid, "60", [open, high, low, close])

// Calculate hour end time
getHourEnd(t) =>
    t + 3600000  // Add 1 hour in milliseconds

// Store values when new hour starts
if newHour
    // Store previous COMPLETED hour's data
    stored_prevHigh := h1_high[1]
    stored_prevLow := h1_low[1]
    stored_prevOpen := h1_open[1]
    stored_prevRange := stored_prevHigh - stored_prevLow
    stored_prev50 := (stored_prevHigh + stored_prevLow) / 2
    
    // Store current hour's opening data
    stored_currHourOpen := open
    stored_hourStartTime := time
    stored_hourEndTime := getHourEnd(time)
    stored_isAbove := open > h1_open[1]
    
    // Reset sweep flags
    highSweptThisHour := false
    lowSweptThisHour := false

// Detect sweeps
bool highSweepNow = false
bool lowSweepNow = false

if not highSweptThisHour and high > stored_prevHigh
    highSweptThisHour := true
    highSweepNow := true

if not lowSweptThisHour and low < stored_prevLow
    lowSweptThisHour := true
    lowSweepNow := true

// Plot sweep markers
if show_sweep_markers
    shape_type = get_shape(marker_shape)
    shape_size = get_size(marker_size)
    
    if highSweepNow
        label.new(bar_index, high, "H", 
             style=label.style_label_down, 
             color=high_sweep_color, 
             textcolor=color.white, 
             size=shape_size,
             yloc=yloc.abovebar)
    
    if lowSweepNow
        label.new(bar_index, low, "L", 
             style=label.style_label_up, 
             color=low_sweep_color, 
             textcolor=color.white, 
             size=shape_size,
             yloc=yloc.belowbar)
