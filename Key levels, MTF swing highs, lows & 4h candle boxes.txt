// ===== CORE FUNCTIONS =====

isRegularFractal(mode) =>
    bool result = false
    if fractalBars == "3"
        // 3-bar fractal: pivot at index 1
        result := mode == 1 ? high[2] < high[1] and high[1] > high[0] : mode == -1 ? low[2] > low[1] and low[1] < low[0] : false
    else if fractalBars == "5"
        // 5-bar fractal: pivot at index 2 (original behaviour)
        result := mode == 1 ? high[4] < high[3] and high[3] < high[2] and high[2] > high[1] and high[1] > high[0] : mode == -1 ? low[4] > low[3] and low[3] > low[2] and low[2] < low[1] and low[1] < low[0] : false
    else if fractalBars == "7"
        // 7-bar fractal: pivot at index 3
        result := mode == 1 ? high[6] < high[5] and high[5] < high[4] and high[4] < high[3] and high[3] > high[2] and high[2] > high[1] and high[1] > high[0] : mode == -1 ? low[6] > low[5] and low[5] > low[4] and low[4] > low[3] and low[3] < low[2] and low[2] < low[1] and low[1] < low[0] : false
    else
        // 9-bar fractal: pivot at index 4
        result := mode == 1 ? high[8] < high[7] and high[7] < high[6] and high[6] < high[5] and high[5] < high[4] and high[4] > high[3] and high[3] > high[2] and high[2] > high[1] and high[1] > high[0] : mode == -1 ? low[8] > low[7] and low[7] > low[6] and low[6] > low[5] and low[5] > low[4] and low[4] < low[3] and low[3] < low[2] and low[2] < low[1] and low[1] < low[0] : false
    result

isBWFractal(mode) =>
    bool result = false
    if fractalBars == "3"
        // 3-bar Bill Williams: same as regular for 3 bars
        result := mode == 1 ? high[2] < high[1] and high[1] > high[0] : mode == -1 ? low[2] > low[1] and low[1] < low[0] : false
    else if fractalBars == "5"
        // 5-bar Bill Williams: pivot at index 2 (original behaviour)
        result := mode == 1 ? high[4] < high[2] and high[3] <= high[2] and high[2] >= high[1] and high[2] > high[0] : mode == -1 ? low[4] > low[2] and low[3] >= low[2] and low[2] <= low[1] and low[2] < low[0] : false
    else if fractalBars == "7"
        // 7-bar Bill Williams: pivot at index 3
        result := mode == 1 ? high[6] < high[3] and high[5] <= high[3] and high[4] <= high[3] and high[3] >= high[2] and high[3] >= high[1] and high[3] > high[0] : mode == -1 ? low[6] > low[3] and low[5] >= low[3] and low[4] >= low[3] and low[3] <= low[2] and low[3] <= low[1] and low[3] < low[0] : false
    else
        // 9-bar Bill Williams: pivot at index 4
        result := mode == 1 ? high[8] < high[4] and high[7] <= high[4] and high[6] <= high[4] and high[5] <= high[4] and high[4] >= high[3] and high[4] >= high[2] and high[4] >= high[1] and high[4] > high[0] : mode == -1 ? low[8] > low[4] and low[7] >= low[4] and low[6] >= low[4] and low[5] >= low[4] and low[4] <= low[3] and low[4] <= low[2] and low[4] <= low[1] and low[4] < low[0] : false
    result

dt = time - time[1]

GetLastPivots() =>
    var float LastPh_v = na
    var float LastPl_v = na
    var int LastPh_t = na
    var int LastPl_t = na
    
    // Calculate offset based on fractal bars setting
    int offset = fractalBars == "3" ? 1 : fractalBars == "5" ? 2 : fractalBars == "7" ? 3 : 4
    
    filteredtopf = filterBW ? isRegularFractal(1) : isBWFractal(1)
    filteredbotf = filterBW ? isRegularFractal(-1) : isBWFractal(-1)

    if filteredtopf
        LastPh_v := high[offset]
        LastPh_t := time[offset]
    if filteredbotf
        LastPl_v := low[offset]
        LastPl_t := time[offset]
        
    [LastPh_v, LastPh_t, LastPl_v, LastPl_t]

StyleToEnum(s) =>
    if s == 'Solid'
        line.style_solid
    else if s == 'Dotted'
        line.style_dotted
    else
        line.style_dashed

GetTextSize(s) =>
    if s == 'Tiny'
        size.tiny
    else if s == 'Small'
        size.small
    else
        size.normal

FindT(t, v) =>
    int result = time
    begin = (time - t) / dt
    if begin < 5000
        for i = begin to 0
            if high[i] == v or low[i] == v
                result := t + (begin-i) * dt
                break
        result
    else
        t

// ===== UPDATE 4H CANDLES =====
update_candle(candle_data c, bool in_session) =>
    if in_session and not in_session[1]
        c.hi := high
        c.lo := low
        c.op := open
        c.start_bar := bar_index
        c.completed := false
        
        if show_4h_boxes
            c.candle_box := box.new(bar_index, high, bar_index, low,
                 bgcolor=c.box_color,
                 border_color=color.new(c.box_color, 0),
                 border_width=1)
            
            // Add time range label
            label_y = low - (high - low) * 0.1
            c.time_label := label.new(bar_index, label_y, c.time_range,
                 style=label.style_none,
                 color=color.new(color.white, 100),
                 textcolor=color.new(color.gray, 0),
                 size=size.small)
    
    if in_session and in_session[1]
        if na(c.hi)
            c.hi := high
        else
            c.hi := math.max(c.hi, high)
        
        if na(c.lo)
            c.lo := low
        else
            c.lo := math.min(c.lo, low)
        
        if na(c.start_bar)
            c.start_bar := bar_index
            c.op := open
        
        // Calculate time elapsed in minutes
        int bars_elapsed = bar_index - c.start_bar
        int minutes_per_bar = timeframe.in_seconds(timeframe.period) / 60
        int minutes_elapsed = bars_elapsed * minutes_per_bar
        
        // Determine which hour we're in
        bool in_hour_1 = minutes_elapsed < 60
        bool in_hour_2 = minutes_elapsed >= 60 and minutes_elapsed < 120
        bool in_hour_3 = minutes_elapsed >= 120 and minutes_elapsed < 180
        bool in_hour_4 = minutes_elapsed >= 180
        
        if show_4h_boxes and na(c.candle_box)
            c.candle_box := box.new(c.start_bar, c.hi, bar_index, c.lo,
                 bgcolor=c.box_color,
                 border_color=color.new(c.box_color, 0),
                 border_width=1)
        
        if show_4h_boxes and not na(c.candle_box)
            box.set_top(c.candle_box, c.hi)
            box.set_bottom(c.candle_box, c.lo)
            box.set_right(c.candle_box, bar_index)
            
            // Update time range label position
            if not na(c.time_label)
                label_y = c.lo - (c.hi - c.lo) * 0.1
                label.set_y(c.time_label, label_y)
                label.set_x(c.time_label, c.start_bar)
        
        // Draw hourly dividers
        if show_hourly_dividers and show_4h_boxes
            // 1H divider
            if in_hour_2 and na(c.divider_1h)
                c.divider_1h := line.new(bar_index, c.lo, bar_index, c.hi,
                     xloc=xloc.bar_index,
                     color=divider_color,
                     style=StyleToEnum(divider_style),
                     width=divider_width)
            else if not na(c.divider_1h)
                line.set_y1(c.divider_1h, c.lo)
                line.set_y2(c.divider_1h, c.hi)
            
            // 2H divider and label
            if in_hour_3 and na(c.divider_2h)
                c.divider_2h := line.new(bar_index, c.lo, bar_index, c.hi,
                     xloc=xloc.bar_index,
                     color=divider_color,
                     style=StyleToEnum(divider_style),
                     width=divider_width)
                
                if show_candle_labels
                    c.candle_label := label.new(bar_index, c.hi, c.name,
                         style=label.style_none,
                         color=color.new(color.white, 100),
                         textcolor=divider_color,
                         size=GetTextSize(label_text_size))
            else if not na(c.divider_2h)
                line.set_y1(c.divider_2h, c.lo)
                line.set_y2(c.divider_2h, c.hi)
                if show_candle_labels and not na(c.candle_label)
                    label.set_y(c.candle_label, c.hi)
