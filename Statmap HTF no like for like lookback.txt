// === Timeframe Conversion Function ===
get_timeframe(tf_string) =>
    string tf = ""
    if tf_string == "None"
        tf := ""
    else if tf_string == "1H"
        tf := "60"
    else if tf_string == "4H"
        tf := "240"
    else if tf_string == "1D"
        tf := "D"
    else if tf_string == "1W"
        tf := "W"
    else if tf_string == "1M"
        tf := "M"
    tf

get_line_style(style_string) =>
    style_string == "Solid" ? line.style_solid : style_string == "Dashed" ? line.style_dashed : line.style_dotted

get_quantile(arr, q) =>
    if array.size(arr) == 0
        na
    else
        sorted = array.copy(arr)  
        array.sort(sorted)        
        index = math.round((q / 100) * (array.size(sorted) - 1))
        array.get(sorted, index)

is_new_candle_tf1 = tf1_converted != "" ? timeframe.change(tf1_converted) : false

var float tf1_close = na

    tf1_low := math.min(tf1_low, low)
    tf1_close := close

var float[] tf1_dist_distances = array.new_float(0)
var float[] tf1_highs = array.new_float(0)  // Stores High - Open distances
var float[] tf1_lows = array.new_float(0)    // Stores Open - Low distances
if is_new_candle_tf1 and not na(tf1_open[1])
    float manip = na
    float dist = na
    float high_dist = tf1_high[1] - tf1_open[1]  // Distance from open to high
    float low_dist = tf1_open[1] - tf1_low[1]    // Distance from open to low
    if tf1_close[1] > tf1_open[1]
        manip := tf1_open[1] - tf1_low[1]
        dist := tf1_high[1] - tf1_open[1]
    else if tf1_close[1] < tf1_open[1]
        manip := tf1_high[1] - tf1_open[1]
        dist := tf1_open[1] - tf1_low[1]
    if not na(manip)
        array.push(tf1_manip_distances, manip)
        array.push(tf1_dist_distances, dist)
        array.push(tf1_highs, high_dist)
        array.push(tf1_lows, low_dist)
        if array.size(tf1_manip_distances) > hist_candles
            array.shift(tf1_manip_distances)
            array.shift(tf1_dist_distances)
            array.shift(tf1_highs)
            array.shift(tf1_lows)

tf1_med_dist = array.size(tf1_dist_distances) > 0 ? array.median(tf1_dist_distances) : na
tf1_avg_high = array.size(tf1_highs) > 0 ? array.avg(tf1_highs) : na  // Average of High - Open
tf1_avg_low = array.size(tf1_lows) > 0 ? array.avg(tf1_lows) : na     // Average of Open - Low
