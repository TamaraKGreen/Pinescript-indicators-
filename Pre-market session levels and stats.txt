// === Helper Functions ===
get_line_style(style_string) =>
    style_string == "Solid" ? line.style_solid : style_string == "Dashed" ? line.style_dashed : line.style_dotted

get_label_size(size_string) =>
    size_string == "Tiny" ? size.tiny : size_string == "Small" ? size.small : size_string == "Normal" ? size.normal : size.large

get_label_style() =>
    label.style_label_left

get_weekday() =>
    day = dayofweek(time, time_zone)
    day == dayofweek.monday ? "Monday" : day == dayofweek.tuesday ? "Tuesday" : day == dayofweek.wednesday ? "Wednesday" : day == dayofweek.thursday ? "Thursday" : day == dayofweek.friday ? "Friday" : "Other"

// Function to format price with appropriate precision
format_price(price) =>
    str.tostring(price, "#." + str.repeat("0", price_precision))

is_session_end_levels = not is_in_session_levels and is_in_session_levels[1]

    session_high := math.max(session_high, high)
    session_low := math.min(session_low, low)

var float[] high_extensions = array.new_float(0)
var float[] low_extensions = array.new_float(0)

    float sess_range = session_high - session_low
    float high_ext = session_high - session_open
    float low_ext = session_open - session_low

        array.push(range_distances, sess_range)
    if not na(high_ext)
        array.push(high_extensions, high_ext)
    if not na(low_ext)
        array.push(low_extensions, low_ext)

std_dev_range = array.size(range_distances) > 1 ? array.stdev(range_distances) : na
mean_high_extension = array.size(high_extensions) > 0 ? array.avg(high_extensions) : na
median_high_extension = array.size(high_extensions) > 0 ? array.median(high_extensions) : na
mean_low_extension = array.size(low_extensions) > 0 ? array.avg(low_extensions) : na

    lower_range_std1 := not na(std_dev_range) ? session_open - (avg_range + (std_dev_level_1 * std_dev_range)) : na
    upper_range_std2 := not na(std_dev_range) ? session_open + (avg_range + (std_dev_level_2 * std_dev_range)) : na
    lower_range_std2 := not na(std_dev_range) ? session_open - (avg_range + (std_dev_level_2 * std_dev_range)) : na
else if std_dev_anchor == "High/Low"
    upper_range_std1 := not na(std_dev_range) ? session_high + (std_dev_level_1 * std_dev_range) : na
    lower_range_std1 := not na(std_dev_range) ? session_low - (std_dev_level_1 * std_dev_range) : na
    upper_range_std2 := not na(std_dev_range) ? session_high + (std_dev_level_2 * std_dev_range) : na
    lower_range_std2 := not na(std_dev_range) ? session_low - (std_dev_level_2 * std_dev_range) : na

    median_high_level := not na(median_high_extension) ? session_high + median_high_extension : na

    median_low_level := not na(median_low_extension) ? session_low - median_low_extension : na
