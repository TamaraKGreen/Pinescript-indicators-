// ---------------------------------------------- Helper Functions --------------------------------------------------

get_line_style(style_string) =>
    switch style_string
        "Solid" => line.style_solid
        "Dashed" => line.style_dashed
        "Dotted" => line.style_dotted
        => line.style_solid

get_label_size(size_string) =>
    switch size_string
        "tiny" => size.tiny
        "small" => size.small
        "normal" => size.normal
        "large" => size.large
        => size.small

get_range(session, session_name, session_css)=>
    var t = 0 
    var max = high
    var min = low
    var sess_open = open
    var box bx = na
    var int start_idx = na
    
    if session > session[1]
        t := time
        max := high
        min := low
        sess_open := open
        start_idx := bar_index
        bx := box.new(bar_index, max, bar_index, min, 
             bgcolor=color.new(session_css, bg_transp), 
             border_color=show_outline ? session_css : na, 
             border_style=line.style_dotted)
    
    if session and session == session[1]
        max := math.max(high, max)
        min := math.min(low, min)
        box.set_top(bx, max)
        box.set_rightbottom(bx, bar_index, min)
    
    [session ? na : max, session ? na : min, sess_open, start_idx]

tf = timeframe.period
var tz = "America/New_York"
is_sesa = math.sign(nz(time(tf, "0800-1600", tz)))
is_sesb = math.sign(nz(time(tf, "0200-0800", tz)))
is_sesc = math.sign(nz(time(tf, "2000-0200", tz)))
is_7to8 = math.sign(nz(time(tf, "0700-0800", tz)))
is_2to3 = math.sign(nz(time(tf, "0200-0300", tz)))
is_midnight = math.sign(nz(time(tf, "0000-0100", tz)))

// Session Variables
var float max_sesa = na
var float min_sesa = na
var float open_sesa = na
var int sesa_start_idx = na

var float max_sesb = na
var float min_sesb = na
var float open_sesb = na
var int sesb_start_idx = na

var float max_sesc = na
var float min_sesc = na
var float open_sesc = na
var int sesc_start_idx = na

var float max_7to8 = na
var float min_7to8 = na
var float max_2to3 = na
var float min_2to3 = na
var float midnight_open_price = na

// London session tracking (tracks Asia levels)
var string london_first_hit = "none"
var bool london_seq_hit_asia_high = false
var bool london_seq_hit_asia_low = false

// Track whether Asia high/low were hit during London session
// Persists into NY session to select the correct Section 8 hit rate
var bool asia_high_hit_in_london = false
var bool asia_low_hit_in_london = false

// NY session tracking (tracks London levels + intraday)
var string ny_first_hit = "none"
var bool ny_seq_hit_london_high = false
var bool ny_seq_hit_london_low = false
var bool ny_seq_hit_7to8_mid = false
var bool ny_seq_hit_midnight_open = false
var bool ny_seq_hit_2to3_mid = false

// Track whether Asia high/low are hit during NY session (for label deletion + table status)
var bool asia_high_hit_in_ny = false
var bool asia_low_hit_in_ny = false

// Get hourly data
if is_7to8 and is_7to8 > is_7to8[1]
    max_7to8 := high
    min_7to8 := low
else if is_7to8
    max_7to8 := math.max(high, max_7to8)
    min_7to8 := math.min(low, min_7to8)

if is_2to3 and is_2to3 > is_2to3[1]
    max_2to3 := high
    min_2to3 := low
else if is_2to3
    max_2to3 := math.max(high, max_2to3)
    min_2to3 := math.min(low, min_2to3)

if is_midnight and is_midnight > is_midnight[1]
    midnight_open_price := open

// Session Ranges
if show_sesa
    [max, min, sess_open, start] = get_range(is_sesa, sesa_txt, sesa_css)
    max_sesa := max
    min_sesa := min
    open_sesa := sess_open
    sesa_start_idx := start

if show_sesb
    [max, min, sess_open, start] = get_range(is_sesb, sesb_txt, sesb_css)
    max_sesb := max
    min_sesb := min
    open_sesb := sess_open
    sesb_start_idx := start

if show_sesc
    [max, min, sess_open, start] = get_range(is_sesc, sesc_txt, sesc_css)
    max_sesc := max
    min_sesc := min
    open_sesc := sess_open
    sesc_start_idx := start

// Reset London flags at start of London session
if is_sesb and is_sesb > is_sesb[1]
    london_first_hit := "none"
    london_seq_hit_asia_high := false
    london_seq_hit_asia_low := false
    asia_high_hit_in_london := false
    asia_low_hit_in_london := false

// Reset NY flags at start of NY session
if is_sesa and is_sesa > is_sesa[1]
    ny_first_hit := "none"
    ny_seq_hit_london_high := false
    ny_seq_hit_london_low := false
    ny_seq_hit_7to8_mid := false
    ny_seq_hit_midnight_open := false
    ny_seq_hit_2to3_mid := false
    asia_high_hit_in_ny := false
    asia_low_hit_in_ny := false

// Track first hit during LONDON session (Asia levels)
if is_sesb and london_first_hit == "none"
    if not na(max_sesc) and high >= max_sesc
        london_first_hit := "asia_high"
    else if not na(min_sesc) and low <= min_sesc
        london_first_hit := "asia_low"

// Track sequential hits during LONDON session
if is_sesb and london_first_hit != "none"
    if london_first_hit == "asia_high"
        if not na(min_sesc) and low <= min_sesc
            london_seq_hit_asia_low := true
    else if london_first_hit == "asia_low"
        if not na(max_sesc) and high >= max_sesc
            london_seq_hit_asia_high := true

// Track Asia high/low hit status throughout entire London session
if is_sesb
    if not na(max_sesc) and high >= max_sesc
        asia_high_hit_in_london := true
    if not na(min_sesc) and low <= min_sesc
        asia_low_hit_in_london := true

// Track first hit during NY session (London levels ONLY)
if is_sesa and ny_first_hit == "none"
    if not na(max_sesb) and high >= max_sesb
        ny_first_hit := "london_high"
    else if not na(min_sesb) and low <= min_sesb
        ny_first_hit := "london_low"

// Track sequential hits during NY session
if is_sesa and ny_first_hit != "none"
    if ny_first_hit == "london_high"
        if not na(min_sesb) and low <= min_sesb
            ny_seq_hit_london_low := true
    else if ny_first_hit == "london_low"
        if not na(max_sesb) and high >= max_sesb
            ny_seq_hit_london_high := true

// Track intraday levels during NY session
if is_sesa
    if not na(max_7to8) and not na(min_7to8)
        mid_7to8 = (max_7to8 + min_7to8) / 2
        if low <= mid_7to8 and mid_7to8 <= high
            ny_seq_hit_7to8_mid := true
    
    if not na(midnight_open_price)
        if low <= midnight_open_price and midnight_open_price <= high
            ny_seq_hit_midnight_open := true
    
    if not na(max_2to3) and not na(min_2to3)
        mid_2to3 = (max_2to3 + min_2to3) / 2
        if low <= mid_2to3 and mid_2to3 <= high
            ny_seq_hit_2to3_mid := true

    // FIX: Use ta.crossover/ta.crossunder for Asia level hits during NY.
    // This fires exactly once when price genuinely crosses the level,
    // requires no open-based gap guards, and correctly handles price
    // approaching from either direction.
    if not na(max_sesc) and ta.crossover(high, max_sesc)
        asia_high_hit_in_ny := true
    if not na(min_sesc) and ta.crossunder(low, min_sesc)
        asia_low_hit_in_ny := true

get_session_pattern(max_sesb, min_sesb, max_sesc, min_sesc) =>
    var string pattern = ""
    if not na(max_sesb) and not na(min_sesb) and not na(max_sesc) and not na(min_sesc)
        if max_sesb > max_sesc and min_sesb < min_sesc
            pattern := "London Engulfs Asia"
        else if max_sesc >= max_sesb and min_sesc <= min_sesb
            pattern := "Asia Engulfs London"
        else if max_sesb > max_sesc and min_sesb >= min_sesc
            pattern := "London Partial Up"
        else if max_sesb <= max_sesc and min_sesb < min_sesc
            pattern := "London Partial Down"
    pattern

get_london_vs_asia_stats(position) =>
    hit_high_pct = position == "above" ? 73.84 : 25.54
    hit_low_pct = position == "above" ? 20.39 : 68.75
    sample_size = position == "above" ? 1785 : 1347
    [hit_high_pct, hit_low_pct, sample_size]

get_london_sequential_stats(position, first_hit) =>
    seq_pct = 0.0
    if position == "above"
        seq_pct := first_hit == "asia_high" ? 25.27 : 29.67
    else
        seq_pct := first_hit == "asia_high" ? 25.29 : 29.48
    seq_pct

get_ny_vs_london_midpoint_stats(position) =>
    london_high_first_pct = position == "above" ? 76.68 : 25.94
    london_low_first_pct = position == "above" ? 22.48 : 73.32
    sample_size = position == "above" ? 1775 : 1357
    [london_high_first_pct, london_low_first_pct, sample_size]

get_ny_sequential_london_stats(position, first_hit) =>
    seq_pct = 0.0
    if position == "above"
        seq_pct := first_hit == "london_high" ? 47.98 : 46.87
    else
        seq_pct := first_hit == "london_high" ? 46.31 : 49.15
    seq_pct

        if ny_position == "above"

            hit_7to8_mid := 97.92

get_asia_in_ny_stats(level, hit_in_london) =>
    rate = 0.0
    if level == "asia_high"
        rate := hit_in_london ? 78.41 : 59.69
    else if level == "asia_low"
        rate := hit_in_london ? 78.54 : 54.64
    rate

get_current_session() =>
    if is_sesa
        "New York"
    else if is_sesb
        "London"
    else if is_sesc
        "Asia"
    else
        "None"

// ---------------------------------------------- Current Conditions --------------------------------------------------

var string london_vs_asia = "unknown"
var string ny_vs_london = "unknown"
var string current_pattern = "unknown"

if not na(max_sesc) and not na(min_sesc) and not na(open_sesb)
    asia_mid = (max_sesc + min_sesc) / 2
    london_vs_asia := open_sesb > asia_mid ? "above" : "below"

if not na(max_sesb) and not na(min_sesb) and not na(open_sesa)
    london_mid = (max_sesb + min_sesb) / 2
    ny_vs_london := open_sesa > london_mid ? "above" : "below"

if not na(max_sesb) and not na(min_sesb) and not na(max_sesc) and not na(min_sesc)
    current_pattern := get_session_pattern(max_sesb, min_sesb, max_sesc, min_sesc)

// ---------------------------------------------- Draw Hit Rate Levels --------------------------------------------------

var line line_asia_high = na
var label label_asia_high = na
var line line_asia_low = na
var label label_asia_low = na
var line line_london_high = na
var label label_london_high = na
var line line_london_low = na
var label label_london_low = na
var line line_7to8_mid = na
var label label_7to8_mid = na
var line line_midnight = na
var label label_midnight = na
var line line_2to3_mid = na
var label label_2to3_mid = na

if show_hit_levels
    sess_line_style = get_line_style(session_line_style)
    lbl_size = get_label_size(label_text_size)

    // -----------------------------------------------------------------------
    // ASIA SESSION END: draw Asia High/Low lines with London first-hit probs
    // -----------------------------------------------------------------------
    if not na(max_sesc) and is_sesc[1] and not is_sesc
        var int asia_end_bar = na
        asia_end_bar := bar_index

        [london_hit_asia_high_pct, london_hit_asia_low_pct, london_sample] = get_london_vs_asia_stats(london_vs_asia)

        line.delete(line_asia_high)
        label.delete(label_asia_high)
        line_asia_high := line.new(asia_end_bar, max_sesc, asia_end_bar, max_sesc, color=sesc_css, style=sess_line_style, width=session_line_width)
        label_asia_high := label.new(asia_end_bar, max_sesc, str.format("Asia High: {0,number,#.##}%", london_hit_asia_high_pct),
             style=label.style_none, textcolor=sesc_css, size=lbl_size, textalign=text.align_left, text_font_family=font.family_monospace)

        line.delete(line_asia_low)
        label.delete(label_asia_low)
        line_asia_low := line.new(asia_end_bar, min_sesc, asia_end_bar, min_sesc, color=sesc_css, style=sess_line_style, width=session_line_width)
        label_asia_low := label.new(asia_end_bar, min_sesc, str.format("Asia Low: {0,number,#.##}%", london_hit_asia_low_pct),

    // -----------------------------------------------------------------------
    // DURING LONDON: update Asia labels based on first-hit status
    // -----------------------------------------------------------------------
    if is_sesb and london_first_hit != "none"
        if london_first_hit == "asia_high"
            if not na(label_asia_low)
                seq_pct = get_london_sequential_stats(london_vs_asia, london_first_hit)
                label.set_text(label_asia_low, str.format("Asia Low: {0,number,#.##}%", seq_pct))
        else if london_first_hit == "asia_low"
            if not na(label_asia_high)
                seq_pct = get_london_sequential_stats(london_vs_asia, london_first_hit)
                label.set_text(label_asia_high, str.format("Asia High: {0,number,#.##}%", seq_pct))

    // Extend Asia lines during London session
    if is_sesb
        curr_bar = bar_index
        if not na(line_asia_high)
            line.set_x2(line_asia_high, curr_bar)
            if not na(label_asia_high)
                label.set_x(label_asia_high, curr_bar)
        if not na(line_asia_low)
            line.set_x2(line_asia_low, curr_bar)
            if not na(label_asia_low)
                label.set_x(label_asia_low, curr_bar)

    // -----------------------------------------------------------------------
    // LONDON SESSION END: switch Asia labels to NY hit rates (Section 8)
    // -----------------------------------------------------------------------
    if not na(max_sesb) and is_sesb[1] and not is_sesb
        var int london_end_bar = na
        london_end_bar := bar_index

        if not na(label_asia_high)
            asia_high_ny_rate = get_asia_in_ny_stats("asia_high", asia_high_hit_in_london)
            label.set_text(label_asia_high, str.format("Asia High: {0,number,#.##}%", asia_high_ny_rate))

        if not na(label_asia_low)
            asia_low_ny_rate = get_asia_in_ny_stats("asia_low", asia_low_hit_in_london)
            label.set_text(label_asia_low, str.format("Asia Low: {0,number,#.##}%", asia_low_ny_rate))

        [ny_hit_london_high_pct, ny_hit_london_low_pct, ny_sample] = get_ny_vs_london_midpoint_stats(ny_vs_london)

        line.delete(line_london_high)
        label.delete(label_london_high)
        line_london_high := line.new(london_end_bar, max_sesb, london_end_bar, max_sesb, color=sesb_css, style=sess_line_style, width=session_line_width)
        label_london_high := label.new(london_end_bar, max_sesb, str.format("London High: {0,number,#.##}%", ny_hit_london_high_pct),
             style=label.style_none, textcolor=sesb_css, size=lbl_size, textalign=text.align_left, text_font_family=font.family_monospace)

        line.delete(line_london_low)
        label.delete(label_london_low)
        line_london_low := line.new(london_end_bar, min_sesb, london_end_bar, min_sesb, color=sesb_css, style=sess_line_style, width=session_line_width)
        label_london_low := label.new(london_end_bar, min_sesb, str.format("London Low: {0,number,#.##}%", ny_hit_london_low_pct),
             style=label.style_none, textcolor=sesb_css, size=lbl_size, textalign=text.align_left, text_font_family=font.family_monospace)

    // -----------------------------------------------------------------------
    // DURING NY: manage label visibility and London label updates
    // -----------------------------------------------------------------------
    if is_sesa
        // Delete Asia labels only once the level is genuinely crossed in NY
        if not na(max_sesc) and asia_high_hit_in_ny and not na(label_asia_high)
            label.delete(label_asia_high)
        if not na(min_sesc) and asia_low_hit_in_ny and not na(label_asia_low)
            label.delete(label_asia_low)

        // Update London labels based on first-hit status
        if ny_first_hit != "none"
            if ny_first_hit == "london_high"
                label.delete(label_london_high)
                if not na(label_london_low)
                    seq_pct = get_ny_sequential_london_stats(ny_vs_london, ny_first_hit)
                    label.set_text(label_london_low, str.format("London Low: {0,number,#.##}%", seq_pct))
            else if ny_first_hit == "london_low"
                label.delete(label_london_low)
                if not na(label_london_high)
                    seq_pct = get_ny_sequential_london_stats(ny_vs_london, ny_first_hit)
                    label.set_text(label_london_high, str.format("London High: {0,number,#.##}%", seq_pct))

    // Extend all lines during NY session
    is_4pm = math.sign(nz(time(tf, "1600-1601", tz)))

    if is_sesa or is_4pm
        curr_bar = bar_index

        if not na(line_asia_high)
            line.set_x2(line_asia_high, curr_bar)
        if not na(line_asia_low)
            line.set_x2(line_asia_low, curr_bar)
        if not na(label_asia_high)
            label.set_x(label_asia_high, curr_bar)
        if not na(label_asia_low)
            label.set_x(label_asia_low, curr_bar)
        if not na(line_london_high)
            line.set_x2(line_london_high, curr_bar)
            if not na(label_london_high)
                label.set_x(label_london_high, curr_bar)
        if not na(line_london_low)
            line.set_x2(line_london_low, curr_bar)
            if not na(label_london_low)
                label.set_x(label_london_low, curr_bar)
        if not na(line_7to8_mid)
            line.set_x2(line_7to8_mid, curr_bar)
            if not na(label_7to8_mid)
                label.set_x(label_7to8_mid, curr_bar)
        if not na(line_midnight)
            line.set_x2(line_midnight, curr_bar)
            if not na(label_midnight)
                label.set_x(label_midnight, curr_bar)
        if not na(line_2to3_mid)
            line.set_x2(line_2to3_mid, curr_bar)
            if not na(label_2to3_mid)
                label.set_x(label_2to3_mid, curr_bar)

// Draw intraday levels at 8am open (NY session only)
