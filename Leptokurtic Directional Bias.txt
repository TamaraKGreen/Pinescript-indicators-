// === MOVING AVERAGE CALCULATION ===
ma(type, src, len) =>
    float result = na
    if type == "SMA"
        result := ta.sma(src, len)
    else if type == "EMA"
        result := ta.ema(src, len)
    else if type == "WMA"
        result := ta.wma(src, len)
    else if type == "RMA"
        result := ta.rma(src, len)
    else if type == "HMA"
        result := ta.hma(src, len)
    else if type == "DEMA"
        e = ta.ema(src, len)
        result := 2 * e - ta.ema(e, len)
    else if type == "TEMA"
        e = ta.ema(src, len)
        result := 3 * (e - ta.ema(e, len)) + ta.ema(ta.ema(e, len), len)
    else if type == "VWMA"
        result := ta.vwma(src, len)
    else if type == "ALMA"
        result := ta.alma(src, len, 6, 0.85)
    result

// === BOLLINGER BANDS TREND STOP CALCULATION ===
htf_close = request.security(syminfo.tickerid, bb_tf, close)
htf_high = request.security(syminfo.tickerid, bb_tf, high)
htf_low = request.security(syminfo.tickerid, bb_tf, low)
htf_avg = request.security(syminfo.tickerid, bb_tf, ta.sma(close, bb_length))
htf_dev = request.security(syminfo.tickerid, bb_tf, ta.stdev(close, bb_length)) * bb_deviation
htf_smax = htf_avg + htf_dev
htf_smin = htf_avg - htf_dev

var trend = 0
if (htf_close > htf_smax[1])
    trend := 1
else if (htf_close < htf_smin[1])
    trend := -1

if (trend > 0 and htf_smin < htf_smin[1])
    htf_smin := htf_smin[1]
if (trend < 0 and htf_smax > htf_smax[1])
    htf_smax := htf_smax[1]

htf_bsmax = htf_smax + 0.5 * (bb_money_risk - 1) * (htf_smax - htf_smin)
htf_bsmin = htf_smin - 0.5 * (bb_money_risk - 1) * (htf_smax - htf_smin)

if (trend > 0 and htf_bsmin < htf_bsmin[1])
    htf_bsmin := htf_bsmin[1]
if (trend < 0 and htf_bsmax > htf_bsmax[1])
    htf_bsmax := htf_bsmax[1]

var float bb_trend_stop = na
if (trend > 0)
    bb_trend_stop := htf_bsmin
if (trend < 0)
    bb_trend_stop := htf_bsmax

// === SOURCE CALCULATION ===
source_data = switch source_type
    "ma" => ma(ma_type, close, ma_length)
    "atr" => ta.atr(atr_length)
    "volume" => ta.sma(volume, volume_length)
    "close" => ta.sma(close, close_length)
    "bb_trend_stop" => bb_trend_stop
    => close

// === KURTOSIS CALCULATION ===
kurtosis(src, len) =>
    mean = ta.sma(src, len)
    variance = ta.sma(math.pow(src - mean, 2), len)
    fourth_moment = ta.sma(math.pow(src - mean, 4), len)
    kurt = variance != 0 ? fourth_moment / math.pow(variance, 2) : na
    kurt

kurtosis_value = kurtosis(source_data, kurtosis_length)
is_leptokurtic = kurtosis_value > 3

// === DIRECTIONAL BIAS LOGIC ===
rsi = ta.rsi(close, rsi_length)
trend_ema = ta.ema(close, trend_ema_length)
price_above_trend = close > trend_ema
recent_momentum = ta.change(close, 3)

// Determine bias
bias_signal = switch bias_mode
    "mean_reversion" => 
        if is_leptokurtic and rsi > 70
            "bearish_reversion"
        else if is_leptokurtic and rsi < 30
            "bullish_reversion"
        else
            "neutral"
    "momentum" => 
        if is_leptokurtic and price_above_trend and recent_momentum > 0
            "bullish_momentum"
        else if is_leptokurtic and not price_above_trend and recent_momentum < 0
            "bearish_momentum"
        else
            "neutral"
    "volatility_only" => 
        if is_leptokurtic
            "high_volatility"
        else
            "normal"
    => "neutral"
