pad2(int v) =>
    (v < 10 ? "0" : "") + str.tostring(v)

bChar(int i) =>
    str.substring("0123456789ab", i, i + 1)

bName(int i) =>
    i == 0  ? "below prev low" :
     i == 11 ? "at/above prev high" :
     str.tostring((i - 1) / 10.0, "0.0") + " - " + str.tostring(i / 10.0, "0.0")

if newRange
    // ---- score the range that just closed ----
    pc = close[1]
    wasUp  = not na(curO) and not na(prvH) and pc > prvH
    wasDn  = not na(curO) and not na(prvL) and pc < prvL
    wasRes = wasUp or wasDn

    // sHit / sDir / sProb still describe the range that just closed, because
    // they are only overwritten further down for the new one.
    if showLive and sHit and not na(curO)
        tResOK.push(wasRes ? 1 : 0)
        tResCl.push(sRes)
        tClaim.push(sProb)
        tOK.push(not wasRes ? na : (sDir == "U" ? wasUp : wasDn) ? 1 : 0)
        while tClaim.size() > trackN
            tClaim.shift()
            tOK.shift()
            tResOK.shift()
            tResCl.shift()

    // ---- colour the range that just closed, by what it actually did ----
    if showBox and not na(bx)
        box.set_bgcolor(bx, wasUp ? colUp : wasDn ? colDn : colIn)

    // ---- now roll the completed range into the previous slot ----
    if not na(curO)
        prvH := curH
        prvL := curL
        prvStart := curStart

    curO := open
    curH := high
    curL := low
    curStart := rStart

    // ---- evaluate the new range against the one before it ----

    sProb := na

    sOP := na
    sBucket := ""
    sDir := ""
    // the previous range must be directly adjacent in time, as in the study
    sAdj := not na(prvStart) and (curStart - prvStart) == tfMs
    sSlot := pad2(hour(curStart, "America/New_York")) + pad2(minute(curStart, "America/New_York"))

    if sAdj and not na(prvH) and prvH > prvL
        sOP := (curO - prvL) / (prvH - prvL)
        bi = sOP < 0 ? 0 : sOP >= 1.0 ? 11 : math.min(10, math.floor(sOP * 10) + 1)
        sBucket := bName(bi)
        sKey := str.tostring(tfIdx) + sSlot + bChar(bi)
        if LUT.contains(sKey)
            v = LUT.get(sKey)
            sDir  := str.substring(v, 0, 1)
            sProb := str.tonumber(str.substring(v, 1, 4))
            sTest := str.tonumber(str.substring(v, 4, 7))
            sN    := math.round(str.tonumber(str.substring(v, 7, 10)))
            sRes  := str.tonumber(str.substring(v, 10, 12))
            sHit  := true

    // ---- open the new box ----
    if showBox
        bx := box.new(curStart, curH, curStart + tfMs, curL, xloc = xloc.bar_time,
             bgcolor = colLive, border_color = color.new(color.gray, 55), border_width = 1)
        boxes.push(bx)
        while boxes.size() > keepN
            box.delete(boxes.shift())

    if sHit and showMarker
        label.new(bar_index, sDir == "U" ? low : high,
             (sDir == "U" ? "▲ " : "▼ ") + str.tostring(sProb, "#") + "%",
             style = sDir == "U" ? label.style_label_up : label.style_label_down,
             color = color.new(sDir == "U" ? color.teal : color.maroon, 20),
             textcolor = color.white, size = size.tiny)

    if sHit and useAlerts and barstate.isconfirmed
        alert(tfSel + "M " + str.substring(sSlot, 0, 2) + ":" + str.substring(sSlot, 2, 4) +
             " ET  opened at " + str.tostring(sOP, "0.00") + "  " +
             (sDir == "U" ? "closes ABOVE previous high " : "closes BELOW previous low ") +
             str.tostring(sProb, "#") + "% of ranges that resolve (" +
             str.tostring(sRes, "#") + "% resolve)", alert.freq_once_per_bar_close)
else
    curH := math.max(curH, high)
    curL := math.min(curL, low)

// keep the live box growing with the range

    box.set_top(bx, curH)
    box.set_bottom(bx, curL)

            kCL.set(i, math.min(kCL.get(i), low))

posOf(string s) =>
    s == "Top left"      ? position.top_left :
     s == "Bottom right" ? position.bottom_right :
     s == "Bottom left"  ? position.bottom_left :
     s == "Middle right" ? position.middle_right : position.top_right

sizeOf(string s) =>
    s == "Tiny" ? size.tiny : s == "Normal" ? size.normal : s == "Large" ? size.large : size.small

pc(float v) =>
    na(v) ? "—" : str.tostring(v, "0.#") + "%"

pc2(float v) =>
    na(v) ? "—" : str.tostring(v, "0") + "%"
