import czoa/SetSessionTimes/1 as Session

//@version=6
indicator("Daily Statistical Range and Levels (Weekday Stats)", overlay=true)

// === Input Definitions ===
// Session Settings for Levels
session_time_levels = input.string("2300-2100", "Levels Session Time (HHMM-HHMM)", tooltip="Session range for calculating levels (e.g., 2300-2100)")
time_zone = input.string("UTC", "Time Zone", options=["UTC", "America/New_York", "America/Chicago", "America/Los_Angeles", "Europe/London", "Asia/Tokyo"], tooltip="Time zone for both sessions")
// Session Settings for Stats
session_time_stats = input.string("0930-1600", "Stats Session Time (HHMM-HHMM)", tooltip="Session range for tracking hit statistics (e.g., 0900-1600)")

// Anchor Settings
show_anchor = input.bool(true, "Show Session Anchor", inline="anchor", tooltip="Toggle vertical line at levels session start")
anchor_style = input.string("Dotted", "", options=["Solid", "Dashed", "Dotted"], inline="anchor")
anchor_color = input.color(color.black, "", inline="anchor")
anchor_width = input.int(1, "", minval=1, maxval=4, inline="anchor")
show_anchor_label = input.bool(true, "Show Anchor Label", tooltip="Toggle label for anchor line")

// Session Open Line Settings
show_session_open = input.bool(true, "Show Session Open Line", inline="session_open", tooltip="Toggle horizontal session open line")
session_open_style = input.string("Dotted", "", options=["Solid", "Dashed", "Dotted"], inline="session_open")
session_open_color = input.color(color.gray, "", inline="session_open")
session_open_width = input.int(1, "", minval=1, maxval=4, inline="session_open")
show_session_open_label = input.bool(true, "Show Session Open Label", tooltip="Toggle label for session open line")

// Manipulation Settings
show_manipulation = input.bool(true, "Show Manipulation Levels", inline="manipulation", tooltip="Toggle horizontal manipulation levels")
manipulation_style = input.string("Dotted", "", options=["Solid", "Dashed", "Dotted"], inline="manipulation")
manipulation_color = input.color(color.red, "", inline="manipulation")
manipulation_width = input.int(1, "", minval=1, maxval=4, inline="manipulation")
show_manipulation_label = input.bool(true, "Show Manipulation Labels", tooltip="Toggle labels for manipulation levels")

// Distribution Settings
show_distribution = input.bool(true, "Show Distribution Levels", inline="distribution", tooltip="Toggle horizontal distribution levels")
distribution_style = input.string("Solid", "", options=["Solid", "Dashed", "Dotted"], inline="distribution")
distribution_color = input.color(color.blue, "", inline="distribution")
distribution_width = input.int(1, "", minval=1, maxval=4, inline="distribution")
show_distribution_label = input.bool(true, "Show Distribution Labels", tooltip="Toggle labels for distribution levels")

// Display Settings
projection_offset = input.int(24, "Projection Offset (Bars)", minval=0, tooltip="Number of bars to extend projection lines beyond session start")
label_size = input.string("Small", "Label Size", options=["Tiny", "Small", "Normal", "Large"], tooltip="Size of labels for all lines")

// === Helper Functions ===
get_line_style(style_string) =>
    style_string == "Solid" ? line.style_solid : style_string == "Dashed" ? line.style_dashed : line.style_dotted

get_label_size(size_string) =>
    size_string == "Tiny" ? size.tiny : size_string == "Small" ? size.small : size_string == "Normal" ? size.normal : size.large

get_label_style() =>
    label.style_label_left

get_weekday() =>
    day = dayofweek(time, time_zone)
    day == dayofweek.monday ? "Monday" : day == dayofweek.tuesday ? "Tuesday" :  day == dayofweek.wednesday ? "Wednesday" : day == dayofweek.thursday ? "Thursday" : day == dayofweek.friday ? "Friday" : "Other"

// Rounding levels to min tick
nround(x) =>
    n = math.round(x / syminfo.mintick) * syminfo.mintick
    n


// === Session Detection ===
// For Levels Calculation and Plotting
is_in_session_levels = not na(time(timeframe.period, session_time_levels, time_zone))
is_session_start_levels = is_in_session_levels and not is_in_session_levels[1]
is_session_end_levels = not is_in_session_levels and is_in_session_levels[1]

// For Stats Tracking
is_in_session_stats = not na(time(timeframe.period, session_time_stats, time_zone))
is_session_start_stats = is_in_session_stats and not is_in_session_stats[1]
is_session_end_stats = not is_in_session_stats and is_in_session_stats[1]

// === Session Data Tracking for Levels ===
var float session_open = na
var float session_high = na
var float session_low = na
var float session_close = na

if is_session_start_levels
    session_open := open
    session_high := high
    session_low := low
    session_close := close
else if is_in_session_levels
    session_high := math.max(session_high, high)
    session_low := math.min(session_low, low)
    session_close := close

// === Arrays and Counters for Levels ===
var float[] manipulation_distances = array.new_float(0)
var float[] distribution_distances = array.new_float(0)

if is_session_end_levels
    float manipulation = na
    float distribution = na
    if session_close > session_open
        manipulation := session_open - session_low
        distribution := session_high - session_open
    else if session_close < session_open
        manipulation := session_high - session_open
        distribution := session_open - session_low
    
    if not na(manipulation) and not na(distribution)
        array.push(manipulation_distances, manipulation)
        array.push(distribution_distances, distribution)

// === Calculate Daily Range Statistics ===
avg_manipulation = array.size(manipulation_distances) > 0 ? array.avg(manipulation_distances) : na
median_manipulation = array.size(manipulation_distances) > 0 ? array.median(manipulation_distances) : na
std_dev_manipulation = array.size(manipulation_distances) > 1 ? array.stdev(manipulation_distances) : na

avg_distribution = array.size(distribution_distances) > 0 ? array.avg(distribution_distances) : na
median_distribution = array.size(distribution_distances) > 0 ? array.median(distribution_distances) : na
std_dev_distribution = array.size(distribution_distances) > 1 ? array.stdev(distribution_distances) : na

// === Standard Deviation Levels Calculation ===
upper_manipulation_std = not na(std_dev_manipulation) ? avg_manipulation + std_dev_manipulation : na
lower_manipulation_std = not na(std_dev_manipulation) ? avg_manipulation - std_dev_manipulation : na
upper_distribution_std = not na(std_dev_distribution) ? avg_distribution + std_dev_distribution : na
lower_distribution_std = not na(std_dev_distribution) ? avg_distribution - std_dev_distribution : na

// === Plotting Levels and Labels ===
var line anchor_line = na
var label anchor_label = na
var line session_open_line = na
var label session_open_label = na
var line upper_manipulation_line = na
var label upper_manipulation_label = na
var line lower_manipulation_line = na
var label lower_manipulation_label = na
var line upper_distribution_line = na
var label upper_distribution_label = na
var line lower_distribution_line = na
var label lower_distribution_label = na
var line median_manipulation_upper_line = na
var label median_manipulation_upper_label = na
var line median_manipulation_lower_line = na
var label median_manipulation_lower_label = na
var line median_distribution_upper_line = na
var label median_distribution_upper_label = na
var line median_distribution_lower_line = na
var label median_distribution_lower_label = na
var line upper_manipulation_std_line = na
var label upper_manipulation_std_label = na
var line lower_manipulation_std_line = na
var label lower_manipulation_std_label = na
var line upper_distribution_std_line = na
var label upper_distribution_std_label = na
var line lower_distribution_std_line = na
var label lower_distribution_std_label = na
var line upper_distribution_std2_line = na
var label upper_distribution_std2_label = na
var line lower_distribution_std2_line = na
var label lower_distribution_std2_label = na

if bar_index == 0
    anchor_line := line.new(bar_index, na, bar_index, na, xloc.bar_index, extend.both, anchor_color, get_line_style(anchor_style), anchor_width)
    session_open_line := line.new(bar_index, na, bar_index+1, na, color=session_open_color, style=get_line_style(session_open_style), width=session_open_width)
    upper_manipulation_line := line.new(bar_index, na, bar_index+1, na, color=manipulation_color, style=get_line_style(manipulation_style), width=manipulation_width)
    lower_manipulation_line := line.new(bar_index, na, bar_index+1, na, color=manipulation_color, style=get_line_style(manipulation_style), width=manipulation_width)
    upper_distribution_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    lower_distribution_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    median_manipulation_upper_line := line.new(bar_index, na, bar_index+1, na, color=manipulation_color, style=get_line_style(manipulation_style), width=manipulation_width)
    median_manipulation_lower_line := line.new(bar_index, na, bar_index+1, na, color=manipulation_color, style=get_line_style(manipulation_style), width=manipulation_width)
    median_distribution_upper_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    median_distribution_lower_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    upper_manipulation_std_line := line.new(bar_index, na, bar_index+1, na, color=manipulation_color, style=get_line_style(manipulation_style), width=manipulation_width)
    lower_manipulation_std_line := line.new(bar_index, na, bar_index+1, na, color=manipulation_color, style=get_line_style(manipulation_style), width=manipulation_width)
    upper_distribution_std_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    lower_distribution_std_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    upper_distribution_std2_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    lower_distribution_std2_line := line.new(bar_index, na, bar_index+1, na, color=distribution_color, style=get_line_style(distribution_style), width=distribution_width)
    // Modified label definitions in the initialization block (replace corresponding lines)
    session_open_label := label.new(bar_index, na, "Session open " + str.tostring(math.round(session_open)), style=get_label_style(), color=color.new(session_open_color, 100), textcolor=session_open_color, size=get_label_size(label_size))
    upper_manipulation_label := label.new(bar_index, na, "-Manipulation " + str.tostring(nround(session_open + avg_manipulation)), style=get_label_style(), color=color.new(manipulation_color, 100), textcolor=manipulation_color, size=get_label_size(label_size))
    lower_manipulation_label := label.new(bar_index, na, "+Manipulation " + str.tostring(nround(session_open - avg_manipulation)), style=get_label_style(), color=color.new(manipulation_color, 100), textcolor=manipulation_color, size=get_label_size(label_size))
    upper_distribution_label := label.new(bar_index, na, "+Distribution " + str.tostring(nround(session_open + avg_distribution)), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    lower_distribution_label := label.new(bar_index, na, "-Distribution " + str.tostring(nround(session_open - avg_distribution)), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    median_manipulation_upper_label := label.new(bar_index, na, "+Median Manipulation " + str.tostring(nround(session_open + median_manipulation)), style=get_label_style(), color=color.new(manipulation_color, 100), textcolor=manipulation_color, size=get_label_size(label_size))
    median_manipulation_lower_label := label.new(bar_index, na, "-Median Manipulation " + str.tostring(nround(session_open - median_manipulation)), style=get_label_style(), color=color.new(manipulation_color, 100), textcolor=manipulation_color, size=get_label_size(label_size))
    median_distribution_upper_label := label.new(bar_index, na, "+Median Distribution " + str.tostring(nround(session_open + median_distribution)), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    median_distribution_lower_label := label.new(bar_index, na, "-Median Distribution " + str.tostring(nround(session_open - median_distribution)), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    upper_manipulation_std_label := label.new(bar_index, na, "+Manipulation stdev " + str.tostring(nround(session_open + (avg_manipulation + std_dev_manipulation))), style=get_label_style(), color=color.new(manipulation_color, 100), textcolor=manipulation_color, size=get_label_size(label_size))
    lower_manipulation_std_label := label.new(bar_index, na, "-Manipulation stdev " + str.tostring(nround(session_open - (avg_manipulation + std_dev_manipulation))), style=get_label_style(), color=color.new(manipulation_color, 100), textcolor=manipulation_color, size=get_label_size(label_size))
    upper_distribution_std_label := label.new(bar_index, na, "+Distribution stdev " + str.tostring(nround(session_open + (avg_distribution + std_dev_distribution))), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    lower_distribution_std_label := label.new(bar_index, na, "-Distribution stdev " + str.tostring(nround(session_open - (avg_distribution + std_dev_distribution))), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    upper_distribution_std2_label := label.new(bar_index, na, "+Distribution 2stdev " + str.tostring(nround(session_open + (avg_distribution + 2 * std_dev_distribution))), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))
    lower_distribution_std2_label := label.new(bar_index, na, "-Distribution 2stdev " + str.tostring(nround(session_open - (avg_distribution + 2 * std_dev_distribution))), style=get_label_style(), color=color.new(distribution_color, 100), textcolor=distribution_color, size=get_label_size(label_size))

// Session Anchor (Vertical Line with Extend Both)
// Session Open Line
if is_session_start_levels and show_session_open
    line.set_xy1(session_open_line, bar_index, session_open)
    line.set_xy2(session_open_line, bar_index + projection_offset, session_open)
if show_session_open_label
    label.set_xy(session_open_label, bar_index + projection_offset, session_open)
    label.set_text(session_open_label, "Session open " + str.tostring(nround(session_open)))

// Manipulation Levels
if is_session_start_levels and not na(avg_manipulation) and not na(avg_distribution) and show_manipulation
    line.set_xy1(upper_manipulation_line, bar_index + projection_offset, session_open + avg_manipulation)
    line.set_xy2(upper_manipulation_line, bar_index + projection_offset, session_open + avg_manipulation)
    line.set_xy1(lower_manipulation_line, bar_index + projection_offset, session_open - avg_manipulation)
    line.set_xy2(lower_manipulation_line, bar_index + projection_offset, session_open - avg_manipulation)
if show_manipulation_label
    label.set_xy(upper_manipulation_label, bar_index + projection_offset, session_open + avg_manipulation)
    label.set_text(upper_manipulation_label, "-Manipulation " + str.tostring(nround(session_open + avg_manipulation)))
    label.set_xy(lower_manipulation_label, bar_index + projection_offset, session_open - avg_manipulation)
    label.set_text(lower_manipulation_label, "+Manipulation " + str.tostring(nround(session_open - avg_manipulation)))

// Distribution Levels
if is_session_start_levels and not na(avg_manipulation) and not na(avg_distribution) and show_distribution
    line.set_xy1(upper_distribution_line, bar_index + projection_offset, session_open + avg_distribution)
    line.set_xy2(upper_distribution_line, bar_index + projection_offset, session_open + avg_distribution)
    line.set_xy1(lower_distribution_line, bar_index + projection_offset, session_open - avg_distribution)
    line.set_xy2(lower_distribution_line, bar_index + projection_offset, session_open - avg_distribution)
if show_distribution_label
    label.set_xy(upper_distribution_label, bar_index + projection_offset, session_open + avg_distribution)
    label.set_text(upper_distribution_label, "+Distribution " + str.tostring(nround(session_open + avg_distribution)))
    label.set_xy(lower_distribution_label, bar_index + projection_offset, session_open - avg_distribution)
    label.set_text(lower_distribution_label, "-Distribution " + str.tostring(nround(session_open - avg_distribution)))

// Median Manipulation Levels
if is_session_start_levels and not na(median_manipulation) and show_manipulation
    line.set_xy1(median_manipulation_upper_line, bar_index + projection_offset, session_open + median_manipulation)
    line.set_xy2(median_manipulation_upper_line, bar_index + projection_offset, session_open + median_manipulation)
    line.set_xy1(median_manipulation_lower_line, bar_index + projection_offset, session_open - median_manipulation)
    line.set_xy2(median_manipulation_lower_line, bar_index + projection_offset, session_open - median_manipulation)
if show_manipulation_label
    label.set_xy(median_manipulation_upper_label, bar_index + projection_offset, session_open + median_manipulation)
    label.set_text(median_manipulation_upper_label, "+Median Manipulation " + str.tostring(nround(session_open + median_manipulation)))
    label.set_xy(median_manipulation_lower_label, bar_index + projection_offset, session_open - median_manipulation)
    label.set_text(median_manipulation_lower_label, "-Median Manipulation " + str.tostring(nround(session_open - median_manipulation)))

// Median Distribution Levels
if is_session_start_levels and not na(median_distribution) and show_distribution
    line.set_xy1(median_distribution_upper_line, bar_index + projection_offset, session_open + median_distribution)
    line.set_xy2(median_distribution_upper_line, bar_index + projection_offset, session_open + median_distribution)
    line.set_xy1(median_distribution_lower_line, bar_index + projection_offset, session_open - median_distribution)
    line.set_xy2(median_distribution_lower_line, bar_index + projection_offset, session_open - median_distribution)
if show_distribution_label
    label.set_xy(median_distribution_upper_label, bar_index + projection_offset, session_open + median_distribution)
    label.set_text(median_distribution_upper_label, "+Median Distribution " + str.tostring(nround(session_open + median_distribution)))
    label.set_xy(median_distribution_lower_label, bar_index + projection_offset, session_open - median_distribution)
    label.set_text(median_distribution_lower_label, "-Median Distribution " + str.tostring(nround(session_open - median_distribution)))

// Standard Deviation Manipulation Levels
if is_session_start_levels and not na(avg_manipulation) and not na(std_dev_manipulation) and show_manipulation
    line.set_xy1(upper_manipulation_std_line, bar_index + projection_offset, session_open + (avg_manipulation + std_dev_manipulation))
    line.set_xy2(upper_manipulation_std_line, bar_index + projection_offset, session_open + (avg_manipulation + std_dev_manipulation))
    line.set_xy1(lower_manipulation_std_line, bar_index + projection_offset, session_open - (avg_manipulation + std_dev_manipulation))
    line.set_xy2(lower_manipulation_std_line, bar_index + projection_offset, session_open - (avg_manipulation + std_dev_manipulation))
if show_manipulation_label
    label.set_xy(upper_manipulation_std_label, bar_index + projection_offset, session_open + (avg_manipulation + std_dev_manipulation))
    label.set_text(upper_manipulation_std_label, "+Manipulation stdev " + str.tostring(nround(session_open + (avg_manipulation + std_dev_manipulation))))
    label.set_xy(lower_manipulation_std_label, bar_index + projection_offset, session_open - (avg_manipulation + std_dev_manipulation))
    label.set_text(lower_manipulation_std_label, "-Manipulation stdev " + str.tostring(nround(session_open - (avg_manipulation + std_dev_manipulation))))

// Standard Deviation Distribution Levels
if is_session_start_levels and not na(avg_distribution) and not na(std_dev_distribution) and show_distribution
    line.set_xy1(upper_distribution_std_line, bar_index + projection_offset, session_open + (avg_distribution + std_dev_distribution))
    line.set_xy2(upper_distribution_std_line, bar_index + projection_offset, session_open + (avg_distribution + std_dev_distribution))
    line.set_xy1(lower_distribution_std_line, bar_index + projection_offset, session_open - (avg_distribution + std_dev_distribution))
    line.set_xy2(lower_distribution_std_line, bar_index + projection_offset, session_open - (avg_distribution + std_dev_distribution))
if show_distribution_label
    label.set_xy(upper_distribution_std_label, bar_index + projection_offset, session_open + (avg_distribution + std_dev_distribution))
    label.set_text(upper_distribution_std_label, "+Distribution stdev " + str.tostring(nround(session_open + (avg_distribution + std_dev_distribution))))
    label.set_xy(lower_distribution_std_label, bar_index + projection_offset, session_open - (avg_distribution + std_dev_distribution))
    label.set_text(lower_distribution_std_label, "-Distribution stdev " + str.tostring(nround(session_open - (avg_distribution + std_dev_distribution))))

// New ±2σ Distribution Levels
if is_session_start_levels and not na(avg_distribution) and not na(std_dev_distribution) and show_distribution
    line.set_xy1(upper_distribution_std2_line, bar_index + projection_offset, session_open + (avg_distribution + 2 * std_dev_distribution))
    line.set_xy2(upper_distribution_std2_line, bar_index + projection_offset, session_open + (avg_distribution + 2 * std_dev_distribution))
    line.set_xy1(lower_distribution_std2_line, bar_index + projection_offset, session_open - (avg_distribution + 2 * std_dev_distribution))
    line.set_xy2(lower_distribution_std2_line, bar_index + projection_offset, session_open - (avg_distribution + 2 * std_dev_distribution))
if show_distribution_label
    label.set_xy(upper_distribution_std2_label, bar_index + projection_offset, session_open + (avg_distribution + 2 * std_dev_distribution))
    label.set_text(upper_distribution_std2_label, "+Distribution 2stdev " + str.tostring(nround(session_open + (avg_distribution + 2 * std_dev_distribution))))
    label.set_xy(lower_distribution_std2_label, bar_index + projection_offset, session_open - (avg_distribution + 2 * std_dev_distribution))
    label.set_text(lower_distribution_std2_label, "-Distribution 2stdev " + str.tostring(nround(session_open - (avg_distribution + 2 * std_dev_distribution))))


// Extend Lines and Labels Beyond Session
if bar_index > bar_index[1]
    if show_session_open
        line.set_x2(session_open_line, bar_index + projection_offset + 3)
    if show_session_open_label
        label.set_x(session_open_label, bar_index + projection_offset + 3)
    if show_manipulation
        line.set_x2(upper_manipulation_line, bar_index + projection_offset)
        line.set_x2(lower_manipulation_line, bar_index + projection_offset)
        line.set_x2(median_manipulation_upper_line, bar_index + projection_offset)
        line.set_x2(median_manipulation_lower_line, bar_index + projection_offset)
        line.set_x2(upper_manipulation_std_line, bar_index + projection_offset)
        line.set_x2(lower_manipulation_std_line, bar_index + projection_offset)
    if show_manipulation_label
        label.set_x(upper_manipulation_label, bar_index + projection_offset)
        label.set_x(lower_manipulation_label, bar_index + projection_offset)
        label.set_x(median_manipulation_upper_label, bar_index + projection_offset)
        label.set_x(median_manipulation_lower_label, bar_index + projection_offset)
        label.set_x(upper_manipulation_std_label, bar_index + projection_offset)
        label.set_x(lower_manipulation_std_label, bar_index + projection_offset)
    if show_distribution
        line.set_x2(upper_distribution_line, bar_index + projection_offset)
        line.set_x2(lower_distribution_line, bar_index + projection_offset)
        line.set_x2(median_distribution_upper_line, bar_index + projection_offset)
        line.set_x2(median_distribution_lower_line, bar_index + projection_offset)

    hit_session_low := low

    hit_session_high := math.max(hit_session_high, high)
    hit_session_low := math.min(hit_session_low, low)

var bool lower_manipulation_first_hit = false
var bool upper_distribution_first_hit = false
var bool lower_distribution_first_hit = false
var bool upper_distribution_std2_first_hit = false
var bool lower_distribution_std2_first_hit = false

    tue_sessions := 0

    thu_sessions := 0
    fri_sessions := 0
    mon_upper_manip_hits := 0
    tue_upper_manip_hits := 0
    wed_upper_manip_hits := 0

    fri_upper_manip_hits := 0

    tue_lower_dist_std2_hits := 0
    wed_lower_dist_std2_hits := 0
    thu_lower_dist_std2_hits := 0
    fri_lower_dist_std2_hits := 0

if is_in_session_stats and not na(avg_manipulation) and not na(avg_distribution) and not na(std_dev_distribution)
    string current_day = get_weekday()
    // Upper Manipulation
    if not upper_manipulation_first_hit and hit_session_high >= hit_session_open + avg_manipulation

        else if current_day == "Tuesday"

        else if current_day == "Friday"
            fri_upper_manip_hits := fri_upper_manip_hits + 1
        upper_manipulation_first_hit := true
    
    // Lower Manipulation
    if not lower_manipulation_first_hit and hit_session_low <= hit_session_open - avg_manipulation

        else if current_day == "Tuesday"

    // Upper Distribution
    if not upper_distribution_first_hit and hit_session_high >= hit_session_open + avg_distribution

            mon_upper_dist_hits := mon_upper_dist_hits + 1
        else if current_day == "Tuesday"
            tue_upper_dist_hits := tue_upper_dist_hits + 1
        else if current_day == "Wednesday"

        else if current_day == "Thursday"

        else if current_day == "Friday"

    if not lower_distribution_std2_first_hit and hit_session_low <= hit_session_open - (avg_distribution + 2 * std_dev_distribution)

            mon_lower_dist_std2_hits := mon_lower_dist_std2_hits + 1
        else if current_day == "Tuesday"
            tue_lower_dist_std2_hits := tue_lower_dist_std2_hits + 1
