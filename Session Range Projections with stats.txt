// Function to track hits (updated to include first hit percentage)
get_OTP_modified(level_index, is_wick, is_tsd) =>
    float pct_minus_to_plus = is_tsd ? (first_hit_minus_tsd.get(level_index) > 0 ? (both_hit_tsd.get(level_index) / first_hit_minus_tsd.get(level_index)) * 100 : 0) : is_wick ? (first_hit_minus_wick.get(level_index) > 0 ? (both_hit_wick.get(level_index) / first_hit_minus_wick.get(level_index)) * 100 : 0) : (first_hit_minus_body.get(level_index) > 0 ? (both_hit_body.get(level_index) / first_hit_minus_body.get(level_index)) * 100 : 0)
    float pct_plus_to_minus = is_tsd ? (first_hit_plus_tsd.get(level_index) > 0 ? (both_hit_tsd.get(level_index) / first_hit_plus_tsd.get(level_index)) * 100 : 0) : is_wick ? (first_hit_plus_wick.get(level_index) > 0 ? (both_hit_wick.get(level_index) / first_hit_plus_wick.get(level_index)) * 100 : 0) : (first_hit_plus_body.get(level_index) > 0 ? (both_hit_body.get(level_index) / first_hit_plus_body.get(level_index)) * 100 : 0)
    float pct_hit_minus = is_tsd ? (total_sessions > 0 ? (first_hit_minus_tsd.get(level_index) / total_sessions) * 100 : 0) : is_wick ? (total_sessions > 0 ? (first_hit_minus_wick.get(level_index) / total_sessions) * 100 : 0) : (total_sessions > 0 ? (first_hit_minus_body.get(level_index) / total_sessions) * 100 : 0)
    float pct_hit_plus = is_tsd ? (total_sessions > 0 ? (first_hit_plus_tsd.get(level_index) / total_sessions) * 100 : 0) : is_wick ? (total_sessions > 0 ? (first_hit_plus_wick.get(level_index) / total_sessions) * 100 : 0) : (total_sessions > 0 ? (first_hit_plus_body.get(level_index) / total_sessions) * 100 : 0)
    bool hit_both = is_tsd ? (hit_minus_tsd.get(level_index) and hit_plus_tsd.get(level_index)) : is_wick ? (hit_minus_wick.get(level_index) and hit_plus_wick.get(level_index)) : (hit_minus_body.get(level_index) and hit_plus_body.get(level_index))
    str_minus_to_plus = str.tostring(math.round(pct_minus_to_plus, 1)) + "%"
    str_plus_to_minus = str.tostring(math.round(pct_plus_to_minus, 1)) + "%"
    str_hit_minus = str.tostring(math.round(pct_hit_minus, 1)) + "%"
    str_hit_plus = str.tostring(math.round(pct_hit_plus, 1)) + "%"
    [str_minus_to_plus, str_plus_to_minus, hit_both, str_hit_minus, str_hit_plus]
