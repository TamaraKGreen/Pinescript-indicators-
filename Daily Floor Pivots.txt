can_draw(a, i) =>
    i >= 0 and i < array.size(a) and not na(array.get(a, i))

    can_draw(a, i) ? array.get(a, i) : na

draw_line(i, pivot, col, is_current = false) =>
    line aLine = na
    okArrTime = array.size(arr_time) > 1 and i >= 0 and i < array.size(arr_time) - 1
    okPivot = i >= 0 and i < array.size(pivot)
    if okArrTime and okPivot
        y = array.get(pivot, i)
        if not na(y)
            x2 = is_current ? time : array.get(arr_time, i + 1)
            aLine := line.new(array.get(arr_time, i), y, x2, y, xloc = xloc.bar_time, color = col, width = line_width)
            array.push(lines, aLine)
    aLine

draw_label(i, y, txt, txt_color, is_gz = false, is_inter_gz = false) =>
    if na(y)
        na
    else
        _showLbl = is_gz ? gz_show_labels : is_inter_gz ? gz_inter_labels : sr_show_labels
        _showPrc = is_gz ? gz_show_prices : is_inter_gz ? gz_inter_prices : sr_show_prices
        if _showLbl or _showPrc
            display = ''
            if _showLbl and _showPrc and txt != ''
                display := str.format('                                {0} - {1}', txt, math.round_to_mintick(y))
            else if _showLbl and txt != ''
                display := '                           ' + txt
            else
                display := str.format('                                {0}', math.round_to_mintick(y))
            rightIndex = math.min(i + 1, array.size(arr_time) - 1)
            leftIndex = math.min(i, array.size(arr_time) - 1)
            if rightIndex >= 0 and leftIndex >= 0
                x = position_labels == 'Left' ? array.get(arr_time, leftIndex) : array.get(arr_time, rightIndex)
                array.push(labels, label.new(x = x, y = y, text = display, textcolor = txt_color, style = label.style_none, color = #00000000, xloc = xloc.bar_time))

// ——— Floor Pivot calculation ———
floor_pivots() =>
    pivot_point = (pivotX_prev_high + pivotX_prev_low + pivotX_prev_close) / 3
    _r1 = (2 * pivot_point) - pivotX_prev_low
    _r2 = pivot_point + (pivotX_prev_high - pivotX_prev_low)
    _r3 = pivotX_prev_high + 2 * (pivot_point - pivotX_prev_low)
    _s1 = (2 * pivot_point) - pivotX_prev_high
    _s2 = pivot_point - (pivotX_prev_high - pivotX_prev_low)
    _s3 = pivotX_prev_low - 2 * (pivotX_prev_high - pivot_point)
    
    array.push(pp, pivot_point)
    array.push(r1, _r1)
    array.push(s1, _s1)
    array.push(r2, _r2)
    array.push(s2, _s2)
    array.push(r3, _r3)
    array.push(s3, _s3)
    
    // PP to R1 Golden Zone
    _pp_r1_range = _r1 - pivot_point
    if gz_from_top
        array.push(gz_pp_r1_high, _r1 - 0.5 * _pp_r1_range)
        array.push(gz_pp_r1_low, _r1 - 0.618 * _pp_r1_range)
    else
        array.push(gz_pp_r1_low, pivot_point + 0.5 * _pp_r1_range)
        array.push(gz_pp_r1_high, pivot_point + 0.618 * _pp_r1_range)
    
    // PP to S1 Golden Zone
    _pp_s1_range = pivot_point - _s1
    if gz_from_top
        array.push(gz_pp_s1_high, _s1 + 0.618 * _pp_s1_range)
        array.push(gz_pp_s1_low, _s1 + 0.5 * _pp_s1_range)
    else
        array.push(gz_pp_s1_high, pivot_point - 0.5 * _pp_s1_range)
        array.push(gz_pp_s1_low, pivot_point - 0.618 * _pp_s1_range)
    
    // R1 to R2 Golden Zone
    _r1_r2_range = _r2 - _r1
    if gz_from_top
        array.push(gz_r1_r2_high, _r2 - 0.5 * _r1_r2_range)
        array.push(gz_r1_r2_low, _r2 - 0.618 * _r1_r2_range)
    else
        array.push(gz_r1_r2_low, _r1 + 0.5 * _r1_r2_range)
        array.push(gz_r1_r2_high, _r1 + 0.618 * _r1_r2_range)
    
    // S1 to S2 Golden Zone
    _s1_s2_range = _s1 - _s2
    if gz_from_top
        array.push(gz_s1_s2_high, _s2 + 0.618 * _s1_s2_range)
        array.push(gz_s1_s2_low, _s2 + 0.5 * _s1_s2_range)
    else
        array.push(gz_s1_s2_high, _s1 - 0.5 * _s1_s2_range)
        array.push(gz_s1_s2_low, _s1 - 0.618 * _s1_s2_range)
    
    // R2 to R3 Golden Zone
    _r2_r3_range = _r3 - _r2
    if gz_from_top
        array.push(gz_r2_r3_high, _r3 - 0.5 * _r2_r3_range)
        array.push(gz_r2_r3_low, _r3 - 0.618 * _r2_r3_range)
    else
        array.push(gz_r2_r3_low, _r2 + 0.5 * _r2_r3_range)
        array.push(gz_r2_r3_high, _r2 + 0.618 * _r2_r3_range)
    
    // S2 to S3 Golden Zone
    _s2_s3_range = _s2 - _s3
    if gz_from_top
        array.push(gz_s2_s3_high, _s3 + 0.618 * _s2_s3_range)
        array.push(gz_s2_s3_low, _s3 + 0.5 * _s2_s3_range)
    else
        array.push(gz_s2_s3_high, _s2 - 0.5 * _s2_s3_range)
        array.push(gz_s2_s3_low, _s2 - 0.618 * _s2_s3_range)

    // Calculate R4/S4
    _r4 = _r3 + (pivotX_prev_high - pivotX_prev_low)
    _s4 = _s3 - (pivotX_prev_high - pivotX_prev_low)
    array.push(r4, _r4)
    array.push(s4, _s4)
    
    // Calculate R5/S5
    _r5 = _r4 + (pivotX_prev_high - pivotX_prev_low)
    _s5 = _s4 - (pivotX_prev_high - pivotX_prev_low)
    array.push(r5, _r5)
    array.push(s5, _s5)
    
    // R3 to R4 Golden Zone
    _r3_r4_range = _r4 - _r3
    if gz_from_top
        array.push(gz_r3_r4_high, _r4 - 0.5 * _r3_r4_range)
        array.push(gz_r3_r4_low, _r4 - 0.618 * _r3_r4_range)
    else
        array.push(gz_r3_r4_low, _r3 + 0.5 * _r3_r4_range)
        array.push(gz_r3_r4_high, _r3 + 0.618 * _r3_r4_range)
    
    // S3 to S4 Golden Zone
    _s3_s4_range = _s3 - _s4
    if gz_from_top
        array.push(gz_s3_s4_high, _s4 + 0.618 * _s3_s4_range)
        array.push(gz_s3_s4_low, _s4 + 0.5 * _s3_s4_range)
    else
        array.push(gz_s3_s4_high, _s3 - 0.5 * _s3_s4_range)
        array.push(gz_s3_s4_low, _s3 - 0.618 * _s3_s4_range)
    
    // R4 to R5 Golden Zone
    _r4_r5_range = _r5 - _r4
    if gz_from_top
        array.push(gz_r4_r5_high, _r5 - 0.5 * _r4_r5_range)
        array.push(gz_r4_r5_low, _r5 - 0.618 * _r4_r5_range)
    else
        array.push(gz_r4_r5_low, _r4 + 0.5 * _r4_r5_range)
        array.push(gz_r4_r5_high, _r4 + 0.618 * _r4_r5_range)
    
    // S4 to S5 Golden Zone
    _s4_s5_range = _s4 - _s5
    if gz_from_top
        array.push(gz_s4_s5_high, _s5 + 0.618 * _s4_s5_range)
        array.push(gz_s4_s5_low, _s5 + 0.5 * _s4_s5_range)
    else
        array.push(gz_s4_s5_high, _s4 - 0.5 * _s4_s5_range)
        array.push(gz_s4_s5_low, _s4 - 0.618 * _s4_s5_range)

calc_golden_zone() =>
    _range = pivotX_prev_high - pivotX_prev_low
    if gz_from_top
        // Calculate from top down (61.8% and 50% from the high)
        gz_50_level = pivotX_prev_high - 0.5 * _range
        gz_618_level = pivotX_prev_high - 0.618 * _range
        array.push(gz_50, gz_50_level)
        array.push(gz_618, gz_618_level)
    else
        // Calculate from bottom up (50% and 61.8% from the low)
        gz_50_level = pivotX_prev_low + 0.5 * _range
        gz_618_level = pivotX_prev_low + 0.618 * _range
        array.push(gz_50, gz_50_level)
        array.push(gz_618, gz_618_level)
        
calc_pivot() =>
    floor_pivots()
    calc_golden_zone()
