// Session Range Function
get_range(session, session_name, session_css, high_array, low_array)=>
    var t = 0 
    var max = high
    var min = low
    var box bx = na
    var label lbl = na 
    var int start_idx = na
    var float vol = 0.0
    var float atr_sum = 0.0
    var int bar_count = 0
    
    if session > session[1]
        t := time
        max := high
        min := low
        start_idx := bar_index
        vol := 0.0
        atr_sum := 0.0
        bar_count := 0
        bx := box.new(bar_index, max, bar_index, min, bgcolor=color.new(session_css, bg_transp), border_color=show_outline ? session_css : na, border_style=line.style_dotted)
        if show_txt
            lbl := label.new(t, max, session_name, xloc=xloc.bar_time, textcolor=session_css, style=label.style_label_down, color=color.new(color.white, 100), size=size.tiny)
    
    if session and session == session[1]
        max := math.max(high, max)
        min := math.min(low, min)
        vol := vol + volume
        atr_sum := atr_sum + (high - low)
        bar_count := bar_count + 1
        box.set_top(bx, max)
        box.set_rightbottom(bx, bar_index, min)
        if show_txt
            label.set_xy(lbl, int(math.avg(t, time)), max)
    
    if session[1] and not session
        array.push(high_array, max)
        array.push(low_array, min)
    
    [session ? na : max, session ? na : min, start_idx, session ? na : vol, session ? na : (bar_count > 0 ? atr_sum / bar_count : na)]

// Session Timezone

// Pattern Detection Function
get_session_patterns(max_sesb, min_sesb, max_sesc, min_sesc) =>
    var string pattern = ""
    
    if not na(max_sesb) and not na(min_sesb) and not na(max_sesc) and not na(min_sesc)
        // Pattern 1: London engulfs Asia (Asia high and low taken)
        if max_sesb > max_sesc and min_sesb < min_sesc
            pattern := "London Engulfs Asia"
        
        // Pattern 2: Asia engulfs London (neither Asia high nor low taken)
        else if max_sesc >= max_sesb and min_sesc <= min_sesb
            pattern := "Asia Engulfs London"
        
        // Pattern 3: London Partially Engulfs Upwards (Asia high taken, but not the low)
        else if max_sesb > max_sesc and min_sesb >= min_sesc
            pattern := "London Partial Up"
        
        // Pattern 4: London Partially Engulfs Downwards (Asia low taken, but not the high)
        else if max_sesb <= max_sesc and min_sesb < min_sesc
            pattern := "London Partial Down"
    
    pattern

// First Taken Detection
var string first_taken = "None"
if is_sesa and not na(max_sesb) and not na(min_sesb)
    if high > max_sesb and first_taken == "None"
        first_taken := "London High"
    else if low < min_sesb and first_taken == "None"
        first_taken := "London Low"

// Reset first_taken at the start of a new day

// NY Outcome Function
get_ny_outcome(pattern, first_taken) =>
    var string outcome = ""
    
    if pattern == "London Engulfs Asia"
        if first_taken == "London High"
            outcome := "98% NY takes Lon Hi/Lo\n45% NY takes Lon Lo (after Hi first)\n43% NY engulfs overnight"
        else if first_taken == "London Low"
            outcome := "98% NY takes Lon Hi/Lo\n44% NY takes Lon Hi (after Lo first)\n43% NY engulfs overnight"
        else
            outcome := "98% NY takes Lon Hi/Lo\n44% NY takes Lon Hi (if Lo first)\n45% NY takes Lon Lo (if Hi first)\n43% NY engulfs overnight"
    
    else if pattern == "Asia Engulfs London"
        outcome := "95% NY takes Lon Hi/Lo\n91% NY takes Asia Hi/Lo\n39% NY engulfs overnight"
    
    else if pattern == "London Partial Up"
        if first_taken == "London Low"
            outcome := "50% NY takes Lon Hi (Lo taken first)\n63% NY takes Lon Lo\n51% NY takes Asia Lo\n35% NY engulfs overnight"
        else if first_taken == "London High"
            outcome := "79% NY takes Lon Hi\n48% NY takes Lon Lo (Hi taken first)\n51% NY takes Asia Lo\n35% NY engulfs overnight"
        else
            outcome := "79% NY takes Lon Hi (50% if Lo first)\n63% NY takes Lon Lo (48% if Hi first)\n51% NY takes Asia Lo\n35% NY engulfs overnight"
    
    else if pattern == "London Partial Down"
        if first_taken == "London High"
            outcome := "66% NY takes Lon Hi\n43% NY takes Lon Lo (Hi taken first)\n54% NY takes Asia Hi\n33% NY engulfs overnight"
        else if first_taken == "London Low"
            outcome := "41% NY takes Lon Hi (Lo taken first)\n73% NY takes Lon Lo\n54% NY takes Asia Hi\n33% NY engulfs overnight"
        else
            outcome := "66% NY takes Lon Hi (41% if Lo first)\n73% NY takes Lon Lo (43% if Hi first)\n54% NY takes Asia Hi\n33% NY engulfs overnight"
    
    outcome == "" ? "No data" : outcome
