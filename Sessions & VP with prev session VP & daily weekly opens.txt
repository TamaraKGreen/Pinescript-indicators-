utcHour    = hour(time(timeframe.period, '0000-2400', 'GMT'), 'GMT')
newTokyo   = utcHour != utcHour[1] + 1 and utcHour != utcHour[1]
endTokyo   = utcHour >= 9  and utcHour[1] < 9
newLondon  = utcHour >= 7  and utcHour[1] < 7
endLondon  = utcHour >= 16 and utcHour[1] < 16
newNewYork = utcHour >= 13 and utcHour[1] < 13
endNewYork = utcHour >= 22 and utcHour[1] < 22
newDaily     = dayofweek  != dayofweek[1]
newWeekly    = weekofyear != weekofyear[1]
newMonthly   = (dayofmonth != dayofmonth[1] + 1) and (dayofmonth != dayofmonth[1])
newYearly    = year != year[1]
newQuarterly = month != month[1] and (month - 1) % 3 == 0

sessionNew(t) =>
    switch t
        'Tokyo'     => newTokyo
        'London'    => newLondon
        'New York'  => newNewYork
        'Daily'     => newDaily
        'Weekly'    => newWeekly
        'Monthly'   => newMonthly
        'Yearly'    => newYearly
        'Quarterly' => newQuarterly

get_vol(y11, y12, y21, y22, height, v) =>
    nz(math.max(math.min(math.max(y11, y12), math.max(y21, y22)) - math.max(math.min(y11, y12), math.min(y21, y22)), 0) * v / height)

profileAdd(o, h, l, c, v, g, vGreen, vRed, zBounds, res, vType) =>
    body_top   = math.max(c, o)
    body_bot   = math.min(c, o)
    itsgreen   = c >= o
    topwick    = h - body_top
    bottomwick = body_bot - l
    body       = body_top - body_bot
    denom      = 2 * topwick + 2 * bottomwick + body
    bodyvol    = denom > 0 ? body           * v / denom : 0.0
    topwvol    = denom > 0 ? 2 * topwick    * v / denom : 0.0
    botwvol    = denom > 0 ? 2 * bottomwick * v / denom : 0.0
    for i = 0 to res - 1
        zTop = array.get(zBounds, i)
        zBot = zTop - g
        if vType == 'Volume'
            gAdd = (itsgreen ? get_vol(zBot, zTop, body_bot, body_top, body, bodyvol) : 0.0) + get_vol(zBot, zTop, body_top, h, topwick, topwvol) / 2 + get_vol(zBot, zTop, body_bot, l, bottomwick, botwvol) / 2
            rAdd = (itsgreen ? 0.0 : get_vol(zBot, zTop, body_bot, body_top, body, bodyvol)) + get_vol(zBot, zTop, body_top, h, topwick, topwvol) / 2 + get_vol(zBot, zTop, body_bot, l, bottomwick, botwvol) / 2
            array.set(vGreen, i, array.get(vGreen, i) + gAdd)
            array.set(vRed,   i, array.get(vRed,   i) + rAdd)
        else if vType == 'Open Interest'
            if v > 0
                array.set(vGreen, i, array.get(vGreen, i) + get_vol(zBot, zTop, body_bot, body_top, body,  v))
            if v < 0
                array.set(vRed,   i, array.get(vRed,   i) + get_vol(zBot, zTop, body_bot, body_top, body, -v))

calcSession(update, pH, pL, res, ltfO, ltfC, ltfH, ltfL, ltfV, vGreen, vRed, zBounds, vType, lookback) =>
    array.fill(vGreen, 0)
    array.fill(vRed, 0)
    if update and lookback > 0 and not na(pH) and not na(pL) and pH != pL
        gap = (pH - pL) / res
        for i = 0 to res - 1
            array.set(zBounds, i, pH - gap * i)
        if array.size(ltfO) > 0
            for j = 0 to array.size(ltfO) - 1
                profileAdd(array.get(ltfO, j), array.get(ltfH, j), array.get(ltfL, j), array.get(ltfC, j), array.get(ltfV, j), gap, vGreen, vRed, zBounds, res, vType)

pocLevel(vGreen, vRed, zBounds, res) =>
    float maxVol = 0
    int   ind    = 0
    for i = 0 to res - 1
        tot = array.get(vRed, i) + array.get(vGreen, i)
        if tot > maxVol
            maxVol := tot
            ind    := i
    float out = na
    if ind != res - 1
        out := array.get(zBounds, ind) - (array.get(zBounds, ind) - array.get(zBounds, ind + 1)) / 2
    out

valueLevels(poc, pH, pL, vGreen, vRed, zBounds, res, vaWidPct) =>
    float gap    = (pH - pL) / res
    float volSum = array.sum(vRed) + array.sum(vGreen)
    float volCnt = 0.0
    float vah    = pH
    float val    = pL
    int   pocInd = 0
    for i = 0 to res - 2
        if array.get(zBounds, i) >= poc and array.get(zBounds, i + 1) < poc
            pocInd := i
    volCnt += array.get(vRed, pocInd) + array.get(vGreen, pocInd)
    for i = 1 to res
        if pocInd + i < res
            volCnt += array.get(vRed, pocInd + i) + array.get(vGreen, pocInd + i)
            if volCnt >= volSum * (vaWidPct / 100)
                break
            else
                val := array.get(zBounds, pocInd + i) - gap
        if pocInd - i >= 0
            volCnt += array.get(vRed, pocInd - i) + array.get(vGreen, pocInd - i)
            if volCnt >= volSum * (vaWidPct / 100)
                break
            else
                vah := array.get(zBounds, pocInd - i)
    [val, vah]

        box.delete(outBox)
        label.delete(sesLabel)
        label.delete(pocLabel)
        label.delete(vahLabel)
        label.delete(valLabel)
        for i = 0 to res - 1
            box.delete(array.get(greenBoxes, i))
            box.delete(array.get(redBoxes,   i))

// Draw the single snapshotted previous session across the current session's bar range
drawPrevOverlay(update, delete, curStart, res, bullC, bearC, pocC, pocW, vahC, vahW, valC, valW, VAbC, showProf, showPoc, showVA, showVAb, showLab, dispMode, sesType, vaWidPct, showLvlLab, pocTxt, vahTxt, valTxt, lvlLabSize, lvlLabCol) =>
    var box[]  gBoxes = array.new_box(res, na)
    var box[]  rBoxes = array.new_box(res, na)
    var line   pocL   = na
    var line   vahL   = na
    var line   valL   = na
    var label  sesLab = na
    var label  pocLb  = na
    var label  vahLb  = na
    var label  valLb  = na
    var box    vab    = na

    if update and snapReady
        // Clear previous drawings
        for i = 0 to res - 1
            box.delete(array.get(gBoxes, i))
            box.delete(array.get(rBoxes, i))
        line.delete(pocL)
        line.delete(vahL)
        line.delete(valL)
        label.delete(sesLab)
        label.delete(pocLb)
        label.delete(vahLb)
        label.delete(valLb)
        box.delete(vab)

        pH   = snapHigh
        pL   = snapLow
        gap  = (pH - pL) / res
        buf  = gap / 10
        // Profile spans left half of current session (matching drawCurZone's right boundary)
        float sLeft  = float(curStart)
        float sRight = bar_index - (bar_index - curStart) * 0.3

        float maxV = 0.0
        for i = 0 to res - 1
            tot = array.get(snapGreen, i) + array.get(snapRed, i)
            if tot > maxV
                maxV := tot

        if showProf and maxV > 0
            for i = 0 to res - 1
                gVol = array.get(snapGreen, i)
                rVol = array.get(snapRed, i)
                zTop = array.get(snapBounds, i)
                zBot = zTop - gap
                gEnd = int(sLeft + (sRight - sLeft) * (gVol / maxV))
                rEnd = int(gEnd  + (sRight - sLeft) * (rVol / maxV))
                if dispMode == 'Mode 2'
                    array.set(gBoxes, i, box.new(int(sLeft), zTop-buf, gEnd, zBot+buf, bgcolor=bullC, border_width=0))
                    array.set(rBoxes, i, box.new(gEnd,       zTop-buf, rEnd, zBot+buf, bgcolor=bearC, border_width=0))
                else if dispMode == 'Mode 1'
                    array.set(gBoxes, i, box.new(int(sLeft), zTop-buf, gEnd, zBot+buf, bgcolor=bullC, border_width=0))
                else
                    array.set(gBoxes, i, box.new(int(sLeft),               zTop-buf, gEnd,       zBot+buf, bgcolor=bullC, border_width=0))
                    array.set(rBoxes, i, box.new(int(sLeft) - rEnd + gEnd, zTop-buf, int(sLeft), zBot+buf, bgcolor=bearC, border_width=0))

        if showLab
            sesLab := label.new(int((sLeft + sRight) / 2), pH, 'Prev ' + sesType, color=color.rgb(0,0,0,100), textcolor=chart.fg_color, style=label.style_label_down, size=size.small)
        if showPoc and not na(snapPoc)
            pocL := line.new(int(sLeft), snapPoc, bar_index, snapPoc, color=pocC, width=pocW, style=line.style_dashed)
            if showLvlLab
                pocLb := label.new(bar_index, snapPoc, 'P: ' + pocTxt, color=color.new(color.black,100), textcolor=lvlLabCol, style=label.style_none, size=labelSize(lvlLabSize))
        if showVA and not na(snapVah) and not na(snapVal)
            vahL := line.new(int(sLeft), snapVah, bar_index, snapVah, color=vahC, width=vahW, style=line.style_dashed)
            valL := line.new(int(sLeft), snapVal, bar_index, snapVal, color=valC, width=valW, style=line.style_dashed)
            if showLvlLab
                vahLb := label.new(bar_index, snapVah, 'P: ' + vahTxt, color=color.new(color.black,100), textcolor=lvlLabCol, style=label.style_none, size=labelSize(lvlLabSize))
                valLb := label.new(bar_index, snapVal, 'P: ' + valTxt, color=color.new(color.black,100), textcolor=lvlLabCol, style=label.style_none, size=labelSize(lvlLabSize))
        if showVAb and not na(snapVah) and not na(snapVal)
            vab := box.new(int(sLeft), snapVah, bar_index, snapVal, border_color=color.rgb(54,58,69,100), bgcolor=VAbC)

    if delete
        for i = 0 to res - 1
            box.delete(array.get(gBoxes, i))
            box.delete(array.get(rBoxes, i))
        line.delete(pocL)
        line.delete(vahL)
        line.delete(valL)
        label.delete(sesLab)
        label.delete(pocLb)
        label.delete(vahLb)
        label.delete(valLb)
        box.delete(vab)
