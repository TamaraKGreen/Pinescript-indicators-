// GAUSSIAN KERNEL SMOOTHING FUNCTIONS
ema_gk(data, alpha, len) =>
    ema = 0.0
    ema := alpha * data + (1 - alpha) * nz(ema[1])

stdev_gk(data, alpha, len) =>
    ema = ema_gk(data, alpha, len)
    squaredDeviation = math.pow(data - ema, 2)
    smoothed = 0.0
    smoothed := alpha * squaredDeviation + (1 - alpha) * nz(smoothed[1])
    math.sqrt(smoothed)

bandwidth_gk(sd, len) =>
    math.pow((4 * math.pow(sd, 5) / (3 * len)), 1.0/5.0)

gaussian_Kernel_Smoothing(data, b, len) =>
    nu = 0.0
    de = 0.0
    for i = 0 to len - 1
        weight = math.exp(-math.pow(data[len-1] - data[i], 2) / (2 * math.pow(b, 2)))
        nu := nu + (weight * data[i])
        de := de + weight
    nu / de

// T3 Moving Average (Tillson T3)
t3_ma(src, length, vFactor) =>
    e1 = ta.ema(src, length)
    e2 = ta.ema(e1, length)
    e3 = ta.ema(e2, length)
    e4 = ta.ema(e3, length)
    e5 = ta.ema(e4, length)
    e6 = ta.ema(e5, length)
    
    c1 = -vFactor * vFactor * vFactor
    c2 = 3 * vFactor * vFactor + 3 * vFactor * vFactor * vFactor
    c3 = -6 * vFactor * vFactor - 3 * vFactor - 3 * vFactor * vFactor * vFactor
    c4 = 1 + 3 * vFactor + vFactor * vFactor * vFactor + 3 * vFactor * vFactor
    
    c1 * e6 + c2 * e5 + c3 * e4 + c4 * e3

// Zero-Lag EMA
zema_ma(src, length) =>
    ema1 = ta.ema(src, length)
    ema2 = ta.ema(ema1, length)
    difference = ema1 - ema2
    ema1 + difference

// Kaufman Adaptive Moving Average (KAMA)
kama_ma(src, length) =>
    xvnoise = math.abs(src - src[1])
    nsignal = math.abs(src - src[length])
    nnoise = math.sum(xvnoise, length)
    nefratio = nnoise != 0 ? nsignal / nnoise : 0
    
    nfastend = 0.666
    nslowend = 0.0645
    nsmooth = math.pow(nefratio * (nfastend - nslowend) + nslowend, 2)
    
    nAMA = 0.0
    nAMA := nz(nAMA[1]) + nsmooth * (src - nz(nAMA[1]))
    nAMA

method st(simple string src, float osc, simple int len) =>
    float o = switch src
        "SMA" => ta.sma(osc, len)
        "EMA" => ta.ema(osc, len)
        "WMA" => ta.wma(osc, len)
        "HMA" => ta.hma(osc, len)
        "RMA" => ta.rma(osc, len)
        "VWMA" => ta.vwma(osc, len)
        "T3" => t3_ma(osc, len, t3VolumeFactor)
        "ZEMA" => zema_ma(osc, len)
        "KAMA" => kama_ma(osc, len)
        "GKEMA" => 
            alpha = 2.0 / (len + 1)
            logData = math.log(math.abs(osc) + 1e-10)  // Add small value to prevent log(0)
            standardDev = stdev_gk(logData, alpha, len)
            movingAvg = ema_gk(logData, alpha, len)
            blc = bandwidth_gk(standardDev, len) * gkBandwidth
            smoothed = gaussian_Kernel_Smoothing(movingAvg, blc, len)
            math.exp(smoothed) * math.sign(osc)  // Restore original sign

         : color.new(x, 0 )

method transp(color x, int t) =>
    color.new(x, t)

// SIGNAL STRENGTH CALCULATION
calculateSignalStrength() =>
    bullishConditions = 0
    bearishConditions = 0
    
    // Condition 1: Main oscillator direction
    if osc.sig > osc.sgD
        bullishConditions += 1
    else
        bearishConditions += 1
    
    // Condition 2: Money flow direction
    if mf.mfi > 0
        bullishConditions += 1
    else
        bearishConditions += 1
    
    // Condition 3: Money flow strength vs average
    if mf.mfi > cf.blT.avg()
        bullishConditions += 1
    if mf.mfi < cf.brT.avg()
        bearishConditions += 1
    
    // Condition 4: Oscillator above/below zero
    if osc.sig > 0
        bullishConditions += 1
    if osc.sig < 0
        bearishConditions += 1
    
    // Condition 5: Extreme levels
    if osc.sig > 30
        bearishConditions += 1  // Overbought
    if osc.sig < -30
        bullishConditions += 1  // Oversold
    
    // Condition 6: Volume confirmation
    vMA = ta.sma(volume, 7)
    if volume > vMA * 1.5
        if osc.sig > osc.sgD
            bullishConditions += 1
        else
            bearishConditions += 1
    
    // Condition 7: Convergence
    if osc.sig > 0 and mf.mfi > 0
        bullishConditions += 1
    if osc.sig < 0 and mf.mfi < 0
        bearishConditions += 1
    
    // Calculate scores (1-10)
    maxConditions = 7
    bullishScore = math.round(bullishConditions * 10 / maxConditions)
    bearishScore = math.round(bearishConditions * 10 / maxConditions)
    
    ss.bullishScore := bullishScore
    ss.bearishScore := bearishScore

    float hi = ta.highest( len)
    float lo = ta.lowest ( len)
    float av = ta.sma (hl2, len)

    osc.sig := ta.ema(ta.linreg((close - math.avg(hi, lo, av)) / (hi - lo) * 100, len, 0), sL)
    osc.sgD := sT.st(osc.sig, 2)
    osc.cO := osc.sig > osc.sgD ? fCSS.css(1) : sCSS.css(1)

    mf.mfi := ta.sma(ta.mfi(hl2, mfL) - 50, mfS)
