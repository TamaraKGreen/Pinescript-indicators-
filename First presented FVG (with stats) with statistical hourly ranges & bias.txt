getHourEndTime(int t, string tz) =>
    int hourStart = timestamp(tz, year(t, tz), month(t, tz), dayofmonth(t, tz), hour(t, tz), 0, 0)
    hourStart + 3600000

getRthCloseTime(int t, string tz) =>
    // 4:15pm on the same NY calendar date
    timestamp(tz, year(t, tz), month(t, tz), dayofmonth(t, tz), 16, 15, 0)

int  currentMinute   = minute(time, sessionTimezone)
bool isNewHour       = na(lastHour) or currentHour != lastHour

    lastHour             := currentHour
    currentHourFvgFound  := false

// Reset RTH window tracking at start of each new day (when we hit 9am or earlier pre-market)

    w1FvgDirection := "None"
    w2FvgDirection := "None"
    w1FvgFound     := false
    w2FvgFound     := false
    w1WindowClosed := false
    w2WindowClosed := false

    w1WindowClosed := true

        w2FvgFound     := true

        w2FvgDirection := "SIBI"

            contProb   := str.tostring(array.get(bearContinuation, h), "#.#") + "% ↓"
            contBgColor := color.new(color.red, 80)
        else
            contProb := "Neutral"
    table.cell(statsTable, 0, 3, "Continuation %",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 3, contProb, bgcolor=contBgColor,
         text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    table.cell(statsTable, 0, 4, "Avg Range %",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 4, str.tostring(array.get(hourlyAvgRange, h), "#.###") + "%",
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    string fvgEffect = ""
    if currentHourBullFvg
        fvgEffect := str.tostring(array.get(bisiEffectiveness, h), "#.#") + "% ↑"
    else if currentHourBearFvg
        fvgEffect := str.tostring(array.get(sibiEffectiveness, h), "#.#") + "% ↓"
    table.cell(statsTable, 0, 5, "FVG Effect %",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 5, fvgEffect,
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    string timeLeft = "-"
    if not na(hourlyOpenTime)
        int remainingMs = 3600000 - (timenow - hourlyOpenTime)
        if remainingMs < 0
            timeLeft := "00:00"
        else
            int remainingMin = math.floor(remainingMs / 60000)
            int remainingSec = math.floor((remainingMs % 60000) / 1000)
            string minStr = remainingMin < 10 ? "0" + str.tostring(remainingMin) : str.tostring(remainingMin)
            string secStr = remainingSec < 10 ? "0" + str.tostring(remainingSec) : str.tostring(remainingSec)
            timeLeft := minStr + ":" + secStr
    table.cell(statsTable, 0, 6, "Time Left",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 6, timeLeft,
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    float  bias    = array.get(hourlyBullishBias, h)
    string biasStr = bias > 50 ? "↑ " + str.tostring(bias, "#.#") + "%" : "↓ " + str.tostring(100 - bias, "#.#") + "%"
    table.cell(statsTable, 0, 7, "Hourly Bias",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 7, biasStr,
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    table.cell(statsTable, 0, 8, "H Open",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 8, not na(hourlyOpen) ? str.tostring(hourlyOpen, format.mintick) : "-",
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    string rangeStr = "-"
    if not na(hourlyOpen)
        float avgRangeVal = array.get(hourlyAvgRange, h)
        rangeStr := str.tostring(hourlyOpen - (avgRangeVal/100*hourlyOpen), format.mintick) + " - " +
                     str.tostring(hourlyOpen + (avgRangeVal/100*hourlyOpen), format.mintick)
    table.cell(statsTable, 0, 9, "Exp Range",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 9, rangeStr,
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    // ── rows 10–15: RTH confluence section ────────────────────────────────

    // Row 10: divider header
    table.cell(statsTable, 0, 10, "── RTH CONFLUENCE ──", bgcolor=color.navy,
         text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 10, "9:30–4:15pm",         bgcolor=color.navy,
         text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    // Row 11: W1 FVG signal
    string w1Label  = w1FvgDirection == "BISI" ? "BISI ↑" : w1FvgDirection == "SIBI" ? "SIBI ↓" : "No FVG"
    color  w1LabelBg = w1FvgDirection == "BISI" ? color.new(#00bfff, 60) : w1FvgDirection == "SIBI" ? color.new(#ff6600, 60) : color.white
    table.cell(statsTable, 0, 11, "W1 (9:30-10)",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 11, w1Label, bgcolor=w1LabelBg,
         text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    // Row 12: W2 FVG signal
    string w2Label   = w2FvgDirection == "BISI" ? "BISI ↑" : w2FvgDirection == "SIBI" ? "SIBI ↓" : w2WindowClosed ? "No FVG" : "Pending"
    color  w2LabelBg = w2FvgDirection == "BISI" ? color.new(#00bfff, 60) : w2FvgDirection == "SIBI" ? color.new(#ff6600, 60) : color.white
    table.cell(statsTable, 0, 12, "W2 (10-11am)",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 12, w2Label, bgcolor=w2LabelBg,
         text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    // Row 13: RTH Bull %
    // Row 14: RTH Bear %
    // Row 15: Sample N
    // Logic: show W1-only after 10am, upgrade to confluence after 11am
    bool showConfluence = w2WindowClosed  // after 11am, show full confluence
    bool showW1Only     = w1WindowClosed and not w2WindowClosed  // 10am-11am window

    float rthBullDisplay = na
    float rthBearDisplay = na
    int   rthNDisplay    = na

    if showConfluence
        int ci = getComboIndex(w1FvgDirection, w2FvgDirection)
        rthBullDisplay := array.get(comboBullPct, ci)
        rthBearDisplay := array.get(comboBearPct, ci)
        rthNDisplay    := array.get(comboSample,  ci)
    else if showW1Only
        int wi = getW1Index(w1FvgDirection)
        rthBullDisplay := array.get(w1BullPct, wi)
        rthBearDisplay := array.get(w1BearPct, wi)
        rthNDisplay    := array.get(w1Sample,  wi)

    string rthBullStr  = not na(rthBullDisplay) ? str.tostring(rthBullDisplay, "#.#") + "%" : "-"
    string rthBearStr  = not na(rthBearDisplay) ? str.tostring(rthBearDisplay, "#.#") + "%" : "-"
    string rthNStr     = not na(rthNDisplay)    ? "n=" + str.tostring(rthNDisplay)           : "-"
    string tableMode   = showConfluence ? "(confluence)" : showW1Only ? "(W1 only)" : ""

    // colour the bias cell by which side is dominant
    color bullDisplayBg = not na(rthBullDisplay) and rthBullDisplay > rthBearDisplay ? color.new(color.green, 70) : color.white
    color bearDisplayBg = not na(rthBearDisplay) and rthBearDisplay > rthBullDisplay ? color.new(color.red,   70) : color.white

    table.cell(statsTable, 0, 13, "RTH Bull% " + tableMode,
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 13, rthBullStr, bgcolor=bullDisplayBg,
         text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    table.cell(statsTable, 0, 14, "RTH Bear%",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 14, rthBearStr, bgcolor=bearDisplayBg,
         text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)

    table.cell(statsTable, 0, 15, "RTH Sample",
         bgcolor=color.black, text_color=color.white, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
    table.cell(statsTable, 1, 15, rthNStr,
         bgcolor=color.white, text_color=color.black, text_size=tableTextSizeConverted, text_font_family=font.family_monospace)
