// Get average open-to-high/low for day of week
get_dow_high_low(ses_stats S, int dow, bool is_high) =>
    dow_values = array.new_float(0)
    for [i, d] in S.day
        if d == dow
            dow_values.push(is_high ? S.open_to_high.get(i) : S.open_to_low.get(i))
    result = dow_values.size() > 0 ? dow_values.avg() : 0
    result

// Session statistics update method
method update_ses_stats(ses_stats S, bool t, int max_days) =>
    if t and not t[1]
        S.count += 1
        S.cur_hi := high
        S.cur_lo := low
        S.cur_op := open
        S.day.unshift(dayofweek)
        S.ranges.unshift(0)
        S.open_to_high.unshift(0)
        S.open_to_low.unshift(0)
        
        // Limit arrays to match session days setting
        if S.ranges.size() > max_days
            S.ranges.pop()
            S.day.pop()
            S.open_to_high.pop()
            S.open_to_low.pop()
        
        // Create session box
        if show_session_boxes
            S.cur_box := box.new(bar_index, high, bar_index, low, 
                 bgcolor = color.new(S.session_color, 80), 
                 border_color = S.session_color)
        
        // Create session label (using expected_range_color)
        if show_session_labels
            label_text = S.session_name + " (" + S.session_time + ")"
            S.cur_label := label.new(bar_index, high, label_text,
                 style = label.style_none,
                 color = color.new(color.white, 100),
                 textcolor = expected_range_color,
                 text_font_family = font.family_monospace,
                 size = get_label_size(session_label_size))
                
    else if t
        S.cur_hi := math.max(S.cur_hi, high)
        S.cur_lo := math.min(S.cur_lo, low)
        
        // Update box
        if show_session_boxes and not na(S.cur_box)
            box.set_top(S.cur_box, S.cur_hi)
            box.set_bottom(S.cur_box, S.cur_lo)
            box.set_right(S.cur_box, bar_index)
        
        // Update label position to center of box
        if show_session_labels and not na(S.cur_label)
            box_left = box.get_left(S.cur_box)
            box_right = bar_index
            center_x = math.floor((box_left + box_right) / 2)
            label.set_y(S.cur_label, S.cur_hi)
            label.set_x(S.cur_label, center_x)
                
    if not t and t[1]
        S.ranges.set(0, S.cur_hi - S.cur_lo)
        S.open_to_high.set(0, S.cur_hi - S.cur_op)
        S.open_to_low.set(0, S.cur_lo - S.cur_op)

// Update session statistics

        // London fully engulfs Asia

            current_condition := "London Engulfs Asia"
        // Asia fully engulfs London

        // London partially engulfs upwards
        else if london_high > asia_high and london_low > asia_low
            current_condition := "London Partially Engulfs Upwards"
        // London partially engulfs downwards
        else if london_low < asia_low and london_high < asia_high

// Get average range for specific day of week
get_dow_stats(ses_stats S, int dow) =>
    sum = 0.0
    count = 0
    for [i, s] in S.day
        if s == dow
            sum += S.ranges.get(i)
            count += 1
    result = count > 0 ? str.tostring(math.round(sum / count, 2)) : "N/A"
    result += sample_size and count > 0 ? (" ["+str.tostring(count)+"]") : ""

// Get total average
get_total_stats(ses_stats S) =>
    avg = S.ranges.size() > 0 ? S.ranges.avg() : 0.0
    result = str.tostring(math.round(avg, 2))
    result += sample_size ? (" ["+str.tostring(S.ranges.size())+"]") : ""
