// Function to parse session time and extract hours/minutes
f_parse_session_time(session_str) =>
    // Extract time portion (e.g., '0955-1020' from '0955-1020:1234567')
    colon_pos = str.pos(session_str, ":")
    time_part = colon_pos > 0 ? str.substring(session_str, 0, colon_pos) : session_str
    // Find the hyphen position
    hyphen_pos = str.pos(time_part, "-")
    // Extract start and end times
    start_time = hyphen_pos > 0 ? str.substring(time_part, 0, hyphen_pos) : time_part
    end_time = hyphen_pos > 0 ? str.substring(time_part, hyphen_pos + 1, str.length(time_part)) : time_part
    // Extract hours and minutes
    start_hour = str.tonumber(str.substring(start_time, 0, 2))
    start_minute = str.tonumber(str.substring(start_time, 2, 4))
    end_hour = str.tonumber(str.substring(end_time, 0, 2))
    end_minute = str.tonumber(str.substring(end_time, 2, 4))
    [start_hour * 60 + start_minute, end_hour * 60 + end_minute]

// Function to determine current session
f_get_dtt_session(_hour, _minute) =>
    time_in_minutes = _hour * 60 + _minute
    // Check sessions, handling overnight sessions (e.g., sess9)
    if time_in_minutes >= sess9_start or time_in_minutes < sess9_end
        i_sess9
    else if time_in_minutes >= sess1_start and time_in_minutes < sess1_end
        i_sess1
    else if time_in_minutes >= sess2_start and time_in_minutes < sess2_end
        i_sess2
    else if time_in_minutes >= sess3_start and time_in_minutes < sess3_end
        i_sess3
    else if time_in_minutes >= sess4_start and time_in_minutes < sess4_end
        i_sess4
    else if time_in_minutes >= sess5_start and time_in_minutes < sess5_end
        i_sess5
    else if time_in_minutes >= sess6_start and time_in_minutes < sess6_end
        i_sess6
    else if time_in_minutes >= sess7_start and time_in_minutes < sess7_end
        i_sess7
    else if time_in_minutes >= sess8_start and time_in_minutes < sess8_end
        i_sess8
    else if time_in_minutes >= sess10_start and time_in_minutes < sess10_end
        i_sess10
    else if time_in_minutes >= sess11_start and time_in_minutes < sess11_end
        i_sess11
    else if time_in_minutes >= sess12_start and time_in_minutes < sess12_end
        i_sess12
    else if time_in_minutes >= sess13_start and time_in_minutes < sess13_end
        i_sess13
    else
        na

                => size.tiny

            lb := label.new(time + (_i_label_offset * timeframe.in_seconds() * 1000), open_price, label_text, xloc.bar_time, yloc.price, na, label.style_label_right, _i_line_color, _mcsize, text.align_right)

        label.delete(lb[1])

f_get_time_by_bar(bar_count) => timeframe.multiplier * bar_count * 60 * 1000
isBelowNMinutes(n) => timeframe.multiplier <= n
isDailyOrHigher(n) => (timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly) and timeframe.multiplier >= n
isHigherTF() => (timeframe.isdaily or timeframe.isweekly or timeframe.ismonthly)
isSupHigherTF() => (timeframe.isweekly or timeframe.ismonthly)

f_get_htftime_by_bar(bar_count) =>
    if timeframe.isintraday
        timeframe.multiplier * bar_count * 60 * 1000
    else
        bar_count * 24 * 60 * 60 * 1000

f_get_period(_session, _start, _lookback) =>
    result = math.max(_start, 1)
    for i = result to _lookback
        if na(_session[i+1]) and _session[i]
            result := i+1
            break
    result

f_render_main(_show, _session, _is_started, _is_ended, _color, _top, _bottom) =>
    var box my_box = na
    var label my_note = na
    var x1 = 0
    var x2 = 0
    var session_open = 0.0
    var session_high = 0.0
    var session_low = 0.0
    x0_1 = ta.valuewhen(na(_session[1]) and _session, bar_index, 1)
    x0_2 = ta.valuewhen(na(_session) and _session[1], bar_index, 0)
    x0_d = math.abs(x0_2 - x0_1)
    limit_bars = 500
    rightbars = x0_d
    if _show
        if _is_started
            x1 := bar_index
            x2 := bar_index + (math.min(x0_d, limit_bars))
            session_open := open
            session_high := _top
            session_low := _bottom
        else if _session
            true_x2 = x1 + x0_d
            rightbars := true_x2 - bar_index
            limit_bars := bar_index + 500
            x2 := math.min(true_x2, limit_bars)
            session_high := _top
            session_low := _bottom
        else if _is_ended
            session_open := na
    [x1, x2, session_open, session_high, session_low, rightbars]

draw(_show, _session, _label, _show_fib, _lookback, _boxes_session, _lines_session, _lines_fib, _labels, _innerlabels, _innerint1, _innerint2, _innerint3, _innertext1, _innertext2, _innertext3, _innerint91, _innerint92, _innerint93) =>
    max = f_get_period(_session, 1, _lookback)
    top = ta.highest(high, max)
    bottom = ta.lowest(low, max)
    is_started = f_get_started(_session)
    is_ended = f_get_ended(_session)
    // Determine color based on session label
    _color = switch _label
        "Moana" => c_moana
        "Aurora" => c_aurora
        "Rapunzel" => c_rapunzel
        "Snow White" => c_snowwhite
        "Jasmine" => c_jasmine
        "Belle" => c_belle
        "Tiana" => c_tiana
        "Elsa" => c_elsa
        "Cinderella" => c_cinderella
        "Mulan" => c_mulan
        "Ariel" => c_ariel
        "Pocahontas" => c_pocahontas
        "Merida" => c_merida
        => i_line_color  // Fallback to original color
    [x1, x2, _open, _high, _low, _rightbars] = f_render_main(_show, _session, is_started, is_ended, _color, top, bottom)
    if (i_sess_box_style == 'Box') and (i_sess_displaytf) and not (isHigherTF())
        f_render_session(_show, _session, is_started, is_ended, _color, top, bottom, x1, x2, _boxes_session)
    _mcsize = switch i_label_size
        "Tiny"   => size.tiny
        "Small"  => size.small
        "Normal" => size.normal
        => size.tiny
    if (i_label_show) and (i_sess_displaytf) and not (isHigherTF())
        f_render_label(_show, _session, is_started, _color, top, bottom, _label, _labels, _mcsize, i_label_position)

    if f_get_ended(session)
        _range = high - low
        dow = dayofweek(time, i_tz)
        switch dow
            2 => array.push(ranges_mon, _range)
            3 => array.push(ranges_tue, _range)
            4 => array.push(ranges_wed, _range)
            5 => array.push(ranges_thu, _range)
            6 => array.push(ranges_fri, _range)

// Function to format session time
f_format_session_time(session_str) =>
    // Extract the time portion (e.g., '0955-1020' from '0955-1020:1234567')
    colon_pos = str.pos(session_str, ":")
    time_part = colon_pos >= 0 ? str.substring(session_str, 0, colon_pos) : session_str
    // Find the hyphen position
    hyphen_pos = str.pos(time_part, "-")
